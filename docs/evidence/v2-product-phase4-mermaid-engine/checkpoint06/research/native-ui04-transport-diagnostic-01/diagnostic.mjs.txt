function assertCapturedPreview(dom, expected, stage) {
  const actual = dom.previews.find(
    (item) => item.from === expected.from && item.to === expected.to,
  );
  assert(
    actual && visible(actual.rect, dom),
    stage + ": same range remains visible",
  );
  assert.equal(actual.inlineSvg, 0, stage + ": no injected inline SVG");
  assert.equal(
    actual.images.length,
    expected.images.length,
    stage + ": all images remain mounted",
  );
  assert(
    actual.images.every(
      (item) =>
        item.src.startsWith("blob:") &&
        item.complete &&
        item.naturalWidth > 0 &&
        item.naturalHeight > 0,
    ),
    stage + ": same range remains fully decoded",
  );
}
function visible(rect, dom) {
  return (
    rect.bottom > Math.max(0, dom.editor.top) &&
    rect.top < Math.min(dom.height, dom.editor.bottom) &&
    rect.right > Math.max(0, dom.editor.left) &&
    rect.left < Math.min(dom.width, dom.editor.right)
  );
}

async function warmup(sample) {
  await seek(sample);
  let dom, preview;
  await app.until(
    async () => {
      dom = await observe();
      preview = dom.previews.find(
        (p) =>
          p.from === sample.from && p.to >= sample.to && p.to <= sample.to + 1,
      );
      return (
        preview?.images.length &&
        preview.images.every(
          (i) => i.complete && i.naturalWidth > 0 && i.naturalHeight > 0,
        )
      );
    },
    "warmup decoded " + sample.id,
    30000,
  );
  await app.delay(250);
  const before = await observe();
  await save("warmup-" + sample.id + "-before", { before, expected: preview });
  assertCapturedPreview(before, preview, "before screenshot");
  await app.shot("warmup-" + sample.id + "-webview");
  execFileSync(
    "import",
    ["-window", "root", path.join(output, "warmup-" + sample.id + "-x11.png")],
    { env: process.env },
  );
  const document = await app.probe(),
    after = await observe();
  await save("warmup-" + sample.id + "-after", {
    document,
    after,
    expected: preview,
  });
  assertCapturedPreview(after, preview, "after screenshot");
  await mark("warmup-" + sample.id + "-complete");
}

    },
    "unique marker search",
    30000,
  );
  await app.key("Escape");
  await app.focus(".milkup-input-proxy");
}
