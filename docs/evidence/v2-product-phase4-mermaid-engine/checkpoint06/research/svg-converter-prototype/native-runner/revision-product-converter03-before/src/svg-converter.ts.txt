    const shallow={...value};delete shallow.children;delete shallow.primary;delete shallow.fallback;delete shallow.stops;
    for(const r of references(shallow))for(let i=0;i<r.multiple;i++)expand(defMap.get(r.id),depth+1);
    for(const k of ['children','primary','fallback','stops'])if(value[k])expand(value[k],depth+1);
  }
  expand(image.children);if(image.rootStyle!==undefined)for(const r of references(styles[image.rootStyle]))expand(defMap.get(r.id));
  // Hidden-root images use their exact clipped viewport. Visible text uses live
  // glyph bounds plus typed stroke/marker/effect envelopes. Visible HTML textBox
  // overflow is rejected until complete glyph-overflow measurement is proven;
  // a foreignObject frame alone is deliberately not treated as paint bounds.
  const rasterBounds=measureSvgPaintBounds(image,paintMeasurements);
  const result:SdkDocumentSvgImage={...image,rasterBounds};
  const bytes=new TextEncoder().encode(JSON.stringify(result)).length;if(bytes>MAX_BYTES)fail('limit-exceeded','AST UTF8 bytes');
  return result;
}
