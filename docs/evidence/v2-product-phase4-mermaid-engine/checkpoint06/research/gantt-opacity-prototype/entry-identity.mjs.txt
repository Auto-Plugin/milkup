// Research only: actual original DOM retained through first SVG and PNG capture.
import mermaid from './mermaid-full-12.2.0.mjs';
import {normalizeBoundedIdentityGroups} from './normalize-identity.mjs';
const base={startOnLoad:false,securityLevel:'strict',htmlLabels:false,suppressErrorRendering:true,maxTextSize:50000,maxEdges:500,deterministicIds:true,deterministicIDSeed:'offline-audit',fontFamily:'Arial, sans-serif',legacyMathML:false,forceLegacyMathML:false,secure:['secure','securityLevel','startOnLoad','maxTextSize','maxEdges','suppressErrorRendering','htmlLabels','fontFamily','themeCSS','legacyMathML','forceLegacyMathML','look','textPlacement','theme','darkMode']};
let preparation={status:'idle'},preparing,sequence=0,task,host,request,evidence;
const encoder=new TextEncoder();
const error=e=>String(e instanceof Error?e.message:e).slice(0,4096);
function ready(){if(preparing)return preparing;preparation={status:'pending'};return preparing=Promise.resolve(document.fonts.ready).then(()=>preparation={status:'ready',result:{engine:'mermaid@12.2.0',diagnosticOnly:true}},e=>{preparation={status:'error',error:error(e)};throw e});}
function readiness(){return preparation;}
function validate(r){if(!r||typeof r!=='object'||Array.isArray(r)||Object.keys(r).some(k=>!['id','source','profile','theme'].includes(k))||typeof r.id!=='string'||!/^[A-Za-z0-9_-]{1,96}$/.test(r.id)||typeof r.source!=='string'||r.source.length>50000||r.profile!=='baseline'||r.theme!=='light')throw Error('invalid fixed request');}
function attachBytes(value){value.serializedBytes=0;value.over256KiB=false;for(let i=0;i<8;i++){const n=encoder.encode(JSON.stringify(value)).length;if(n===value.serializedBytes)break;value.serializedBytes=n;value.over256KiB=n>262144;}return value;}
async function run(r){const start=performance.now();try{host?.remove();host=document.createElement('div');host.id='svg-converter-prototype-host';host.style.width='1000px';document.body.append(host);mermaid.initialize(base);const rendered=await mermaid.render('convert_'+r.id,r.source,host);host.innerHTML=rendered.svg;request=r;evidence=undefined;return attachBytes({ok:true,id:r.id,profile:r.profile,theme:r.theme,diagramType:rendered.diagramType,elapsedMs:performance.now()-start,diagnosticOnly:true,converterSkipped:true});}catch(e){return attachBytes({ok:false,error:{reason:error(e)},diagnosticOnly:true});}}
function start(r){validate(r);if(preparation.status!=='ready'||task?.status==='pending')throw Error('not ready or busy');const taskId='gantt_'+(++sequence);task={taskId,status:'pending'};Promise.resolve().then(()=>run(r)).then(result=>{task={taskId,status:'complete',result};},e=>{task={taskId,status:'error',error:error(e)};});return {taskId,status:'pending'};}
function poll(id){return task?.taskId===id?task:{status:'unknown'};}
function geometry(root){const box=r=>[r.x,r.y,r.width,r.height],m=root.getCTM();return {rect:box(root.getBoundingClientRect()),bbox:box(root.getBBox()),ctm:m?[m.a,m.b,m.c,m.d,m.e,m.f]:null,viewBox:root.getAttribute('viewBox'),width:root.getAttribute('width'),height:root.getAttribute('height'),overflow:getComputedStyle(root).overflow};}
function probeInspect(includeSource=false){const root=host?.querySelector('svg');if(!root)throw Error('missing original or candidate');return {diagnosticOnly:true,geometry:geometry(root),requestId:request.id,evidence:evidence??null,sourceKind:evidence?'candidate-clone':'current-run-unmodified-original',...(includeSource?{svg:new XMLSerializer().serializeToString(root)}:{})};}
function probeChange(){if(task?.status==='pending')throw Error('busy');const original=host?.querySelector('svg');if(!original||evidence)throw Error('single normalization only');const taskId='normalize_'+(++sequence);task={taskId,status:'pending'};Promise.resolve().then(()=>{const candidate=original.cloneNode(true);evidence=normalizeBoundedIdentityGroups(original,candidate);if(!evidence?.groups?.length)throw Error('no eligible disjoint groups; cannot label no-op comparison as proof');original.replaceWith(candidate);return new Promise(resolve=>requestAnimationFrame(()=>requestAnimationFrame(resolve)));}).then(()=>{task={taskId,status:'complete',result:attachBytes({ok:true,diagnosticOnly:true,evidence})};},e=>{task={taskId,status:'complete',result:attachBytes({ok:false,error:{reason:error(e)},diagnosticOnly:true})};});return {taskId,status:'pending'};}
window.__milkupSvgConverterProbe=Object.freeze({ready,readiness,start,poll,probeInspect,probeChange});
