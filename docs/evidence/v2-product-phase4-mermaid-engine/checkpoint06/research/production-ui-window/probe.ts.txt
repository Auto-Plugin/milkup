 const base='/workspace/milkup-mermaid-engine-evidence/research/svg-converter-prototype/native-runner/runs/';
 const images=['00/scan_000','01/scan_002','02/scan_004'].map(p=>JSON.parse(readFileSync(base+'product-converter-fourth-'+p+'-result.json','utf8')).image);

 return result;
 }});
 for(const toLine of [23,58,73]){

 observations.push({response:{toLine,complete:response.result.complete,fragments:response.result.fragments.length,contextKnown:response.request.contextKnown}});
 }
 writeFileSync('/workspace/milkup-mermaid-engine-evidence/research/production-ui-window/budget-probe.json',JSON.stringify({scope:'Controlled public service probe with actual unchanged captured ASTs; not native request tracing',singles,observations},null,2));
}
