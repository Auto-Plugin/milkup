"""Offline full-frame pixel comparison only; no crop, threshold mask, or runtime."""
from pathlib import Path
from PIL import Image,ImageChops
import json,hashlib
R=Path(__file__).resolve().parent;run=R/'runs/paint-first01';out=R/'pixel-comparison.json'
assert not out.exists()
rows=[]
for left,right in [('original-full-canvas','canonical-full-canvas'),('original-top-left','canonical-top-left'),('original-bottom-right','canonical-bottom-right')]:
 a=Image.open(run/(left+'.png')).convert('RGBA');b=Image.open(run/(right+'.png')).convert('RGBA');assert a.size==b.size
 diff=ImageChops.difference(a,b);pixels=list(diff.getdata());count=sum(any(v) for v in pixels)
 visible=diff.convert('RGB');bbox=visible.getbbox();visible.save(run/(left+'-versus-'+right+'-diff.png'))
 rows.append({'original':left,'canonical':right,'size':a.size,'differentPixels':count,'rgbDifferenceBounds':bbox,'originalSha256':hashlib.sha256((run/(left+'.png')).read_bytes()).hexdigest(),'canonicalSha256':hashlib.sha256((run/(right+'.png')).read_bytes()).hexdigest()})
out.write_text(json.dumps({'scope':'Whole native snapshot frames; no ink crop or accepted-error mask; actual images still require review','comparisons':rows},indent=2)+'\n')
