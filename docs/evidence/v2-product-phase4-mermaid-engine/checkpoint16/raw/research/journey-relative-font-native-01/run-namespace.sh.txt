#!/usr/bin/env bash
# Research only; invoked by exact-owned display launcher after separate runtime GO.
set -euo pipefail
source /workspace/.cloud/milkup-v2/env.sh
root=/workspace/milkup-mermaid-engine-evidence/research/journey-relative-font-native-01
verified_base=/workspace/milkup-mermaid-engine-evidence/prototype
runtime=/workspace/.cloud/milkup-v2/sysroot
phase=paint
zoom=${1:?native zoom 1 or 2}
[[ "$zoom" == 1 || "$zoom" == 2 ]]
run="zoom$zoom-first01"
[[ "$run" =~ ^[a-zA-Z0-9_-]{1,64}$ ]] || { printf 'Invalid run name\n' >&2; exit 2; }
profile="$root/profiles/$run"
output="$root/runs/$run"
[[ ! -e "$profile" && ! -e "$output" ]] || { printf 'Fresh paths required\n' >&2; exit 2; }
[[ -S /tmp/.X11-unix/X128 ]] || { printf 'Dedicated X128 socket missing\n' >&2; exit 2; }
mkdir -p "$profile"/{home,config,data,cache}
uid=$(id -u)
# The supervisor remains OUTSIDE the child namespaces, owning the host PIDs,
# monotonic watchdog, process-group signals and subreaper waitpid audit.
exec /usr/bin/python3 "$root/supervise.py" --output "$output" --phase "$phase" --zoom "$zoom" -- \
  /usr/bin/bwrap \
  --die-with-parent --unshare-user --unshare-pid --unshare-ipc --unshare-uts --unshare-net \
  --clearenv \
  --ro-bind /usr /usr \
  --overlay-src /usr/bin --tmp-overlay /usr/bin \
  --ro-bind "$runtime/usr/bin/xdg-dbus-proxy" /usr/bin/xdg-dbus-proxy --remount-ro /usr/bin \
  --symlink usr/bin /bin --symlink usr/lib /lib --symlink usr/lib64 /lib64 \
  --ro-bind "$runtime/usr/lib/x86_64-linux-gnu" /runtime/lib \
  --ro-bind "$runtime/usr/share/glib-2.0/schemas" /runtime/schemas \
  --overlay-src /usr/lib/x86_64-linux-gnu --tmp-overlay /usr/lib/x86_64-linux-gnu \
  --ro-bind "$runtime/usr/lib/x86_64-linux-gnu/webkit2gtk-4.1" /usr/lib/x86_64-linux-gnu/webkit2gtk-4.1 \
  --remount-ro /usr/lib/x86_64-linux-gnu \
  --ro-bind /etc/fonts /etc/fonts \
  --ro-bind "$verified_base/narrow-config/passwd" /etc/passwd \
  --ro-bind "$verified_base/narrow-config/group" /etc/group \
  --ro-bind "$verified_base/narrow-config/hostname" /etc/hostname \
  --ro-bind "$verified_base/narrow-config/hosts" /etc/hosts \
  --ro-bind "$verified_base/narrow-config/nsswitch.conf" /etc/nsswitch.conf \
  --ro-bind "$verified_base/narrow-config/machine-id" /etc/machine-id \
  --ro-bind "$verified_base/narrow-config/loaders.cache" /probe/loaders.cache \
  --ro-bind "$root/bin/$phase" /probe/bin/helper \
  --ro-bind "$root/extensions/libpaired_resource_guard.so" /probe/extensions/libpaired_resource_guard.so \
  --bind "$profile" /profile \
  --tmpfs /tmp --dir /tmp/.X11-unix --ro-bind /tmp/.X11-unix/X128 /tmp/.X11-unix/X128 \
  --tmpfs /run --perms 0700 --dir "/run/user/$uid" \
  --dir /sys/block --dir /sys/bus --dir /sys/class --dir /sys/dev --dir /sys/devices \
  --dev /dev --proc /proc \
  --setenv PATH /usr/bin:/bin \
  --setenv HOME /profile/home \
  --setenv XDG_CONFIG_HOME /profile/config --setenv XDG_DATA_HOME /profile/data \
  --setenv XDG_CACHE_HOME /profile/cache --setenv XDG_RUNTIME_DIR "/run/user/$uid" \
  --setenv LD_LIBRARY_PATH /runtime/lib \
  --setenv WEBKIT_INJECTED_BUNDLE_PATH /runtime/lib/webkit2gtk-4.1/injected-bundle \
  --setenv GSETTINGS_SCHEMA_DIR /runtime/schemas \
  --setenv GDK_PIXBUF_MODULE_FILE /probe/loaders.cache \
  --setenv MILKUP_PROBE_EXTENSIONS /probe/extensions \
  --setenv MILKUP_PROBE_ZOOM "$zoom" \
  --setenv DISPLAY :128 --setenv GDK_BACKEND x11 --setenv LIBGL_ALWAYS_SOFTWARE 1 \
  --setenv LANG C.UTF-8 --setenv LC_ALL C.UTF-8 --setenv TZ UTC \
  --setenv NO_AT_BRIDGE 1 \
  --remount-ro / --chdir /profile -- \
  /usr/bin/dbus-run-session -- /probe/bin/helper
