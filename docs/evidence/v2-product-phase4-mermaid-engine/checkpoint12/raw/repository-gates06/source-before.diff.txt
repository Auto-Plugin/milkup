+}) => {
+  await page.goto('/')
+  const result = await page.evaluate(
+    async ({ module }) => {
+      const { DocumentPresentation } = await import(module)
+      const errors: string[] = []
+      const onError = (event: ErrorEvent) => errors.push(event.message)
+      window.addEventListener('error', onError)
+      const root = document.createElement('section')
+      Object.assign(root.style, {
+        position: 'fixed',
+        top: '50px',
+        left: '20px',
+        width: '300px',
+        padding: '0',

+      line.textContent = source
+      content.append(line)
+      root.append(content)
+      document.body.append(root)
+      const identity = {}
+      const selection = { anchor: 0, head: 2 }
+      const widths: number[] = []
+      let layouts = 0
+      const manager = new DocumentPresentation(
+        root,
+        content,
+        {
+          provide(window) {
+            widths.push(window.width)
+            return {
+              declarations: [],
+              request: {
+                documentId: 'finite-resize',
+                version: 0,
+                source: window.source,
+                from: window.from,
+                to: window.to,
+                fromLine: window.fromLine,
+                toLine: window.toLine,

+                completeEnd: window.completeEnd,

+                blocks: [],
+                theme: 'light',
+                target: 'screen',

+          },
