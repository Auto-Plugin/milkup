//! External Linux mechanism probe. No Wry fork, no Tauri bridge, no Mermaid yet.
use std::{io::{self,BufRead,Read,Write},sync::{Arc,atomic::{AtomicBool,Ordering}}};
use tao::{event::{Event,WindowEvent},event_loop::{ControlFlow,EventLoopBuilder},window::WindowBuilder};
use tao::platform::unix::WindowExtUnix;
use gtk::{prelude::*,gio,glib};
use webkit2gtk::{WebViewExt,WebContextExt,SettingsExt,PermissionRequestExt,FileChooserRequestExt,
 URISchemeRequestExt,URISchemeResponseExt,PolicyDecisionExt,NavigationPolicyDecisionExt,URIRequestExt,
 DownloadExt,SecurityManagerExt};
use javascriptcore::ValueExt;
use serde_json::{json,Value};
#[derive(Debug)]enum User { Loaded,Ready(String),Task(String),Result(String),Eof }
const URL:&str="milkup-render://job/index.html";
const CSP:&str="default-src 'none'; script-src milkup-render://job/engine.js; style-src 'unsafe-inline'; connect-src 'none'; img-src 'none'; font-src 'none'; frame-src 'none'; child-src 'none'; worker-src 'none'; object-src 'none'; base-uri 'none'; form-action 'none'";
fn emit(v:Value){let mut out=io::stdout().lock();writeln!(out,"{}",v).unwrap();out.flush().unwrap();}
fn evaluated(result:Result<javascriptcore::Value,glib::Error>)->String {
 match result {Ok(value)=>value.to_json(0).map(|v|v.to_string()).unwrap_or("null".into()),Err(error)=>json!({"nativeEvaluationError":error.to_string()}).to_string()}
}
fn main(){
 let extension=std::env::var("MILKUP_PROBE_EXTENSIONS").expect("explicit extension path required");
 let events=EventLoopBuilder::<User>::with_user_event().build();
 let window=WindowBuilder::new().with_title("Isolated DOM mechanism probe").with_inner_size(tao::dpi::LogicalSize::new(640.,200.)).with_visible(false).build(&events).unwrap();
 // Public native API: set sandbox BEFORE creating any WebView / WebProcess.
 let context=webkit2gtk::WebContext::new_ephemeral();
 context.set_sandbox_enabled(true);assert!(context.is_sandbox_enabled());
 context.set_web_extensions_directory(&extension);
 context.set_automation_allowed(false);
 context.security_manager().unwrap().register_uri_scheme_as_secure("milkup-render");
 context.register_uri_scheme("milkup-render",|request|{
  let uri=request.uri().map(|s|s.to_string()).unwrap_or_default();
  let method=request.http_method().map(|s|s.to_string()).unwrap_or_default();
  let (status,mime,body)=match(method.as_str(),uri.as_str()){
   ("GET","milkup-render://job/index.html")=>(200,"text/html; charset=utf-8",include_bytes!("../assets/index.html").as_slice()),
   ("GET","milkup-render://job/engine.js")=>(200,"text/javascript; charset=utf-8",include_bytes!("../assets/engine.js").as_slice()),
   _=>(403,"text/plain",b"denied".as_slice())};
  let bytes=glib::Bytes::from_static(body);let stream=gio::MemoryInputStream::from_bytes(&bytes);
  let response=webkit2gtk::URISchemeResponse::new(&stream,body.len() as i64);
  response.set_status(status,None);response.set_content_type(mime);
  let headers=soup::MessageHeaders::new(soup::MessageHeadersType::Response);
  headers.append("Content-Security-Policy",CSP);headers.append("X-Content-Type-Options","nosniff");headers.append("Cache-Control","no-store");
  response.set_http_headers(headers);request.finish_with_response(&response);
 });
 context.connect_download_started(|_,download|download.cancel());
 let native=webkit2gtk::WebView::with_context(&context);
 let settings=WebViewExt::settings(&native).unwrap();
 settings.set_allow_file_access_from_file_urls(false);settings.set_allow_universal_access_from_file_urls(false);
 settings.set_enable_webrtc(false);settings.set_enable_media_stream(false);settings.set_enable_webgl(false);
 settings.set_enable_html5_database(false);settings.set_enable_html5_local_storage(false);settings.set_enable_dns_prefetching(false);
 settings.set_javascript_can_open_windows_automatically(false);settings.set_enable_developer_extras(false);
 native.connect_permission_request(|_,p|{p.deny();true});native.connect_run_file_chooser(|_,p|{p.cancel();true});
 native.connect_script_dialog(|_,dialog|{dialog.close();true});
 let nav=Arc::new(AtomicBool::new(false));let nav2=nav.clone();
 native.connect_decide_policy(move|_,decision,kind|{
  match kind {
   webkit2gtk::PolicyDecisionType::NavigationAction=>{
    let allowed=decision.downcast_ref::<webkit2gtk::NavigationPolicyDecision>().and_then(|d|d.navigation_action()).and_then(|a|a.request()).and_then(|r|r.uri()).map(|u|u==URL&&!nav2.swap(true,Ordering::SeqCst)).unwrap_or(false);
    if allowed{decision.use_();}else{decision.ignore();}true
   },
   webkit2gtk::PolicyDecisionType::NewWindowAction=>{decision.ignore();true},
   _=>false
  }
 });
 let loaded=events.create_proxy();native.connect_load_changed(move|view,event|{if matches!(event,webkit2gtk::LoadEvent::Finished)&&view.uri().as_deref()==Some(URL){let _=loaded.send_event(User::Loaded);}});
 window.default_vbox().unwrap().add(&native);native.set_size_request(640,200);native.show();
 emit(json!({"type":"native-policy","sandbox":context.is_sandbox_enabled(),"ephemeral":native.is_ephemeral(),"tauriBridge":false,"wryIpcStub":false,"allowedResources":2,"settings":"file/universal/webrtc/media/webgl/storage/dns/windows/devtools denied"}));
 let input=events.create_proxy();std::thread::spawn(move||{
  let mut reader=io::stdin().lock();
  loop{let mut bytes=Vec::new();match reader.by_ref().take(130).read_until(b'\n',&mut bytes){
   Ok(0)=>break,Ok(_)if bytes.len()<=129&&bytes.last()==Some(&b'\n')=>{bytes.pop();match String::from_utf8(bytes){Ok(line)=>{let _=input.send_event(User::Task(line));},Err(_)=>break}},_=>break
  }}let _=input.send_event(User::Eof);
 });
 native.load_uri(URL);let callback=events.create_proxy();let mut ready=false;
 events.run(move|event,_,flow|{*flow=ControlFlow::Wait;match event{
  Event::UserEvent(User::Loaded)=>{let cb=callback.clone();native.evaluate_javascript("window.__fixedDomProbe.ready()",None,None,None::<&gio::Cancellable>,move|r|{let _=cb.send_event(User::Ready(evaluated(r)));});},
  Event::UserEvent(User::Ready(s))=>{let value:Value=serde_json::from_str(&s).unwrap_or(Value::Null);ready=value.get("ready")==Some(&Value::Bool(true));emit(json!({"type":"ready","value":value}));},
  Event::UserEvent(User::Task(task))=>{
   if !ready||!matches!(task.as_str(),"bbox"|"loop"|"policy"){emit(json!({"type":"rejected"}));*flow=ControlFlow::Exit;return}
   let cb=callback.clone();let script=format!("window.__fixedDomProbe.run({})",serde_json::to_string(&task).unwrap());
   native.evaluate_javascript(&script,None,None,None::<&gio::Cancellable>,move|r|{let _=cb.send_event(User::Result(evaluated(r)));});
  },
  Event::UserEvent(User::Result(s))=>{if s.len()>65536{emit(json!({"type":"oversize"}));*flow=ControlFlow::Exit;}else{emit(json!({"type":"result","value":serde_json::from_str::<Value>(&s).unwrap_or(Value::Null)}));}},
  Event::UserEvent(User::Eof)|Event::WindowEvent{event:WindowEvent::CloseRequested,..}=>*flow=ControlFlow::Exit,
  _=>{}
 }});
}
