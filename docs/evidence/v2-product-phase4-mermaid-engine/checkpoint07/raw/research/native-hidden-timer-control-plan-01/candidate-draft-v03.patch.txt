+// Fixed scalar observations only; absent/lost diagnostics are not proof of state.
+fn trace_visibility_control(
+    phase: &str,
+    container: &'static str,
+    window: &gtk::Bin,
+    view: &webkit2gtk::WebView,
+    request_id: Option<&str>,
+    state: Option<&Value>,
+) {
+    let allocation = view.allocation();
+    native_trace_details(
+        phase,
+        None,
+        request_id,
+        Some(NativeTraceDetails::VisibilityControl {
+            container,
+            container_mapped: window.is_mapped(),
+            container_realized: window.is_realized(),
+            view_mapped: view.is_mapped(),
+            view_realized: view.is_realized(),
+            view_visible: view.is_visible(),
+            allocation_width: allocation.width(),
+            allocation_height: allocation.height(),
+            document_hidden: state.and_then(|value| value["hidden"].as_bool()),
+            visibility_state: match state.and_then(|value| value["visibility"].as_str()) {
+                Some("hidden") => "hidden",
+                Some("visible") => "visible",
+                _ => "unobserved",
+            },
+            document_now_ms: state
+                .and_then(|value| value["now"].as_f64())
+                .filter(|value| value.is_finite()),
+        }),
+    );
