//! Linux process owner. The monotonic watchdog never holds pipe or GUI locks.
use super::*;
use std::{
    collections::{HashMap, HashSet},
    fs::File,
    io::{BufRead, BufReader, Read, Write},
    os::{
        fd::{AsRawFd, FromRawFd},
        unix::process::CommandExt,
    },
    process::{Child, ChildStdin, Command, Stdio},
    sync::Condvar,
    thread,
};
const INIT_DEADLINE: Duration = Duration::from_secs(10);
const RENDER_DEADLINE: Duration = Duration::from_secs(2);

pub(super) struct Instance {
    pub id: String,
    plugin_id: String,
    manifest: Value,
    child: Mutex<Child>,
    pid: i32,
    birth: u64,
    stdin: Mutex<ChildStdin>,
    guardian_stop: Option<Mutex<File>>,
    input_ready: Mutex<bool>,
    input_wake: Condvar,
    control: Mutex<Control>,
    wake: Condvar,
    owned: Mutex<HashMap<i32, u64>>,
    emit: Emit,
    extension: helper::PrivateDirectory,
}
struct Control {
    stopped: bool,
    initialized: bool,
    init_deadline: Instant,
    activation: Option<String>,
    active: Option<(String, Instant)>,
    rpc: Option<String>,
    rpc_deadline: Option<Instant>,
}
impl Instance {
    pub fn start(plugin_id: String, snapshot: Snapshot, emit: Emit) -> Result<Arc<Self>, String> {
        Self::launch(
            plugin_id,
            snapshot,
            emit,
            std::env::current_exe().map_err(|e| e.to_string())?,
        )
    }
    fn launch(
        plugin_id: String,
        snapshot: Snapshot,
        emit: Emit,
        executable: std::path::PathBuf,
    ) -> Result<Arc<Self>, String> {
        // Only the production current_exe call above reaches this method outside
        // unit tests. No executable/path option is exposed through a command.
        let id = opaque_id()?;
        let start = Instant::now();
        let extension = helper::private_extension()?;
        let mut fds = [-1; 2];
        if unsafe { libc::pipe2(fds.as_mut_ptr(), libc::O_CLOEXEC) } != 0 {
            return Err(std::io::Error::last_os_error().to_string());
        }
        let reader = unsafe { File::from_raw_fd(fds[0]) };
        let writer = unsafe { File::from_raw_fd(fds[1]) };
        let write_fd = writer.as_raw_fd();
        let mut control_fds = [-1; 2];
        if unsafe { libc::pipe2(control_fds.as_mut_ptr(), libc::O_CLOEXEC | libc::O_NONBLOCK) } != 0
        {
            return Err(std::io::Error::last_os_error().to_string());
        }
        let control_reader = unsafe { File::from_raw_fd(control_fds[0]) };
        let control_writer = unsafe { File::from_raw_fd(control_fds[1]) };
        let control_fd = control_reader.as_raw_fd();
        let mut command = Command::new(executable);
        command
            .arg("--milkup-internal-dom-helper")
            .env("MILKUP_INTERNAL_DOM_GUARD_DIR", &extension.0)
            .env("MILKUP_INTERNAL_DOM_INSTANCE_ID", &id)
            .stdin(Stdio::piped())
            .stdout(Stdio::null())
            .stderr(Stdio::inherit());
        let parent_pid = unsafe { libc::getpid() };
        unsafe {
            command.pre_exec(move || {
                if libc::setsid() < 0 {
                    return Err(std::io::Error::last_os_error());
                }
                if libc::getppid() != parent_pid {
                    return Err(std::io::Error::new(
                        std::io::ErrorKind::BrokenPipe,
                        "DOM owner exited during spawn",
                    ));
                }
                if libc::dup2(write_fd, 3) < 0 || libc::fcntl(3, libc::F_SETFD, 0) < 0 {
                    return Err(std::io::Error::last_os_error());
                }
                if libc::dup2(control_fd, 4) < 0 || libc::fcntl(4, libc::F_SETFD, 0) < 0 {
                    return Err(std::io::Error::last_os_error());
                }
                Ok(())
            });
        }
        let mut child = command.spawn().map_err(|e| e.to_string())?;
        drop(writer);
        drop(control_reader);
        let pid = child.id() as i32;
        let birth = match proc_stat(pid) {
            Some(p) => p.birth,
            None => {
                drop(control_writer);
                let _ = child.wait();
                return Err("DOM guardian exited before ownership was established".into());
            }
        };
        let stdin = match child.stdin.take() {
            Some(stdin) => stdin,
            None => {
                drop(control_writer);
                let _ = child.wait();
                return Err("DOM helper input pipe unavailable".into());
            }
        };
        let instance = Arc::new(Self {
            id,
            plugin_id,
            manifest: snapshot.manifest.clone(),
            child: Mutex::new(child),
            pid,
            birth,
            stdin: Mutex::new(stdin),
            guardian_stop: Some(Mutex::new(control_writer)),
            input_ready: Mutex::new(false),
            input_wake: Condvar::new(),
            control: Mutex::new(Control {
                stopped: false,
                initialized: false,
                init_deadline: start + INIT_DEADLINE,
                activation: None,
                active: None,
                rpc: None,
                rpc_deadline: None,
            }),
            wake: Condvar::new(),
            owned: Mutex::new(HashMap::from([(pid, birth)])),
            emit,
            extension,
        });
        // Arm the owner before reading/writing anything to the child.
        let watcher = instance.clone();
        thread::spawn(move || watcher.watch());
        let sampler = instance.clone();
        thread::spawn(move || {
            while sampler.control.lock().is_ok_and(|c| !c.stopped) {
                sampler.sample_owned();
                thread::sleep(Duration::from_millis(25));
            }
        });
        let output = instance.clone();
        thread::spawn(move || {
            let mut reader = BufReader::new(reader);
            loop {
                match read_frame(&mut reader, OUTPUT_WIRE_LIMIT) {
                    Ok(Some(bytes)) => match serde_json::from_slice(&bytes) {
                        Ok(message) => output.receive(message),
                        Err(_) => {
                            output.fail("DOM helper emitted invalid JSON".into());
                            break;
                        }
                    },
                    Ok(None) => {
                        output.fail("DOM helper result pipe closed".into());
                        break;
                    }
                    Err(e) => {
                        output.fail(e);
                        break;
                    }
                }
            }
        });
        let input = instance.clone();
        thread::spawn(move || {
            // Binary length prefix avoids JSON expanding an 8 MiB UTF-8 module
            // to an unbounded preparation frame. This is trusted snapshot data.
            let result = (|| {
                let bytes = serde_json::to_vec(&snapshot).map_err(|e| e.to_string())?;
                let mut pipe = input.stdin.lock().map_err(|e| e.to_string())?;
                pipe.write_all(&(bytes.len() as u32).to_le_bytes())
                    .and_then(|_| pipe.write_all(&bytes))
                    .and_then(|_| pipe.flush())
                    .map_err(|e| e.to_string())
            })();
            if let Ok(mut ready) = input.input_ready.lock() {
                *ready = true;
                input.input_wake.notify_all();
            }
            if let Err(error) = result {
                input.fail(error);
            }
        });
        Ok(instance)
    }
    pub fn send(&self, message: Value) -> Result<(), String> {
        let bytes = serde_json::to_vec(&message).map_err(|e| e.to_string())?;
        if bytes.len() > INPUT_WIRE_LIMIT {
            return Err("DOM input frame exceeds its bounded wire budget".into());
        }
        {
            let mut control = self.control.lock().map_err(|e| e.to_string())?;
            if control.stopped {
                return Err("DOM instance was revoked".into());
            }
            let id = valid_id(&message)
                .ok_or("Invalid DOM request ID")?
                .to_string();
            match (message["protocol"].as_str(), message["type"].as_str()) {
                (Some(RENDER_PROTOCOL), Some("render")) => {
                    if !control.initialized {
                        return Err("DOM instance is not activated".into());
                    }
                    if control.active.is_some() || control.rpc.is_some() {
                        return Err("DOM renderer is busy".into());
                    }
                    validate_render(&message, &self.plugin_id, &self.manifest)?;
                    // This dispatch is authoritative, including serialization in
                    // the helper, layout, and result transfer back to the owner.
                    native_trace("dispatch", Some(&self.id), Some(&id));
                    control.active = Some((id, Instant::now() + RENDER_DEADLINE));
                    self.wake.notify_all();
                }
                (Some(RENDER_PROTOCOL), Some("cancel")) => {
                    if !keys(&message, &["protocol", "type", "id"]) {
                        return Err("Invalid DOM cancellation".into());
                    }
                    // Cancellation neither settles nor resets the active deadline.
                    if !control
                        .active
                        .as_ref()
                        .is_some_and(|(active, _)| active == &id)
                    {
                        return Ok(());
                    }
                }
                (Some(RPC_PROTOCOL), Some("request")) => {
                    if !keys(&message, &["protocol", "type", "id", "method", "payload"])
                        || message["payload"]["pluginId"] != self.plugin_id
                    {
                        return Err("Invalid DOM lifecycle request".into());
                    }
                    if control.rpc.is_some() || control.active.is_some() {
                        return Err("DOM instance is busy".into());
                    }
                    match message["method"].as_str() {
                        Some("activate")
                            if !control.initialized && control.activation.is_none() =>
                        {
                            if approved_identity(&message["payload"]["manifest"])
                                != approved_identity(&self.manifest)
                                || message["payload"]["permissions"] != json!(["document:read"])
                            {
                                return Err(
                                    "DOM activation does not match its approved snapshot".into()
                                );
                            }
                            control.activation = Some(id.clone());
                        }
                        Some("deactivate" | "dispose") if control.initialized => {}
                        _ => return Err("Unsupported DOM lifecycle method".into()),
                    }
                    control.rpc = Some(id);
                    control.rpc_deadline = Some(if control.initialized {
                        Instant::now() + INIT_DEADLINE
                    } else {
                        control.init_deadline
                    });
                    self.wake.notify_all();
                }
                _ => return Err("Unsupported DOM message".into()),
            }
        }
        // Diagnostic-only work starts after the original active deadline was armed.
        let diagnostic_message =
            native_diagnostics_enabled().then(|| DiagnosticMessage::from_value(&message));
        if let Some(frame) = &diagnostic_message {
            if frame.protocol == "document-render" && frame.kind == "render" {
                let request = &message["payload"]["request"];
                if let Some(source) = request["source"].as_str() {
                    native_trace_details(
                        "request-meta",
                        Some(&self.id),
                        frame.id.as_deref(),
                        Some(NativeTraceDetails::Request {
                            from: request["from"].as_u64(),
                            to: request["to"].as_u64(),
                            source_bytes: source.len(),
                            source_sha256: format!("{:x}", Sha256::digest(source.as_bytes())),
                            width: request["width"].as_f64().filter(|v| v.is_finite()),
                            theme: match request["theme"].as_str() {
                                Some("light") => "light",
                                Some("dark") => "dark",
                                _ => "unknown",
                            },
                        }),
                    );
                }
            }
        }
        let result = (|| {
            let mut ready = self.input_ready.lock().map_err(|e| e.to_string())?;
            while !*ready {
                if self.control.lock().map_err(|e| e.to_string())?.stopped {
                    return Err("DOM instance was revoked".into());
                }
                ready = self
                    .input_wake
                    .wait_timeout(ready, Duration::from_millis(50))
                    .map_err(|e| e.to_string())?
                    .0;
            }
            drop(ready);
            let mut pipe = self.stdin.lock().map_err(|e| e.to_string())?;
            if self.control.lock().map_err(|e| e.to_string())?.stopped {
                return Err("DOM instance was revoked".into());
            }
            pipe.write_all(&bytes)
                .and_then(|_| pipe.write_all(b"\n"))
                .and_then(|_| pipe.flush())
                .map_err(|e| e.to_string())
        })();
        if result.is_ok() {
            if let Some(frame) = &diagnostic_message {
                native_trace_details(
                    "pipe-write-complete",
                    Some(&self.id),
                    frame.id.as_deref(),
                    Some(NativeTraceDetails::Frame { message: frame }),
                );
            }
        }
        if let Err(error) = &result {
            self.fail(error.clone());
        }
        result
    }
    fn receive(&self, message: Value) {
        // Entry is after stdout frame JSON parsing, not first pipe-byte arrival.
        // Preserve this observation even when the original deadline branch rejects it.
        let diagnostic_entry = native_diagnostic_time();
        let diagnostic_message = diagnostic_entry.map(|_| DiagnosticMessage::from_value(&message));
        if let (Some(entry), Some(frame)) = (diagnostic_entry, &diagnostic_message) {
            native_trace_details(
                "receive-entry",
                Some(&self.id),
                frame.id.as_deref(),
                Some(NativeTraceDetails::Received {
                    message: frame,
                    entry_monotonic_ns: entry,
                    ok: message["ok"].as_bool(),
                }),
            );
        }
        let mut diagnostic_settled = false;
        // All child parsing and payload sizing precede the final monotonic
        // settlement decision; no expensive serialization owns control.
        let oversized = message
            .get("value")
            .is_some_and(|v| serde_json::to_vec(v).map_or(true, |b| b.len() > 262144));
        let mut control = match self.control.lock() {
            Ok(c) => c,
            Err(_) => return,
        };
        if control.stopped {
            return;
        }
        let now = Instant::now();
        if !control.initialized && now >= control.init_deadline {
            drop(control);
            self.fail("DOM initialization exceeded 10 seconds".into());
            return;
        }
        if let Some((id, deadline)) = &control.active {
            if now >= *deadline {
                let id = id.clone();
                control.stopped = true;
                control.active = None;
                drop(control);
                self.finish(Some(id), None);
                return;
            }
            if !is_result(&message, id) {
                return;
            }
            if let (Some(entry), Some(frame)) = (diagnostic_entry, &diagnostic_message) {
                native_trace_details(
                    "matching-result-received",
                    Some(&self.id),
                    frame.id.as_deref(),
                    Some(NativeTraceDetails::Received {
                        message: frame,
                        entry_monotonic_ns: entry,
                        ok: message["ok"].as_bool(),
                    }),
                );
            }
            if oversized {
                drop(control);
                self.fail("DOM output exceeds the public payload budget".into());
                return;
            }
            if Instant::now() >= *deadline {
                let id = id.clone();
                control.stopped = true;
                control.active = None;
                drop(control);
                self.finish(Some(id), None);
                return;
            }
            control.active = None;
            diagnostic_settled = true;
        } else if let Some(id) = &control.rpc {
            if !is_rpc_result(&message, id) {
                return;
            }
            if control.rpc_deadline.is_some_and(|d| Instant::now() >= d) {
                drop(control);
                self.fail("DOM lifecycle deadline exceeded".into());
                return;
            }
            if control.activation.as_ref() == Some(id) {
                if message["ok"] != true {
                    drop(control);
                    self.fail("DOM plugin activation failed".into());
                    return;
                }
                control.initialized = true;
            }
            control.rpc = None;
            control.rpc_deadline = None;
        } else {
            return;
        }
        self.wake.notify_all();
        drop(control);
        if diagnostic_settled {
            if let Some(frame) = &diagnostic_message {
                native_trace_details(
                    "result-settled",
                    Some(&self.id),
                    frame.id.as_deref(),
                    Some(NativeTraceDetails::Settled {
                        message: frame,
                        ok: message["ok"].as_bool(),
                    }),
                );
            }
        }
        (self.emit)(self.event("message", Some(message), None, None));
    }
    fn watch(&self) {
        loop {
            let mut control = match self.control.lock() {
                Ok(c) => c,
                Err(_) => return,
            };
            if control.stopped {
                return;
            }
            let deadline = if !control.initialized {
                Some(control.init_deadline)
            } else {
                control
                    .active
                    .as_ref()
                    .map(|(_, d)| *d)
                    .or(control.rpc_deadline)
            };
            if deadline.is_some_and(|d| Instant::now() >= d) {
                let request_id = control.active.take().map(|(id, _)| id);
                control.stopped = true;
                drop(control);
                let error = if request_id.is_none() {
                    Some("DOM initialization or lifecycle exceeded 10 seconds".into())
                } else {
                    None
                };
                self.finish(request_id, error);
                return;
            }
            let remaining = deadline
                .map(|d| d.saturating_duration_since(Instant::now()))
                .unwrap_or(Duration::from_millis(25));
            // /proc sampling uses a separate owner thread, never this watchdog.
            let wait = remaining.min(Duration::from_millis(25));
            if let Ok((next, _)) = self.wake.wait_timeout(control, wait) {
                control = next;
            } else {
                return;
            }
            drop(control);
        }
    }
    fn event(
        &self,
        kind: &'static str,
        message: Option<Value>,
        request_id: Option<String>,
        error: Option<String>,
    ) -> DomEvent {
        DomEvent {
            plugin_id: self.plugin_id.clone(),
            instance_id: self.id.clone(),
            kind,
            message,
            request_id,
            error,
        }
    }
    fn finish(&self, request_id: Option<String>, error: Option<String>) {
        if request_id.is_some() {
            native_trace("deadline-observed", Some(&self.id), request_id.as_deref());
        }
        self.wake.notify_all();
        let cleanup = self.kill_owned();
        let error = match cleanup {
            Ok(()) => error,
            Err(e) => Some(format!(
                "{}; {e}",
                error.unwrap_or_else(|| "DOM context exceeded its deadline".into())
            )),
        };
        // Cleanup precedes trusted control delivery. OS scheduling can delay
        // this observation; there is no zero scheduling latency claim.
        let kind = if error.is_some() {
            "error"
        } else {
            "document-render-timeout"
        };
        native_trace("control-emitted", Some(&self.id), request_id.as_deref());
        (self.emit)(self.event(kind, None, request_id, error));
    }
    fn fail(&self, error: String) {
        let mut control = match self.control.lock() {
            Ok(c) => c,
            Err(_) => return,
        };
        if control.stopped {
            return;
        }
        control.stopped = true;
        control.active = None;
        drop(control);
        self.finish(None, Some(error));
    }
    pub fn stop(&self) -> Result<(), String> {
        {
            let mut control = self.control.lock().map_err(|e| e.to_string())?;
            control.stopped = true;
            control.active = None;
        }
        self.wake.notify_all();
        self.kill_owned()
    }
    fn sample_owned(&self) {
        let table = proc_table();
        let Ok(mut owned) = self.owned.lock() else {
            return;
        };
        let leader_matches = table.get(&self.pid).is_some_and(|p| p.birth == self.birth);
        let mut selected: HashSet<_> = table
            .iter()
            .filter(|(pid, p)| {
                owned.get(pid) == Some(&p.birth) || (leader_matches && p.group == self.pid)
            })
            .map(|(pid, _)| *pid)
            .collect();
        loop {
            let additional: Vec<_> = table
                .iter()
                .filter(|(pid, p)| !selected.contains(pid) && selected.contains(&p.parent))
                .map(|(pid, _)| *pid)
                .collect();
            if additional.is_empty() {
                break;
            }
            selected.extend(additional);
        }
        for pid in selected {
            if let Some(p) = table.get(&pid) {
                owned.insert(pid, p.birth);
            }
        }
    }
    fn kill_owned(&self) -> Result<(), String> {
        // Serialize cleanup/reaping only, never request writes or watchdog state.
        let mut child = self.child.lock().map_err(|e| e.to_string())?;
        if let Some(channel) = &self.guardian_stop {
            // Never SIGKILL the guardian ahead of its cleanup. The dedicated
            // nonblocking pipe cannot be held up by renderer input or its GUI.
            if child.try_wait().map_err(|e| e.to_string())?.is_none() {
                let channel = channel.lock().map_err(|e| e.to_string())?;
                let byte = [1u8];
                let result = unsafe { libc::write(channel.as_raw_fd(), byte.as_ptr().cast(), 1) };
                if result < 0
                    && ![Some(libc::EAGAIN), Some(libc::EPIPE)]
                        .contains(&std::io::Error::last_os_error().raw_os_error())
                {
                    return Err(std::io::Error::last_os_error().to_string());
                }
            }
            let until = Instant::now() + Duration::from_secs(4);
            loop {
                if let Some(status) = child.try_wait().map_err(|e| e.to_string())? {
                    if !status.success() {
                        return Err(format!(
                            "DOM guardian exited before confirming descendant cleanup: {status}"
                        ));
                    }
                    self.extension.cleanup();
                    native_trace("cleanup-confirmed", Some(&self.id), None);
                    return Ok(());
                }
                if Instant::now() >= until {
                    return Err("DOM guardian has not confirmed descendant cleanup; it remains alive to finish cleanup".into());
                }
                thread::sleep(Duration::from_millis(5));
            }
        }
        // Test-only standalone process fixtures use the exact identity cleanup
        // path below. Production instances always have a dedicated guardian.
        // The unreaped leader reserves its PID/process group. SIGKILL first;
        // audit and escaped-child signals follow, never a graceful wait first.
        if proc_stat(self.pid).is_some_and(|p| p.birth == self.birth) {
            unsafe {
                libc::kill(-self.pid, libc::SIGKILL);
            }
        }
        self.sample_owned();
        let until = Instant::now() + Duration::from_secs(3);
        loop {
            self.sample_owned();
            let owned = self.owned.lock().map_err(|e| e.to_string())?.clone();
            let mut alive = Vec::new();
            for (pid, birth) in owned {
                if let Some(stat) = proc_stat(pid).filter(|s| s.birth == birth) {
                    signal_exact(pid, birth)?;
                    if pid != self.pid {
                        unsafe {
                            libc::waitpid(pid, std::ptr::null_mut(), libc::WNOHANG);
                        }
                    }
                    if stat.state != 'Z' || proc_stat(pid).is_some() {
                        alive.push(pid);
                    }
                }
            }
            let _ = child.try_wait().map_err(|e| e.to_string())?;
            if alive.is_empty() {
                self.extension.cleanup();
                return Ok(());
            }
            if Instant::now() >= until {
                return Err(format!("DOM owned descendants did not clear: {alive:?}"));
            }
            thread::sleep(Duration::from_millis(10));
        }
    }
}
pub(super) fn signal_exact(pid: i32, birth: u64) -> Result<(), String> {
    // A pidfd fixes identity across the check-to-signal interval.
    let fd = unsafe { libc::syscall(libc::SYS_pidfd_open, pid, 0) as i32 };
    if fd < 0 {
        if std::io::Error::last_os_error().raw_os_error() == Some(libc::ESRCH) {
            return Ok(());
        }
        return Err(format!(
            "Cannot acquire exact DOM process identity: {}",
            std::io::Error::last_os_error()
        ));
    }
    let file = unsafe { File::from_raw_fd(fd) };
    if proc_stat(pid).is_some_and(|p| p.birth == birth) {
        let result = unsafe {
            libc::syscall(
                libc::SYS_pidfd_send_signal,
                file.as_raw_fd(),
                libc::SIGKILL,
                std::ptr::null::<libc::siginfo_t>(),
                0,
            )
        };
        if result < 0 && std::io::Error::last_os_error().raw_os_error() != Some(libc::ESRCH) {
            return Err(std::io::Error::last_os_error().to_string());
        }
    }
    Ok(())
}
pub(super) struct ProcStat {
    pub(super) parent: i32,
    group: i32,
    pub(super) birth: u64,
    state: char,
}
fn proc_stat(pid: i32) -> Option<ProcStat> {
    let text = std::fs::read_to_string(format!("/proc/{pid}/stat")).ok()?;
    let fields: Vec<_> = text.rsplit_once(')')?.1.split_whitespace().collect();
    Some(ProcStat {
        state: fields.first()?.chars().next()?,
        parent: fields.get(1)?.parse().ok()?,
        group: fields.get(2)?.parse().ok()?,
        birth: fields.get(19)?.parse().ok()?,
    })
}
pub(super) fn proc_table() -> HashMap<i32, ProcStat> {
    std::fs::read_dir("/proc")
        .into_iter()
        .flatten()
        .filter_map(|e| e.ok())
        .filter_map(|e| e.file_name().to_str()?.parse::<i32>().ok())
        .filter_map(|pid| proc_stat(pid).map(|p| (pid, p)))
        .collect()
}
pub(super) fn read_frame(
    reader: &mut impl BufRead,
    limit: usize,
) -> Result<Option<Vec<u8>>, String> {
    let mut bytes = Vec::new();
    let n = reader
        .take((limit + 2) as u64)
        .read_until(b'\n', &mut bytes)
        .map_err(|e| e.to_string())?;
    if n == 0 {
        return Ok(None);
    }
    if bytes.last() != Some(&b'\n') || bytes.len() > limit + 1 {
        return Err("DOM result pipe exceeded its bounded frame budget".into());
    }
    bytes.pop();
    Ok(Some(bytes))
}
fn keys(value: &Value, allowed: &[&str]) -> bool {
    value
        .as_object()
        .is_some_and(|o| o.keys().all(|k| allowed.contains(&k.as_str())))
}
fn valid_id(message: &Value) -> Option<&str> {
    message["id"].as_str().filter(|id| {
        !id.is_empty()
            && id.len() <= 80
            && id
                .bytes()
                .all(|b| b.is_ascii_alphanumeric() || b == b'_' || b == b'-')
    })
}
fn is_result(message: &Value, id: &str) -> bool {
    if !keys(message, &["protocol", "type", "id", "ok", "value", "error"])
        || message["protocol"] != RENDER_PROTOCOL
        || message["type"] != "result"
        || valid_id(message) != Some(id)
    {
        return false;
    }
    match message["ok"].as_bool() {
        Some(true) => message.get("value").is_some() && message.get("error").is_none(),
        Some(false) => {
            message.get("value").is_none()
                && message["error"].as_str().is_some_and(|e| {
                    [
                        "invalid-input",
                        "invalid-output",
                        "revoked",
                        "aborted",
                        "busy",
                        "timeout",
                        "unavailable",
                    ]
                    .contains(&e)
                })
        }
        None => false,
    }
}
fn is_rpc_result(message: &Value, id: &str) -> bool {
    if !keys(message, &["protocol", "type", "id", "ok", "value", "error"])
        || message["protocol"] != RPC_PROTOCOL
        || message["type"] != "response"
        || valid_id(message) != Some(id)
    {
        return false;
    }
    match message["ok"].as_bool() {
        Some(true) => message.get("value").is_some() && message.get("error").is_none(),
        Some(false) => {
            message.get("value").is_none()
                && keys(&message["error"], &["message"])
                && message["error"]["message"]
                    .as_str()
                    .is_some_and(|s| s.len() <= 4096)
        }
        None => false,
    }
}
fn validate_render(message: &Value, plugin_id: &str, manifest: &Value) -> Result<(), String> {
    let payload = &message["payload"];
    let source = payload["request"]["source"]
        .as_str()
        .ok_or("DOM render source must be UTF-8 text")?;
    if !keys(message, &["protocol", "type", "id", "payload"])
        || !keys(payload, &["pluginId", "rendererId", "request"])
        || payload["pluginId"] != plugin_id
        || !manifest["contributes"]["documentRenderers"]
            .as_array()
            .is_some_and(|a| a.iter().any(|r| r["id"] == payload["rendererId"]))
        || source.encode_utf16().count() > 65536
        || source.split('\n').count() > 256
    {
        return Err("Invalid bounded DOM document render request".into());
    }
    Ok(())
}
