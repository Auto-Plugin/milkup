function assertCapturedPreview(dom, expected, stage) {
  const actual = dom.previews.find((item) => item.from === expected.from && item.to === expected.to)
  assert(actual && visible(actual.rect, dom), stage + ': same range remains visible')
  assert.equal(actual.inlineSvg, 0, stage + ': no injected inline SVG')
  assert.equal(actual.images.length, expected.images.length, stage + ': all images remain mounted')
  assert(
    actual.images.every(
      (item) =>
        item.src.startsWith('blob:') &&
        item.complete &&
        item.naturalWidth > 0 &&
        item.naturalHeight > 0,
    ),
    stage + ': same range remains fully decoded',
  )
}
function noticeIsVisible(notice, dom) {
  const rect = notice.rect
  return (
    notice.display !== 'none' &&
    notice.visibility === 'visible' &&
    notice.opacity > 0 &&
    rect.width > 0 &&
    rect.height > 0 &&
    rect.top >= Math.max(0, dom.editor.top) &&
    rect.bottom <= Math.min(dom.height, dom.editor.bottom) &&
    rect.left >= Math.max(0, dom.editor.left) &&
    rect.right <= Math.min(dom.width, dom.editor.right)
  )
}
function assertCapturedNotice(dom, expected, stage) {
  assert(
    dom.noticeBoxes.some((notice) => notice.text === expected && noticeIsVisible(notice, dom)),
    stage + ': the current isolated fallback notice is fully visible',
  )
  assert.equal(dom.previews.length, 0, stage + ': single fallback remains full source')
}
let assertionFailureSequence = 0
async function preserveAssertion(kind, dom, expected, stage, document = null) {
  try {
    if (kind === 'preview') assertCapturedPreview(dom, expected, stage)
    else assertCapturedNotice(dom, expected, stage)
  } catch (error) {
    const file =
      'assertion-failure-' +
      String(++assertionFailureSequence).padStart(3, '0') +
      '-' +
      kind +
      '-' +
      stage.replaceAll(' ', '-') +
      '.json'
    // Preserve the observations already made by the failing check. A before-shot
    // check has no document probe; do not add one and change the observed timing.
    await writeFile(
      path.join(output, file),
      JSON.stringify(
        {
          recordedAt: new Date().toISOString(),
          kind,
          stage,
          document,
          documentBoundary:
            document === null
              ? 'Not read at this stage; no extra browser observation was made.'
              : 'Already-read public document probe from this record call.',
          dom,
          expected,
          error: {
            message: String(error),
            stack: error.stack,
            actual: error.actual,
            actualIsUndefined: error.actual === undefined,
            expected: error.expected,
            operator: error.operator,
          },
        },
        null,
        2,
      ) + '\n',
      { flag: 'wx' },
    ).catch((writeError) => {
      console.error('Failed to preserve assertion snapshot ' + file + ': ' + String(writeError))
    })
    throw error
  }
}
async function record(name, extra = {}) {
  const document = await app.probe()
  const dom = await observe()
  if (extra.preview)
    await preserveAssertion('preview', dom, extra.preview, 'after screenshot', document)
  if (extra.fallbackNotice) {
    assert.equal(document.path, activeFile)
    await preserveAssertion('notice', dom, extra.fallbackNotice, 'after screenshot', document)
  }
  const currentRegistry = await registry()
  const audit = await diagramsAudit()
  if (extra.coldBundled) {
    const provider = currentRegistry.find((row) => row.id === 'milkup.diagrams')
    assert(provider?.sourcePath.startsWith('bundled:'))
    assert.equal(provider.state, 'enabled')
    assert.equal(provider.enabled, true)
    assert.deepEqual(provider.errors, [])
    assert(
      !audit.some((item) => item.denied && /execution deadline/i.test(item.detail || '')),
      'Cold bundled instance must not have a deadline-denied audit',
    )
  }
  results.push({
    name,
    at: new Date().toISOString(),
    ...extra,
    document,
    dom,
    registry: currentRegistry,
    diagramsAudit: audit,
  })
  await writeFile(journal, JSON.stringify(results, null, 2) + '\n')
  console.log('PASS ' + name)
}
async function shot(name, extra = {}) {
  await app.delay(250)
  if (extra.preview)
    await preserveAssertion('preview', await observe(), extra.preview, 'before screenshot')
  if (extra.fallbackNotice)
    await preserveAssertion('notice', await observe(), extra.fallbackNotice, 'before screenshot')
  await app.shot(name + '-webview')
  execFileSync('import', ['-window', 'root', path.join(output, name + '-x11.png')], {
    env: process.env,
  })
  const screenshots = await Promise.all(
    ['webview', 'x11'].map(async (kind) => {
      const file = name + '-' + kind + '.png',
        bytes = await readFile(path.join(output, file))
      return { file, bytes: bytes.length, sha256: sha256(bytes), reviewed: false }
    }),
  )
  await record(name, { ...extra, screenshots })
}

async function assertSample(sample, mode) {
  if (mode === 'source' || sample.expected === 'fallback') {
    if (mode === 'live')
      await app.until(
        async () => (await observe()).notices.length > 0,
        'explicit full-source fallback',
        30_000,
      )
    const dom = await observe()
    assert(
      !dom.previews.some((p) => p.from < sample.to && p.to > sample.from),
      'Literal/fallback range is never replaced',
    )
    const lines = await app.execute(
      'return [...document.querySelectorAll(".milkup-line")].filter(e=>Number(e.dataset.from)<arguments[1]&&Number(e.dataset.to)>=arguments[0]).map(e=>({from:Number(e.dataset.from),to:Number(e.dataset.to),text:e.textContent,rect:e.getBoundingClientRect().toJSON()}))',
      sample.from,
      sample.to,
    )
    assert(
      lines.some((line) => line.text.includes('```mermaid')),
      'Mapped opening fence remains present',
    )
    // Windowed source need not mount the whole long fence at once; exact file
    // bytes are asserted separately, and the visible source window is recorded.
    return { expected: sample.expected, sourceLines: lines, pixelReview: 'pending' }
  }
  let dom, preview
  await app.until(
    async () => {
      dom = await observe()
      preview = dom.previews.find(
        (p) => p.from === sample.from && p.to >= sample.to && p.to <= sample.to + 1,
      )
      return (
        preview?.images.length &&
        preview.images.every((i) => i.complete && i.naturalWidth > 0 && i.naturalHeight > 0)
      )
    },
    'actual image decode ' + sample.id,
    30_000,
  )
  assert(visible(preview.rect, dom), 'Preview must intersect the actual native viewport')
  assert.equal(preview.inlineSvg, 0)
  assert(preview.label)
  for (const image of preview.images) {
    assert(image.src.startsWith('blob:'))
    assert.equal(image.draggable, false)
    assert(image.tabIndex < 0)
    assert.equal(image.slotOverflow, 'visible')
    assert(image.rect.width > 0 && image.rect.height > 0)
  }
  return { preview, pixelReview: 'pending' }
}

async function isolatedFallback(sample) {
  await openDocument(sample.name)
  await seek(sample, 'source')
  await app.until(
    async () => (await observe()).notices.length === 0,
    'source mode clears any previous document notice',
  )
  const before = await app.probe()
  await seek(sample, 'live')
  const result = await assertSample(sample, 'live')
  const dom = await observe()
  assert.equal(dom.previews.length, 0, 'Single-fence fallback has no replacement anywhere')
  assert(
    dom.notices.some((notice) => notice.trim().length > 0),
    'Explicit fresh fallback notice',
  )
  const text = contents.get(sample.name)
  const lines = result.sourceLines.sort((a, b) => a.from - b.from)
  assert.equal(lines[0]?.from, sample.from)
  assert.equal(lines.at(-1)?.to, sample.to)
  for (let index = 0; index < lines.length; index++) {
    const line = lines[index]
    assert.equal(line.text, text.slice(line.from, line.to), 'Entire fallback source line retained')
    if (index > 0) assert.equal(line.from, lines[index - 1].to + 1, 'No omitted source lines')
  }
  assert.equal((await app.probe()).version, before.version)
  await saved(text)
  await shot('isolated-' + sample.name, {
    sample: sample.id,
    ...result,
    isolatedSingleFence: true,
    completeSourceLines: true,
    byteExact: true,
    notices: dom.notices,
  })
  await app.focus('.milkup-input-proxy')
  await app.key('ctrl+Home')
  await app.until(async () => (await app.probe()).cursor === 0, 'native Home to document start')
  const home = await observe()
  const px = Math.min(home.width, home.editor.right) - 30
  const py = Math.max(0, home.editor.top) + 40
  assert(
    await app.execute(
      'return document.querySelector(".milkup-editor")?.contains(document.elementFromPoint(arguments[0],arguments[1]))',
      px,
      py,
    ),
    'Native upward wheel targets the document editor',
  )
  const scale =
    Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', nativeWindow))[1]) / home.width
  x(
    'mousemove',
    '--window',
    nativeWindow,
    String(Math.round(px * scale)),
    String(Math.round(py * scale)),
    'click',
    '--repeat',
    '6',
    '--delay',
    '70',
    '4',
  )
  const message = dom.notices.find((notice) => notice.trim().length > 0)
  await app.until(async () => {
    const current = await observe()
    return current.noticeBoxes.some(
      (notice) => notice.text === message && noticeIsVisible(notice, current),
    )
  }, 'current single-file fallback notice is fully visible after native navigation')
  assert.equal((await app.probe()).version, before.version)
  assert.deepEqual(await readFile(activeFile), Buffer.from(text))
  await shot('notice-' + sample.name, {
    sample: sample.id,
    isolatedSingleFence: true,
    fallbackNotice: message,
    navigation: ['native ctrl+Home', 'native wheel button4 six times'],
    byteExact: true,
  })
}

async function history(name, mode) {
  await openDocument(name)
  await app.focus('.milkup-input-proxy')
  await app.key(mode === 'source' ? 'ctrl+1' : 'ctrl+2')
  await app.until(async () => (await app.probe()).mode === mode, 'history mode ' + mode)
  await app.key('ctrl+Home')
  await app.until(async () => (await app.probe()).cursor === 0, 'document start')
  const original = contents.get(name),
    insert = 'EDIT-中文é🙂-'
  const release = await ownNativeClipboard(insert)
  try {
    await app.key('ctrl+v')
    await app.until(async () => (await app.probe()).dirty, 'native edit')
  } finally {
    release()
  }

  await app.key('ctrl+z')

async function scrollEdges(sample, key) {
  const read = async () => (await observe()).previews.find((p) => p.from === sample.from)
  const before = await read()
  const dom = await observe(),
    rect = before.rect
  const px =
    (Math.max(rect.left, dom.editor.left) + Math.min(rect.right, dom.editor.right, dom.width)) / 2
  const py =
    (Math.max(rect.top, dom.editor.top) + Math.min(rect.bottom, dom.editor.bottom, dom.height)) / 2
  assert(
    await app.execute(
      'return document.querySelector(arguments[0])?.contains(document.elementFromPoint(arguments[1],arguments[2]))',
      selectorFor(sample),
      px,
      py,
    ),
    'Real wheel target hit test',
  )
  const scale =
    Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', nativeWindow))[1]) / dom.width
  x(
    'mousemove',
    '--window',
    nativeWindow,
    String(Math.round(px * scale)),
    String(Math.round(py * scale)),
  )
  for (const [corner, axes] of [
    [
      'bottom-right',
      [
        ['5', 'top', before.scrollHeight - before.clientHeight],
        ['7', 'left', before.scrollWidth - before.clientWidth],
      ],
    ],
    [
      'top-left',
      [
        ['4', 'top', 0],
        ['6', 'left', 0],
      ],
    ],
  ]) {
    for (const [button, axis, target] of axes) {
      for (let step = 0; step < 30 && Math.abs((await read())[axis] - target) > 1; step++) {
        x('click', '--repeat', '3', '--delay', '70', button)
        await app.delay(120)
      }
      assert(
        Math.abs((await read())[axis] - target) <= 1,
        'Wheel reaches diagram ' + axis + ' ' + corner,
      )
    }
    await shot(`edges-${key}-${sample.id}-${corner}`, {
      sample: sample.id,
      before,
      after: await read(),
    })
  }
}

// Frozen sampler CLI adapter; execution still requires root runtime approval.
// Verify the sampler before App startup; never pause the product to await sampling.
assert(process.env.MILKUP_COLD_CPU_SAMPLER, 'A reviewed sampler path is required')
assert.match(process.env.MILKUP_COLD_CPU_SAMPLER_SHA256 ?? '', /^[a-f0-9]{64}$/)
const samplerPath = await realpath(process.env.MILKUP_COLD_CPU_SAMPLER)
assert.equal(sha256(await readFile(samplerPath)), process.env.MILKUP_COLD_CPU_SAMPLER_SHA256)
const samplerOut = await open(path.join(output, 'cpu-sampler-stdout.log'), 'wx')
const samplerErr = await open(path.join(output, 'cpu-sampler-stderr.log'), 'wx')
let samplerCompletion
let samplerInputError = null

try {
  activeFile = path.join(output, 'ordinary.md')
  app = await startProduction(activeFile, output, { diagnostics: false })
  const runtimeUrl = new URL(await app.execute('return location.href'))
  assert(
    (runtimeUrl.protocol === 'tauri:' && runtimeUrl.hostname === 'localhost') ||
      (['http:', 'https:'].includes(runtimeUrl.protocol) &&
        runtimeUrl.hostname === 'tauri.localhost'),
    'Only packaged Tauri assets qualify; no development server or generic browser page',
  )
  const owner = await ownedApplication(binary, activeFile)
  const processEnvironment = (await readFile(`/proc/${owner.identity.pid}/environ`, 'utf8')).split(
    '\0',
  )
  const verifiedProfile = {}
  for (const key of [
    'XDG_DATA_HOME',
    'XDG_CONFIG_HOME',
    'XDG_CACHE_HOME',
    'MILKUP_DOM_DIAGNOSTICS',
  ]) {
    const actual = processEnvironment
      .find((value) => value.startsWith(key + '='))
      ?.slice(key.length + 1)
    assert.equal(actual, process.env[key], 'Actual application must inherit the isolated ' + key)
    verifiedProfile[key] = actual
  }
  await writeFile(
    path.join(output, 'native-owned-profile.json'),
    JSON.stringify({ ...owner.identity, runtimeUrl: runtimeUrl.href, verifiedProfile }, null, 2) +
      '\n',
  )
  // The first native dispatch may precede this handshake. The sampler must
  // retain that original dispatch timestamp and report the missed prefix.
  const samplerIdentity = { ...owner.identity }
  await writeFile(
    path.join(output, 'cpu-sampler-handshake.json'),
    JSON.stringify(samplerIdentity, null, 2) + '\n',
  )
  const sampler = spawn('/usr/bin/python3', [
    samplerPath,
    '--output', path.join(output, 'cpu-observer'),
    '--app-log', process.env.MILKUP_NATIVE_APP_LOG,
    '--cgroup-cpu-stat',
  ], { stdio: ['pipe', samplerOut.fd, samplerErr.fd] })
  samplerCompletion = new Promise((resolve) => {
    sampler.once('error', (error) => resolve({ error: String(error), exitCode: null }))
    sampler.once('close', (exitCode, signal) => resolve({ exitCode, signal }))
  })
  sampler.stdin.on('error', (error) => {
    samplerInputError = String(error)
  })
  sampler.stdin.end(JSON.stringify(samplerIdentity) + '\n')
  // No await of armed/sample/dispatch: continue the original public UI gate.
  await app.execute(
    `window.__productionDiagnostics={errors:[],cspViolations:[]};addEventListener('error',e=>window.__productionDiagnostics.errors.push(e.message));addEventListener('unhandledrejection',e=>window.__productionDiagnostics.errors.push(String(e.reason)));addEventListener('securitypolicyviolation',e=>window.__productionDiagnostics.cspViolations.push({directive:e.violatedDirective,blocked:e.blockedURI}));`,
  )
  nativeWindow = ui.windowId()
  x('windowsize', nativeWindow, '1100', '760')
  await coldBundledWindow()
  assert.equal(sha256(await readFile(binary)), process.env.MILKUP_MERMAID_BINARY_SHA256)
  assert.equal(sha256(await readFile(archivePath)), process.env.MILKUP_MERMAID_ARCHIVE_SHA256)
  await record('DIAGNOSTIC-COLD-ONLY-COMPLETE', {
    diagnosticOnly: true,
    acceptance: false,
    stoppedAfter: 'preinstalled cold window; no independent install or later prefix',
    completeAcceptanceMatrix: false,
    observer: 'Native stage traces and bounded external read-only CPU sampler; no JS wire observer.',
  })
} catch (error) {
  await writeFile(
    path.join(output, 'failure.json'),
    JSON.stringify({ error: String(error), stack: error.stack }, null, 2) + '\n',
  )
  if (app) await shot('FAIL', { error: String(error) }).catch(() => {})
  throw error
} finally {
  // Preserve App close timing. Sampler never keeps the product alive.
  try {
    if (app) await app.close()
  } finally {
    if (samplerCompletion) {
      const completion = await samplerCompletion
      await writeFile(path.join(output, 'cpu-sampler-completion.json'), JSON.stringify({ ...completion, stdinError: samplerInputError }, null, 2) + '\n')
    }
    await samplerOut.close()
    await samplerErr.close()
  }
}
// A sampler failure stops the batch even when the cold UI gate passed.
if (samplerCompletion) {
  const completion = await samplerCompletion
  assert.equal(completion.exitCode, 0, 'CPU diagnostic sampler must complete successfully')
  assert.equal(samplerInputError, null, 'CPU sampler identity stdin must be delivered')
}
