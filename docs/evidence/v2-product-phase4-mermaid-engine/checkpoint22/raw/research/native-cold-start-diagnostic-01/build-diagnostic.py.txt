def utc():return datetime.datetime.now(datetime.timezone.utc).isoformat()
assert sys.argv[1:]==['--execute-root-reviewed-build'], 'Static candidate: explicit root build GO required'
manifest=BASE/'build-inputs.json';go=json.loads((BASE/'root-build-go.json').read_text())

inputs=json.loads(manifest.read_text()); assert inputs['status']=='ROOT_REVIEWED_BUILD_INPUTS'

lock=(BASE/'build-exclusive.lock').open('a');fcntl.flock(lock,fcntl.LOCK_EX|fcntl.LOCK_NB)

assert go['exclusiveRepositoryAndCargoTarget'] is True

for row in provenance['frontend']:

def verify_frontend():

 assert actual==expected_front, 'All 12 production08 frontend files must be identical'

verify_frontend()
for row in inputs['files']: assert sha(row['path'])==row['sha256'],row['path']
for row in json.loads((PROD/'source-before.json').read_text())['files']:
 assert sha(REPO/row['path'])==row['sha256'], 'Source drift from production08: '+row['path']

backups={}; child=None;record={'head':head,'startedUtc':utc(),'classification':'DIAGNOSTIC_ONLY_NOT_PRODUCTION_ACCEPTANCE','frontendBefore':expected_front,'sourceDelta':inputs['overlays']}
def interrupted(signum,frame):raise InterruptedError('Build interrupted by signal '+str(signum))

try:
 for overlay in inputs['overlays']:
  target=REPO/overlay['repositoryPath'];candidate=Path(overlay['candidatePath'])
  assert target.is_relative_to(REPO/'apps/desktop/src-tauri/src')
  assert target.name in {'plugin_dom.rs','plugin_dom_process.rs','plugin_dom_request_diagnostic.rs'}
  assert sha(candidate)==overlay['candidateSha256']
  backups[target]=target.read_bytes() if target.exists() else None
  if backups[target] is not None:
   assert hashlib.sha256(backups[target]).hexdigest()==overlay['originalSha256']
   backup=attempt/'restore'/overlay['repositoryPath'];backup.parent.mkdir(parents=True,exist_ok=True);backup.write_bytes(backups[target])
  else:assert overlay['originalSha256'] is None
  target.write_bytes(candidate.read_bytes())
 record['command']=['bash','-lc','source /workspace/.cloud/milkup-v2/env.sh\nunset CARGO_TARGET_X86_64_UNKNOWN_LINUX_GNU_RUNNER PROOT_NO_SECCOMP\nexec cargo build --manifest-path apps/desktop/src-tauri/Cargo.toml --release --locked --offline --features tauri/custom-protocol --bin milkup-desktop']
