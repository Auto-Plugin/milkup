B='/frozen/milkup-desktop';I='a'*48
class Clock:
    def __init__(self):self.value=1_000_000_000_000
    def __call__(self):self.value+=100;return self.value
class FakeFiles:
    def __init__(self):self.data={};self.taskmap={};self.reads=[];self.taskcalls=[]
    def process(self,pid,birth,parent,argv,children=(),env=None):
        w=['0']*22;w[0]='S';w[1]=str(parent);w[11]='3';w[12]='2';w[19]=str(birth)
        self.data[f'/proc/{pid}/stat']=f'{pid} (test process) '+ ' '.join(w)
        self.data[f'/proc/{pid}/cmdline']='\0'.join(argv)+'\0'
        self.data[f'/proc/{pid}/schedstat']='100 20 4'
        self.data[f'/proc/{pid}/environ']=env or ''
        self.taskmap[pid]=[str(pid)];self.data[f'/proc/{pid}/task/{pid}/children']=' '.join(map(str,children))
    def read(self,path,limit):
        self.reads.append(path)
        if path not in self.data:raise FileNotFoundError(2,'ENOENT',path)
        v=self.data[path];v=v.encode() if isinstance(v,str) else v
        if len(v)>limit:raise ValueError('read-size-limit')
        return v
    def tasks(self,pid,limit):
        self.taskcalls.append(pid);v=self.taskmap[pid]
        if len(v)>limit:raise ValueError('round-task-limit')
        return v

class Tests(unittest.TestCase):
    def test_binding_exact_and_no_env_output(self):
        o,fs,c,e=fixture();o.bind_identity_event(e,o.native_identity_source)
        self.assertEqual(o.app['birth'],100);self.assertEqual(o.known[(20,110)]['role'],'guardian')
        self.assertNotIn('SECRET',json.dumps(o.summary()));self.assertNotIn('hidden',json.dumps(o.summary()))
    def test_environment_permission_gap_with_native_proof(self):
        o,fs,c,e=fixture();original=fs.read
        def unavailable(path,limit):
            if path.endswith('/environ'):raise PermissionError(13,'Permission denied')
            return original(path,limit)
        fs.read=unavailable;o.bind_identity_event(e,o.native_identity_source)
        self.assertEqual(o.binding_events[0]['instanceEnvironmentCheck']['status'],'unavailable')
        self.assertIsNone(o.binding_events[0]['instanceEnvironmentMatched'])
        self.assertEqual(o.summary()['environmentUnavailableBindings'],1)
        self.assertIn((20,110),o.known)
    def test_environment_unavailable_without_native_proof_rejected(self):
        o,fs,c,e=fixture();original=fs.read
        def unavailable(path,limit):
            if path.endswith('/environ'):raise PermissionError(13,'Permission denied')
            return original(path,limit)
        fs.read=unavailable
        with self.assertRaisesRegex(ValueError,'creation-proof'):o.bind_identity_event(e)
        self.assertFalse(o.guardians)
    def test_readable_environment_mismatch_rejected(self):
        o,fs,c,e=fixture();fs.data['/proc/20/environ']='MILKUP_INTERNAL_DOM_INSTANCE_ID='+'b'*48+'\0'
        with self.assertRaisesRegex(ValueError,'instance-environment'):o.bind_identity_event(e,o.native_identity_source)
        self.assertFalse(o.guardians)
    def test_native_schema_required(self):

        self.assertFalse(o.guardians)
    def test_birth_mismatch_rejected(self):
        o,fs,c,e=fixture();e['guardianBirth']=111
        with self.assertRaises(ValueError):o.bind_identity_event(e,o.native_identity_source)
        self.assertFalse(o.guardians)
    def test_app_cannot_rebind(self):

    def test_static_no_global_proc_enumeration_or_signal(self):
        tree=ast.parse(Path(m.__file__).read_text());scans=[]
        for node in ast.walk(tree):
            if isinstance(node,ast.Call):
                name=ast.unparse(node.func)
                self.assertNotIn(name,['os.kill','os.killpg','signal.pthread_kill','subprocess.run','subprocess.Popen'])
                if name=='os.scandir':scans.append(ast.unparse(node.args[0]))
        self.assertEqual(scans,["f'/proc/{pid}/task'"])
