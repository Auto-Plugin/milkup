+const SCHEMA: &str = "milkup.webprocess-self-pidfd.v1";

+const MAX_RECORD: usize = 8192;
+const MAX_PROC: u64 = 4096;

+fn mono_ns() -> u64 {
+    let mut t = libc::timespec {
+        tv_sec: 0,
+        tv_nsec: 0,
+    };
+    if unsafe { libc::clock_gettime(libc::CLOCK_MONOTONIC, &mut t) } == 0 {
+        t.tv_sec as u64 * 1_000_000_000 + t.tv_nsec as u64
+    } else {
+        0
+    }
+}
+fn bounded_read(path: PathBuf) -> io::Result<String> {
+    let mut bytes = Vec::new();
+    File::open(path)?
+        .take(MAX_PROC + 1)
+        .read_to_end(&mut bytes)?;
+    if bytes.len() > MAX_PROC as usize {
+        return Err(io::Error::other("proc read cap"));
+    }
+    String::from_utf8(bytes).map_err(io::Error::other)
+}
+fn identity(pid: u32) -> io::Result<Value> {
+    let text = bounded_read(PathBuf::from(format!("/proc/{pid}/stat")))?;
+    let (prefix, rest) = text
+        .rsplit_once(')')
+        .ok_or_else(|| io::Error::other("stat framing"))?;
+    if prefix.split_once('(').map(|p| p.0.trim()) != Some(pid.to_string().as_str()) {
+        return Err(io::Error::other("stat pid mismatch"));
+    }
+    let w: Vec<_> = rest.split_whitespace().collect();
+    let parse = |n: usize| -> io::Result<u64> {
+        w.get(n)
+            .ok_or_else(|| io::Error::other("short stat"))?
+            .parse()
+            .map_err(io::Error::other)
+    };
+    Ok(json!({"pid":pid,"birth":parse(19)?,"parent":parse(1)?,"state":w[0]}))
+}
+fn owned_dir(file: &File) -> io::Result<()> {
+    let m = file.metadata()?;
+    if !m.is_dir() || m.uid() != unsafe { libc::geteuid() } || m.mode() & 0o7777 != 0o700 {
+        return Err(io::Error::other("diagnostic directory owner/mode"));
+    }
+    Ok(())
+}
+fn openat_dir(parent: &File, name: &str) -> io::Result<File> {
+    let n = CString::new(name)?;
+    let fd = unsafe {
+        libc::openat(
+            parent.as_raw_fd(),
+            n.as_ptr(),
+            libc::O_RDONLY | libc::O_DIRECTORY | libc::O_NOFOLLOW | libc::O_CLOEXEC,
+        )
+    };
+    if fd < 0 {
+        return Err(io::Error::last_os_error());
+    }
+    let file = unsafe { File::from_raw_fd(fd) };
+    owned_dir(&file)?;
+    Ok(file)
+}

+pub(super) struct Diagnostic {
+    directory: File,
+    stem: String,
+    instance: String,
+    token: String,
+    helper: Value,

+impl Diagnostic {
+    pub(super) fn prepare() -> Option<Rc<Self>> {
+        if std::env::var("MILKUP_WEBPROCESS_PIDFD_DIAGNOSTICS").as_deref() != Ok("1") {
+            return None;
+        }
+        match Self::prepare_inner() {
+            Ok(v) => Some(Rc::new(v)),
+            Err(e) => {
+                // Bounded once per gated helper. Never propagate into rendering.
+                eprintln!(
+                    "WEBPROCESS_PIDFD_DIAGNOSTIC setup failed errno={:?}",
+                    e.raw_os_error()
+                );
+                None
+            }
+        }
+    }
+    fn prepare_inner() -> io::Result<Self> {
+        let start = mono_ns();
+        let output = PathBuf::from(
+            std::env::var_os("MILKUP_WEBPROCESS_PIDFD_OUTPUT_DIR")
+                .ok_or_else(|| io::Error::other("missing diagnostic output directory"))?,
+        );
+        if !output.is_absolute() {
+            return Err(io::Error::other("diagnostic path must be absolute"));
+        }
+        // Final component only: ancestors are trusted run-owned preparation inputs.
+        let root = OpenOptions::new()
+            .read(true)
+            .custom_flags(libc::O_DIRECTORY | libc::O_NOFOLLOW | libc::O_CLOEXEC)
+            .open(output)?;
+        owned_dir(&root)?;
+        let instance =
+            std::env::var("MILKUP_INTERNAL_DOM_INSTANCE_ID").map_err(io::Error::other)?;
+        if instance.len() != 48 || !instance.bytes().all(|b| b.is_ascii_hexdigit()) {
+            return Err(io::Error::other("invalid instance"));
+        }
+        let helper = identity(std::process::id())?;
+        let parent = helper["parent"]
+            .as_u64()
+            .ok_or_else(|| io::Error::other("missing parent"))? as u32;
+        let guardian = identity(parent)?;
+        let token = super::opaque_id().map_err(io::Error::other)?;
+        // One helper claims this entire fresh run directory. Never reopen a slot.
+        if unsafe { libc::mkdirat(root.as_raw_fd(), c"slot".as_ptr(), 0o700) } != 0 {
+            return Err(io::Error::last_os_error());
+        }
+        let directory = openat_dir(&root, "slot")?;
+        let stem = format!(
+            "helper-{}-{}",
+            std::process::id(),
+            helper["birth"].as_u64().unwrap_or(0)
+        );
+        let result = Self {
+            directory,
+            stem,
+            instance,
+            token,
+            helper,
+            guardian,
+            received: Cell::new(false),
+        };
+        let mut event = result.base("armed");
+        event["status"] = json!("receiver-prepared");
+        event["prepareStartNs"] = json!(start);
+        event["captureEndNs"] = json!(mono_ns());
+        event["extensionSha256"] = json!(format!("{:x}", Sha256::digest(super::EXTENSION)));
+        event["tokenPrepared"] = json!(true);
+        result.publish("armed", &event)?;
+        Ok(result)
+    }
+    fn base(&self, kind: &str) -> Value {
+        json!({"schema":SCHEMA,"kind":kind,"final":true,"instanceId":self.instance,
+            "helper":self.helper,"guardian":self.guardian,"nativeMonotonicNs":mono_ns(),
+            "contextProvenance":"owned extension -> one private ephemeral native WebContext; no JavaScript channel",
+            "clock":"local CLOCK_MONOTONIC; cross-process comparability must be verified",
+            "recordLimitBytes":MAX_RECORD})
+    }
+    fn publish(&self, kind: &str, value: &Value) -> io::Result<()> {
+        let bytes = serde_json::to_vec(value).map_err(io::Error::other)?;
+        if bytes.len() > MAX_RECORD {
+            return Err(io::Error::other("record byte cap"));
+        }
+        let temp = CString::new(format!("{}-{kind}.tmp", self.stem))?;
+        let final_name = CString::new(format!("{}-{kind}.json", self.stem))?;
+        let d = self.directory.as_raw_fd();
+        let fd = unsafe {
+            libc::openat(
+                d,
+                temp.as_ptr(),
+                libc::O_WRONLY | libc::O_CREAT | libc::O_EXCL | libc::O_NOFOLLOW | libc::O_CLOEXEC,
+                0o600,
+            )
+        };
+        if fd < 0 {
+            return Err(io::Error::last_os_error());
+        }
+        let mut file = unsafe { File::from_raw_fd(fd) };
+        let outcome = (|| {
+            let m = file.metadata()?;
+            if !m.is_file() || m.uid() != unsafe { libc::geteuid() } || m.mode() & 0o7777 != 0o600 {
+                return Err(io::Error::other("record owner/mode/type"));
+            }
+            file.write_all(&bytes)?;
+            // linkat publishes complete bytes without replacing any existing name.
+            if unsafe { libc::linkat(d, temp.as_ptr(), d, final_name.as_ptr(), 0) } != 0 {
+                return Err(io::Error::last_os_error());
+            }
+            Ok(())
+        })();
+        unsafe {
+            libc::unlinkat(d, temp.as_ptr(), 0);
+        }
+        outcome
+    }

+    pub(super) fn receive(&self, message: &webkit2gtk::UserMessage) {
+        // A single callback consumes this slot, including malformed/failure cases.
+        if self.received.replace(true) {
+            return;
+        }
+        let start = mono_ns();
+        let mut event = self.base("result");
+        event["receiveStartNs"] = json!(start);
+        event["status"] = json!("failed");
+        let outcome = self.receive_inner(message, &mut event);
+        if let Err(e) = outcome {
+            event["receiverErrno"] = json!(e.raw_os_error());
+            event["receiverError"] = json!(e.to_string().chars().take(256).collect::<String>());
+        }
+        event["captureEndNs"] = json!(mono_ns());
+        // No stdout/stderr write from the GTK callback. A failed exclusive
+        // publication leaves the result absent; the external observer must
+        // classify that as incomplete, never as successful FD delivery.
+        let _ = self.publish("result", &event);
+    }
+    fn receive_inner(
+        &self,
+        message: &webkit2gtk::UserMessage,
+        event: &mut Value,
+    ) -> io::Result<()> {
+        use gio::prelude::*;
+        type Payload = (String, String, i32, i32, u32, i32, u64, u64, u64);
+        let p: Payload = message
+            .parameters()
+            .and_then(|v| {
+                if v.size() > 1024 {
+                    None
+                } else {
+                    v.get::<Payload>()
+                }
+            })

+            return Err(io::Error::other("native context token mismatch"));

+        event["nativeStage"] = json!(p.2);
+        event["nativeErrno"] = json!(p.3);
+        event["nativeGErrorDomain"] = json!(p.4);
+        event["nativeGErrorCode"] = json!(p.5);
+        event["webProcessCallbackStartNs"] = json!(p.6);
+        event["webProcessCallbackEndNs"] = json!(p.7);
+        event["pageId"] = json!(p.8);
+        let list = message.fd_list();
+        let count = list.as_ref().map(|l| l.length()).unwrap_or(0);
+        event["receivedFdCount"] = json!(count);
+        if p.2 != 0 {
+            if count != 0 {
+                return Err(io::Error::other("failure message unexpectedly has FDs"));
+            }
+            return Err(io::Error::other("WebProcess self-pidfd preparation failed"));
+        }
+        if p.3 != 0 || count != 1 {
+            return Err(io::Error::other(
+                "success requires exactly one FD and no errno",
+            ));
+        }
+        // GUnixFDList.get duplicates: this receiver exclusively owns the duplicate.
+        let raw = list
+            .ok_or_else(|| io::Error::other("missing FD list"))?
+            .get(0)
+            .map_err(io::Error::other)?;
+        let retained = unsafe { File::from_raw_fd(raw) };
+        let flags = unsafe { libc::fcntl(raw, libc::F_GETFD) };
+        if flags < 0 || unsafe { libc::fcntl(raw, libc::F_SETFD, flags | libc::FD_CLOEXEC) } < 0 {
+            return Err(io::Error::last_os_error());
+        }
+        let read_start = mono_ns();
+        let fdinfo = bounded_read(PathBuf::from(format!("/proc/self/fdinfo/{raw}")))?;
+        if !fdinfo.lines().any(|l| l.starts_with("Pid:")) {
+            return Err(io::Error::other("pidfd has no Pid field"));
+        }
+        event["heldFd"] = json!(raw);
+        event["receiverFdinfoRaw"] = json!(fdinfo);
+        event["receiverReadStartNs"] = json!(read_start);
+        event["receiverReadEndNs"] = json!(mono_ns());

+        );

+        // guardian's existing helper exit closes it; no new cleanup authority.
+        std::mem::forget(retained);
+        Ok(())
+    }
