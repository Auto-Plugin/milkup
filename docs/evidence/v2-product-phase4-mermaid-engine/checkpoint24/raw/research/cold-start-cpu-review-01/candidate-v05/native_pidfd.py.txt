"""Bounded native record + exact-helper held-pidfd proof. No signals/scan/FD duplication.
Native protocol validator is filled only from the frozen native schema.
"""
import hashlib
import json
import os
import stat

RECORD_BYTES=8192
MAX_CHAIN_DEPTH=8
MAX_CHAIN_NODES=16
MAX_CHAIN_READS=32
MAX_NATIVE_STAT_READS=40
TERMINAL=('Z','X','x')


def integer(value,minimum=0,maximum=(1<<63)-1):
    if type(value) is not int or not minimum<=value<=maximum:raise ValueError('native-integer')
    return value


def read_record(directory,pid,birth,kind,limit=RECORD_BYTES):
    """No directory enumeration. Fixed path parts derived from bound helper only."""
    integer(pid,1,(1<<31)-1);integer(birth,1)
    if kind not in ('armed','result'):raise ValueError('record-kind-path')
    handles=[]
    try:
        root=os.open(directory,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW);handles.append(root)
        root_stat=os.fstat(root)
        if root_stat.st_uid!=os.getuid() or stat.S_IMODE(root_stat.st_mode)!=0o700:raise ValueError('native-root-mode-owner')
        slot=os.open('slot',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW,dir_fd=root);handles.append(slot)
        slot_stat=os.fstat(slot)
        if slot_stat.st_uid!=os.getuid() or stat.S_IMODE(slot_stat.st_mode)!=0o700:raise ValueError('native-slot-mode-owner')
        name=f'helper-{pid}-{birth}-{kind}.json'
        fd=os.open(name,os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK,dir_fd=slot);handles.append(fd)
        before=os.fstat(fd)
        if not stat.S_ISREG(before.st_mode) or before.st_uid!=os.getuid() or stat.S_IMODE(before.st_mode)!=0o600:raise ValueError('native-file-type-mode-owner')
        if before.st_size>limit:raise ValueError('native-record-byte-limit')
        raw=os.pread(fd,limit+1,0)
        after=os.fstat(fd)
        attrs=('st_dev','st_ino','st_size','st_mtime_ns','st_ctime_ns')
        if len(raw)>limit or len(raw)!=before.st_size:raise ValueError('native-record-size-race')
        if any(getattr(before,k)!=getattr(after,k) for k in attrs):raise ValueError('native-record-mutated-during-read')
        return raw,dict(rootDevice=root_stat.st_dev,rootInode=root_stat.st_ino,slotDevice=slot_stat.st_dev,slotInode=slot_stat.st_ino,
                       device=before.st_dev,inode=before.st_ino,bytes=len(raw),name=name,uid=before.st_uid,mode=stat.S_IMODE(before.st_mode))
    finally:
        for fd in reversed(handles):os.close(fd)


def native_stat(o,pid,budget,chain=False):
    o.check_budget(budget)
    if budget.get('nativeStatReads',0)>=MAX_NATIVE_STAT_READS:raise ValueError('native-stat-cap')
    if chain and budget.get('chainReads',0)>=MAX_CHAIN_READS:raise ValueError('native-chain-read-cap')
    budget['nativeStatReads']=budget.get('nativeStatReads',0)+1
    if chain:budget['chainReads']=budget.get('chainReads',0)+1
    start=o.now();row=o.identity(integer(pid,1,(1<<31)-1));end=o.now()
    if row['state'] in TERMINAL:raise ValueError(f'native-terminal-node-pid-{pid}')
    o.check_budget(budget)
    return dict(identity=row,readStartNs=start,readEndNs=end)


def stable(old,new):
    if any(old[k]!=new[k] for k in ('pid','birth','parent')):raise ValueError(f'native-chain-identity-or-edge-changed-pid-{old["pid"]}')
    if new['state'] in TERMINAL:raise ValueError('native-terminal-node')


def chain_phase_a(o,target_pid,helper,budget):
    nodes=[];seen=set();pid=target_pid
    while True:
        o.check_budget(budget)
        if pid in seen:raise ValueError('native-chain-cycle')
        if pid<=1:raise ValueError('native-chain-escaped-helper')
        if len(nodes)>=MAX_CHAIN_DEPTH:raise ValueError('native-chain-depth-cap')
        if budget.get('chainNodes',0)>=MAX_CHAIN_NODES:raise ValueError('native-chain-node-cap')
        budget['chainNodes']=budget.get('chainNodes',0)+1;seen.add(pid)
        item=native_stat(o,pid,budget,chain=True);row=item['identity']
        if nodes and nodes[-1]['identity']['birth']<row['birth']:raise ValueError('native-chain-birth-order')
        nodes.append(item)
        if pid==helper['pid']:
            stable(helper,row)
            if len(nodes)<2:raise ValueError('native-target-is-helper')
            return nodes
        pid=row['parent']


def chain_phase_b(o,nodes,budget):
    repeated=[]
    for item in reversed(nodes):
        after=native_stat(o,item['identity']['pid'],budget,chain=True)
        stable(item['identity'],after['identity']);repeated.append(after)
    return list(reversed(repeated))


def held_proof(o,helper,fd,budget):
    """Read kernel fdinfo metadata, NEVER /proc/PID/fd/FD target contents."""
    guardian=o.guardians[helper['instanceId']]
    guardian_before=native_stat(o,guardian['pid'],budget);stable(guardian,guardian_before['identity'])
    initial=native_stat(o,helper['pid'],budget);stable(helper,initial['identity'])
    integer(fd,0,(1<<31)-1)
    path=f'/proc/{helper["pid"]}/fdinfo/{fd}'
    first_start=o.now();before=o.read_fdinfo(helper,path,budget);first_end=o.now()
    fields=o.pidfd_parse(before)
    if fields is None:raise ValueError('native-held-fd-not-pidfd')
    integer(fields['pid'],1,(1<<31)-1)
    if fields['nspid'] is not None and any(type(x) is not int or x<=0 for x in fields['nspid']):raise ValueError('native-invalid-nspid')
    if len(before)>1024:raise ValueError('native-fdinfo-proof-byte-cap')
    nodes=chain_phase_a(o,fields['pid'],helper,budget)
    # argv is bracketed by ALL of Phase B, not read after identity verification.
    role,argv=o.child_role(nodes[0]['identity'],helper,budget)
    repeated=chain_phase_b(o,nodes,budget)
    second_start=o.now();after=o.read_fdinfo(helper,path,budget);second_end=o.now()
    if before!=after:raise ValueError('native-held-fdinfo-changed')
    final=native_stat(o,helper['pid'],budget);stable(initial['identity'],final['identity'])
    guardian_after=native_stat(o,guardian['pid'],budget);stable(guardian_before['identity'],guardian_after['identity'])
    o.check_budget(budget)
    return dict(row=nodes[0]['identity'],role=role,argvEvidence=argv,
                proof=dict(helperBefore=initial,helperAfter=final,guardianBefore=guardian_before,guardianAfter=guardian_after,fd=fd,pidFields=fields,
                           fdinfoBefore=before.decode('ascii'),fdinfoAfter=after.decode('ascii'),
                           firstReadStartNs=first_start,firstReadEndNs=first_end,secondReadStartNs=second_start,secondReadEndNs=second_end,
                           chainBefore=nodes,chainAfter=repeated,
                           qualification='Observed helper-anchored ancestry brackets, not atomic snapshot or full tree; no NSpid translation/ABA exclusion'))

SCHEMA='milkup.webprocess-self-pidfd.v1'
PROVENANCE='owned extension -> one private ephemeral native WebContext; no JavaScript channel'


def strict_json(raw):
    depth=0;quoted=False;escaped=False
    for char in raw.decode('utf-8'):
        if quoted:
            if escaped:escaped=False
            elif char=='\\':escaped=True
            elif char=='"':quoted=False
        elif char=='"':quoted=True
        elif char in '[{':
            depth+=1
            if depth>8:raise ValueError('native-record-json-depth')
        elif char in ']}':depth-=1
    def constant(value):raise ValueError('native-record-nonfinite-json')
    def pairs(items):
        result={}
        for key,value in items:
            if key in result:raise ValueError('native-record-duplicate-key')
            result[key]=value
        return result
    value=json.loads(raw,object_pairs_hook=pairs,parse_constant=constant)
    if not isinstance(value,dict):raise ValueError('native-record-object')
    return value


def record(o,helper,kind,budget):
    o.check_budget(budget)
    if budget.get('nativeRecordReads',0)>=2:raise ValueError('native-record-read-cap')
    budget['nativeRecordReads']=budget.get('nativeRecordReads',0)+1
    start=o.now()
    reader=getattr(o.fs,'native_record',read_record)
    try:raw,metadata=reader(o.native_dir,helper['pid'],helper['birth'],kind,RECORD_BYTES)
    finally:o.native_record_io_ns+=o.now()-start
    o.check_budget(budget)
    if len(raw)>RECORD_BYTES:raise ValueError('native-record-byte-limit')
    digest=hashlib.sha256(raw).hexdigest();data=strict_json(raw);o.check_budget(budget)
    identity=(metadata['rootDevice'],metadata['rootInode'],metadata['slotDevice'],metadata['slotInode'],metadata['device'],metadata['inode'],digest)
    if kind in o.native_sources and tuple(o.native_sources[kind]['identity'])!=identity:raise ValueError('native-record-replaced-no-rebind')
    if data.get('schema')!=SCHEMA or data.get('kind')!=kind or data.get('final') is not True:raise ValueError('native-record-schema-kind-final')
    if data.get('instanceId')!=helper['instanceId']:raise ValueError('native-record-instance')
    guardian=o.guardians[helper['instanceId']]
    if helper['parent']!=guardian['pid']:raise ValueError('native-helper-guardian-link')
    for name,expected in [('helper',helper),('guardian',guardian)]:
        row=data.get(name)
        if not isinstance(row,dict):raise ValueError('native-record-identity-object')
        for field in ('pid','birth','parent'):
            integer(row.get(field),1)
            if row[field]!=expected[field]:raise ValueError('native-record-'+name+'-'+field)
        if row.get('state') not in ('R','S','D','T','t','I','P','W'):raise ValueError('native-record-terminal-or-unknown-state')
    ns=integer(data.get('nativeMonotonicNs'),1);end=integer(data.get('captureEndNs'),1)
    if not o.started<=ns<=end<=o.now():raise ValueError('native-record-clock-window')
    if data.get('contextProvenance')!=PROVENANCE or data.get('recordLimitBytes')!=RECORD_BYTES:raise ValueError('native-record-provenance-shape')
    if kind=='armed':
        if data.get('status')!='receiver-prepared' or data.get('tokenPrepared') is not True:raise ValueError('native-record-not-armed')
        if data.get('extensionSha256')!=o.native_extension_sha256:raise ValueError('native-extension-hash')
        if not o.started<=integer(data.get('prepareStartNs'),1)<=ns:raise ValueError('native-prepare-clock')
    elif data.get('status') not in ('held','failed'):raise ValueError('native-result-status')
    proof=dict(identity=list(identity),metadata=metadata,sha256=digest,readStartNs=start,readEndNs=o.now(),record=data)
    # Two immutable source records maximum, not a per-round growing payload log.
    if kind not in o.native_sources:o.native_sources[kind]=proof
    return data


def observe(o,helper,budget):
    """At most one target attempt per round, from one pinned helper; no fallback."""
    if not o.native_dir:return None
    key=(helper['pid'],helper['birth'])
    if o.native_helper is None:o.native_helper=key
    if o.native_helper!=key:
        o.native_status('additional-helper-not-supported',helper,budget,gap=True);return None
    if budget.get('nativeTargetAttempts',0)>=1:
        o.native_status('native-target-round-cap',helper,budget,gap=True);return None
    budget['nativeTargetAttempts']=1
    stage='record-armed'
    try:
        armed=record(o,helper,'armed',budget);stage='record-result'
        result=record(o,helper,'result',budget)
        if result['status']=='failed':
            o.native_status('native-receiver-failed',helper,budget,dict(receiverErrno=result.get('receiverErrno'),receiverError=str(result.get('receiverError',''))[:256],nativeStage=result.get('nativeStage'),nativeErrno=result.get('nativeErrno')),gap=True);return None
        stage='held-record-validation'
        if result.get('tokenMatched') is not True or type(result.get('receivedFdCount')) is not int or result['receivedFdCount']!=1:raise ValueError('native-token-or-fd-count')
        if any(type(result.get(k)) is not int or result[k]!=0 for k in ('nativeStage','nativeErrno','nativeGErrorDomain','nativeGErrorCode')):raise ValueError('native-held-error-fields')
        integer(result.get('pageId'),0,(1<<64)-1);fd=integer(result.get('heldFd'),0,(1<<31)-1)
        raw=result.get('receiverFdinfoRaw')
        if not isinstance(raw,str) or len(raw.encode('utf-8'))>4096:raise ValueError('native-receiver-fdinfo-shape')
        times=[armed['captureEndNs']]+[integer(result.get(k),1) for k in ('webProcessCallbackStartNs','webProcessCallbackEndNs','receiveStartNs','receiverReadStartNs','receiverReadEndNs','captureEndNs')]
        if times!=sorted(times):raise ValueError('native-callback-clock-order')
        if not result['receiveStartNs']<=result['nativeMonotonicNs']<=result['captureEndNs']:raise ValueError('native-receive-clock')
        stage='external-held-fd-chain'
        found=held_proof(o,helper,fd,budget)
        found['proof']['bindingArgvEvidence']=found['argvEvidence']
        found['proof']['nativeSourceDigests']={k:v['sha256'] for k,v in o.native_sources.items()}
        if found['role']!='webkit-web':
            o.native_unbound_proof=found
            o.native_status('native-target-argv-not-webprocess',helper,budget,dict(targetPid=found['row']['pid'],targetBirth=found['row']['birth']),gap=True);return None
        o.native_status('native-verified-webprocess',helper,budget,dict(targetPid=found['row']['pid'],targetBirth=found['row']['birth']))
        return found
    except (OSError,ValueError,KeyError,TypeError,UnicodeError) as exc:
        o.native_status('native-proof-incomplete',helper,budget,dict(stage=stage,errorClass=type(exc).__name__,errno=getattr(exc,'errno',None),detail=str(exc)[:160]),gap=True)
        return None
