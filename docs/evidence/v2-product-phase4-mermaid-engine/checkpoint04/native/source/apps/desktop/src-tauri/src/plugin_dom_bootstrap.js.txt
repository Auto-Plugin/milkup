// Trusted generic DOM renderer bootstrap. No editor bridge or host capabilities.
Object.defineProperty(globalThis, '__milkupDomInit', { configurable: true, writable: false, value: (token, manifest) => {
  'use strict';
  delete globalThis.__milkupDomInit;
  const stringify = JSON.stringify.bind(JSON);
  const parse = JSON.parse.bind(JSON);
  const bytes = new TextEncoder();
  const encode = bytes.encode.bind(bytes);
  const freeze = Object.freeze.bind(Object);
  const define = Object.defineProperty.bind(Object);
  const Abort = AbortController;
  const RPC = 'milkup.plugin.isolation.rpc.v1';
  const RENDER = 'milkup.plugin.document-render.v1';
  let output = null, active = false, activation, running, generation = 0;
  function push(message) {
    let text;
    try { text = stringify(message); } catch { throw new Error('invalid-output'); }
    if (encode(text).length > 266240) throw new Error('invalid-output');
    if (output !== null) throw new Error('busy');
    output = text;
  }
  function errorCode(error) {
    const code = error?.code ?? error?.message;
    return ['invalid-input', 'invalid-output', 'revoked', 'aborted', 'busy', 'timeout', 'unavailable'].includes(code) ? code : 'unavailable';
  }
  // The private token gates input and draining; plugin code has no result setter.
  define(globalThis, '__milkupDom', { configurable: false, writable: false, value: freeze((key, message) => {
    if (key !== token) return null;
    if (message === null) { const result = output; output = null; return result; }
    void dispatch(message);
    return null;
  }) });
  const modulePromise = import('milkup-dom://job/module.js').then(namespace => {
    const module = namespace.default ?? namespace;
    if (module === null || typeof module !== 'object' || Array.isArray(module)) throw new Error('Invalid plugin module export');
    return module;
  });
  async function dispatch(message) {
    if (message.protocol === RPC) {
      try {
        if (message.payload?.pluginId !== manifest.id) throw new Error('Plugin identity mismatch');
        const module = await modulePromise;
        let value = null;
        if (message.method === 'activate') {
          if (active) throw new Error('Already activated');
          generation++;
          activation = await module.activate?.(freeze({ manifest: freeze(manifest), pluginId: manifest.id,
            permissions: freeze(['document:read']), host: freeze({}) }));
          active = true;
          value = { commands: [], renderers: [], documentRenderers: manifest.contributes.documentRenderers
            .filter(r => typeof (activation?.documentRenderers?.[r.id] ?? module.documentRenderers?.[r.id]) === 'function').map(r => r.id) };
        } else if (message.method === 'deactivate' || message.method === 'dispose') {
          active = false; generation++; running?.controller.abort();
          if (message.method === 'deactivate') { await module.deactivate?.(); activation = undefined; }
          else await activation?.dispose?.();
        } else throw new Error('DOM host supports document renderers only');
        push({ protocol: RPC, type: 'response', id: message.id, ok: true, value });
      } catch (error) {
        push({ protocol: RPC, type: 'response', id: message.id, ok: false, error: { message: String(error?.message ?? 'DOM activation failed').slice(0, 1024) } });
      }
      return;
    }
    if (message.protocol !== RENDER) return;
    if (message.type === 'cancel') { if (running?.id === message.id) running.controller.abort(); return; }
    if (message.type !== 'render') return;
    const current = generation;
    const controller = new Abort();
    try {
      if (!active) throw new Error('revoked');
      if (running) throw new Error('busy');
      const { pluginId, rendererId, request } = message.payload;
      if (pluginId !== manifest.id || !manifest.contributes.documentRenderers.some(r => r.id === rendererId)) throw new Error('unavailable');
      const module = await modulePromise;
      const renderer = activation?.documentRenderers?.[rendererId] ?? module.documentRenderers?.[rendererId];
      if (typeof renderer !== 'function') throw new Error('unavailable');
      running = { id: message.id, controller };
      const value = await renderer(request, { signal: controller.signal });
      if (!active || generation !== current) throw new Error('revoked');
      if (controller.signal.aborted) throw new Error('aborted');
      // Serialize inside the native render deadline, then enforce the public payload budget.
      const copied = stringify(value);
      if (typeof copied !== 'string' || encode(copied).length > 262144) throw new Error('invalid-output');
      push({ protocol: RENDER, type: 'result', id: message.id, ok: true, value: parse(copied) });
    } catch (error) {
      push({ protocol: RENDER, type: 'result', id: message.id, ok: false, error: errorCode(error) });
    } finally { if (running?.id === message.id) running = undefined; }
  }
} });
