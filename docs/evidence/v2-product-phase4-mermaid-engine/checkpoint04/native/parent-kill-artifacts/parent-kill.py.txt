def stamp(): return {'mono':time.monotonic(),'wall':time.time()}
def sha(path):
    h=hashlib.sha256()
    with open(path,'rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''): h.update(b)
    return h.hexdigest()
def write_json(path,value): Path(path).write_text(json.dumps(value,ensure_ascii=False,indent=2)+'\n')
def emit(fd,value):
    data=(json.dumps(value,separators=(',',':'))+'\n').encode()
    assert len(data)<=16384
    os.write(fd,data)
def proc(pid):
    try:
        root=Path('/proc')/str(pid)
        raw=(root/'stat').read_text(); rest=raw.rsplit(')',1)[1].split()
        result={'pid':pid,'birth':int(rest[19]),'parent':int(rest[1]),'group':int(rest[2]),'session':int(rest[3]),'state':rest[0],'ticks':int(rest[11])+int(rest[12]),'comm':raw[raw.find('(')+1:raw.rfind(')')]}
        result['cmdline']=(root/'cmdline').read_bytes()[:512].replace(b'\0',b' ').decode(errors='replace')
        return result
    except (OSError,ValueError,IndexError): return None
def table():
    return {p['pid']:p for q in Path('/proc').iterdir() if q.name.isdigit() and (p:=proc(int(q.name))) is not None}
def descendants(tab,seeds):
    chosen={pid for pid,birth in seeds.items() if pid in tab and tab[pid]['birth']==birth}
    while True:
        add={pid for pid,p in tab.items() if p['parent'] in chosen}-chosen
        if not add: return chosen
        chosen.update(add)
def kill_exact(pid,birth):
    try: fd=os.pidfd_open(pid)
    except ProcessLookupError: return False
    try:
        p=proc(pid)
        if p and p['birth']==birth:
            signal.pidfd_send_signal(fd,signal.SIGKILL); return True
        return False
    finally: os.close(fd)
def pipe_high(flags=0):
    pair=os.pipe2(os.O_CLOEXEC|flags); high=[]
    for fd in pair:
        high.append(fcntl.fcntl(fd,fcntl.F_DUPFD_CLOEXEC,20));os.close(fd)
    return high

def receive_line(fd,deadline,limit=LIMIT):
    data=bytearray()
    while True:
        left=deadline-time.monotonic()
        if left<=0: raise TimeoutError('bounded frame deadline')
        if not select.select([fd],[],[],left)[0]: raise TimeoutError('bounded frame deadline')
        chunk=os.read(fd,1)
        if not chunk: raise EOFError('frame channel closed')
        if chunk==b'\n':
            value=json.loads(data)
            if time.monotonic()>=deadline: raise TimeoutError('JSON parsing exceeded frame deadline')
            return value
        data.extend(chunk)
        if len(data)>limit: raise ValueError('frame too large')
def send_request(fd,message):
    data=(json.dumps(message,separators=(',',':'))+'\n').encode()
    assert len(data)<=1024*1024
    while data:
        written=os.write(fd,data);data=data[written:]
def render_request(identifier,source):
    return {'protocol':RENDER,'type':'render','id':identifier,'payload':{'pluginId':MANIFEST['id'],'rendererId':'test.renderer','request':{'source':source}}}

def owner(channel,binary,guard,out):
    """Only this process ever owns the stop-pipe write end."""
    in_r,in_w=pipe_high();res_r,res_w=pipe_high();stop_r,stop_w=pipe_high(os.O_NONBLOCK)
    stdout=os.open(out/'guardian.stdout.log',os.O_CREAT|os.O_EXCL|os.O_WRONLY,0o600)
    stderr=os.open(out/'guardian.stderr.log',os.O_CREAT|os.O_EXCL|os.O_WRONLY,0o600)
    begun=time.monotonic();guardian=os.fork()
    if guardian==0:
        os.setsid()
        for source,target in [(in_r,0),(stdout,1),(stderr,2),(res_w,3),(stop_r,4)]: os.dup2(source,target,inheritable=True)
        for q in list(Path('/proc/self/fd').iterdir()):
            if int(q.name)>4:
                try: os.close(int(q.name))
                except OSError: pass
        env=dict(os.environ,MILKUP_INTERNAL_DOM_GUARD_DIR=str(guard),MILKUP_INTERNAL_DOM_INSTANCE_ID='parent-kill-fixture',MILKUP_DOM_DIAGNOSTICS='1')
        os.execve(binary,[binary,'--milkup-internal-dom-helper'],env)
    for fd in (in_r,res_w,stop_r,stdout,stderr): os.close(fd)
    identity=proc(guardian);assert identity
    emit(channel,{'event':'guardian-spawned',**stamp(),'guardian':identity,'preparationStart':begun,'owner':proc(os.getpid()),'stopWriterFd':stop_w})
    try:
        snapshot={'manifest':MANIFEST,'module':MODULE,'sha256':hashlib.sha256(MODULE.encode()).hexdigest()}
        payload=json.dumps(snapshot,separators=(',',':'),ensure_ascii=False).encode()
        data=struct.pack('<I',len(payload))+payload
        while data: data=data[os.write(in_w,data):]
        send_request(in_w,{'protocol':RPC,'type':'request','id':'activate1','method':'activate','payload':{'pluginId':MANIFEST['id'],'manifest':MANIFEST,'permissions':['document:read'],'hostCapabilities':[]}})
        activation=receive_line(res_r,begun+10)
        assert activation.get('ok') is True and activation.get('id')=='activate1',activation
        assert activation['value']['documentRenderers']==['test.renderer'],activation
        emit(channel,{'event':'activated',**stamp(),'elapsed':time.monotonic()-begun,'result':activation})
        before=time.monotonic();send_request(in_w,render_request('bbox1','bbox'))
        result=receive_line(res_r,before+2)
        assert result.get('ok') is True and result.get('id')=='bbox1',result
        assert result['value']['width']>0 and result['value']['height']>0 and result['value']['viewport']>0,result
        emit(channel,{'event':'bbox-ready',**stamp(),'elapsed':time.monotonic()-before,'result':result})
        command=receive_line(channel,time.monotonic()+5,16384)
        assert command=={'command':'loop'}
        before=time.monotonic();send_request(in_w,render_request('loop1','loop'))
        emit(channel,{'event':'loop-dispatched',**stamp(),'dispatchMono':before,'fixedExecutionDeadline':before+2})
        # Keep FD4's sole writer alive; do not send stop. Observer must SIGKILL us.
        # Guardrail: if observer fails to kill, close at the fixed execution limit.
        left=before+2-time.monotonic()
        if left>0: select.select([channel],[],[],left)
        raise TimeoutError('observer did not kill owner within 2s loop budget')
    except BaseException as error:
        try: emit(channel,{'event':'owner-error',**stamp(),'error':repr(error)})
        except OSError: pass
        os.close(stop_w)
        os._exit(2)

def main():
    parser=argparse.ArgumentParser();parser.add_argument('--output',required=True);parser.add_argument('--expected-binary-sha',required=True);parser.add_argument('--binary',default='/probe/bin/milkup-desktop');parser.add_argument('--guard-library',default='/probe/guard.so')
    args=parser.parse_args();out=Path(args.output);out.mkdir(parents=True,exist_ok=False)
    events=[];samples=[];known={};first_seen={};reaped={};owner_pid=None;owner_identity=None;guardian=None;success=False
    result={'status':'not-started','productAcceptance':False,'scope':'Actual product guardian/helper/WebKit; external simulated App SIGKILL; no Rust Instance deadline inference','escapedChildren':'Only naturally observed groups are recorded; no arbitrary OS child is injected through DOM. The separate shared-guardian escaped-child unit remains independent evidence.'}
    guard=out/'private-guard';guard.mkdir(mode=0o700);shutil.copyfile(args.guard_library,guard/'libmilkup_dom_guard.so');os.chmod(guard/'libmilkup_dom_guard.so',0o600);os.chmod(guard,0o500)
    def event(value): events.append(value);write_json(out/'events.json',events)

            try: owner(b.fileno(),args.binary,guard,out)
            except BaseException as error:
                try:emit(b.fileno(),{'event':'owner-error',**stamp(),'error':repr(error)})
                except OSError:pass
            os._exit(2)
        b.close();owner_identity=proc(owner_pid);assert owner_identity
        setup_limit=time.monotonic()+13
        while True:
            value=receive_line(a.fileno(),setup_limit,16384);event(value)
            assert value['event']!='owner-error',value
            if value['event']=='guardian-spawned':
                guardian=value['guardian'];known[guardian['pid']]=guardian['birth']
            sample()

        baseline=sample();baseline_ticks={p['pid']:(p['birth'],p['ticks']) for p in baseline['members'] if 'WebKitWebProcess' in p['cmdline'] or p['comm'].startswith('WebKitWebPro')}

    return 0 if success else 1
if __name__=='__main__':sys.exit(main())
