// Independent converter parsers. These do not call or trust the host validator.
export class ConversionError extends Error {
  constructor(code,reason,path=''){super(reason);this.code=code;this.path=path;}
}
export const fail=(code,reason,path='')=>{throw new ConversionError(code,reason,path)};
export function finite(value,label='number',limit=8192){const n=Number(value);if(!Number.isFinite(n)||Math.abs(n)>limit)fail('limit-exceeded',label);return n;}
const NUMBER=/^[+-]?(?:\d+\.?\d*|\.\d+)(?:e[+-]?\d+)?/i;
export function length(value,defaultValue='0') {
  value=String(value??defaultValue).trim();
  const m=/^([+-]?(?:\d+\.?\d*|\.\d+)(?:e[+-]?\d+)?)(px|em|%)?$/i.exec(value);
  if(!m)fail('unsupported-feature','unsupported length: '+value);
  return {value:finite(m[1],'length'),unit:m[2]==='%'?'percent':m[2]??'user'};
}
export function numberList(value){
  let text=String(value??'').trim(),out=[];
  while(text){const m=NUMBER.exec(text);if(!m)fail('invalid-output','invalid numeric list');out.push(finite(m[0]));text=text.slice(m[0].length);const sep=/^[\s,]*/.exec(text)[0];if((sep.match(/,/g)||[]).length>1)fail('invalid-output','repeated comma');text=text.slice(sep.length);if(sep.includes(',')&&!text)fail('invalid-output','trailing comma');if(out.length>8192)fail('limit-exceeded','numeric list');}
  return out;
}
export function lengthList(value){if(!value?.trim())return undefined;return value.trim().split(/[\s,]+/).map(x=>length(x));}
export function color(value){
  value=String(value).trim();if(value==='transparent')return [0,0,0,0];
  let m=/^rgba?\((.*)\)$/i.exec(value);
  if(m){const a=m[1].trim().split(/[\s,/]+/).filter(Boolean);if(a.length!==3&&a.length!==4)fail('unsupported-feature','color arity');const channels=a.slice(0,3).map(x=>x.endsWith('%')?finite(x.slice(0,-1))*2.55:finite(x));const alpha=a[3]===undefined?1:a[3].endsWith('%')?finite(a[3].slice(0,-1))/100:finite(a[3]);if(channels.some(x=>x<0||x>255)||alpha<0||alpha>1)fail('invalid-output','color range');return [...channels,alpha];}
  m=/^color\(srgb\s+(.+)\)$/i.exec(value);if(m){const a=m[1].split(/[\s/]+/).map(Number);if(a.length<3||a.length>4||a.some(x=>!Number.isFinite(x)||x<0||x>1))fail('unsupported-feature','color srgb');return [a[0]*255,a[1]*255,a[2]*255,a[3]??1];}
  m=/^#([a-f\d]{3}|[a-f\d]{6}|[a-f\d]{8})$/i.exec(value);if(m){let h=m[1];if(h.length===3)h=[...h].map(x=>x+x).join('');return [parseInt(h.slice(0,2),16),parseInt(h.slice(2,4),16),parseInt(h.slice(4,6),16),h.length===8?parseInt(h.slice(6),16)/255:1];}
  fail('unsupported-feature','unsupported computed color: '+value);
}
export function parsePath(value){
  if(typeof value!=='string'||value.length>262144)fail('limit-exceeded','path bytes');
  const arities={M:2,L:2,H:1,V:1,C:6,S:4,Q:4,T:2,A:7,Z:0};let p=0,command,first=true;const out=[];
  const ws=()=>{while(/\s/.test(value[p]??'')&&p<value.length)p++;};
  while(true){ws();if(p>=value.length)break;let explicit=false;
    if(/[a-z]/i.test(value[p])){command=value[p++];explicit=true;if(!Object.hasOwn(arities,command.toUpperCase()))fail('invalid-output','path command');}
    if(!command)fail('invalid-output','path must start with command');const upper=command.toUpperCase();
    if(first&&upper!=='M')fail('invalid-output','path starts without moveto');first=false;
    if(upper==='Z'){out.push([command]);command=undefined;continue;}
    const vals=[];
    for(let j=0;j<arities[upper];j++){
      ws();if(value[p]===','){if(j===0&&explicit)fail('invalid-output','comma after command');p++;ws();if(value[p]===',')fail('invalid-output','repeated comma');}
      if(upper==='A'&&(j===3||j===4)){if(value[p]!=='0'&&value[p]!=='1')fail('invalid-output','arc flag');vals.push(Number(value[p++]));}
      else {const m=NUMBER.exec(value.slice(p));if(!m)fail('invalid-output','path parameter');vals.push(finite(m[0],'path coordinate'));p+=m[0].length;}
    }
    if(upper==='A'&&(vals[0]<0||vals[1]<0))fail('invalid-output','negative arc radius');
    out.push([command,...vals]);if(out.length>4096)fail('limit-exceeded','path command count');
    if(upper==='M')command=command==='M'?'L':'l';
  }
  return out;
}
export function localMatrix(value){
  if(!value||value==='none')return undefined;
  const m=/^matrix\(([^)]+)\)$/.exec(value);if(!m)fail('unsupported-feature','3D or unknown transform');
  const values=numberList(m[1]);if(values.length!==6)fail('invalid-output','matrix arity');return [{kind:'matrix',values}];
}
export function fontId(value){
  const s=value.toLowerCase().replace(/["']/g,'').replace(/\s+/g,' ').trim();
  const ids=new Map([['arial, sans-serif','arial-sans'],['arial,sans-serif','arial-sans'],['open sans, sans-serif','open-sans-sans'],['open sans,sans-serif','open-sans-sans'],['trebuchet ms, verdana, arial, sans-serif','trebuchet-verdana-arial-sans'],['sans-serif','sans'],['serif','serif'],['monospace','monospace']]);
  if(ids.has(s))return ids.get(s);fail('font-unavailable','unmapped offline font sequence: '+value);
}
