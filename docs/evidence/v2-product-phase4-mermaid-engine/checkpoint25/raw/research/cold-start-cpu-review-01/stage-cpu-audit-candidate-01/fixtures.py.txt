def wire(request='r1'):

                requestId=request, wire=raw, wireBytes=len(raw.encode()), wireSha256=hashlib.sha256(raw.encode()).hexdigest())


def sample(at, ticks, pid=101, birth=11, instance=None):
    return dict(pid=pid, birth=birth, instanceId=instance, status='sample', readStartNs=at,
                readEndNs=at+2, userTicks=ticks, systemTicks=0,
                leaderSchedstat=dict(runtimeNs=ticks*3, waitNs=ticks*2, timeslices=ticks))


def refresh_status(data):
    for sink in data['stageSinks']:
        n = len(sink['events'])
        status = dict(schema='milkup.cold-stage-status.v1', pid=sink['pid'], birth=sink['birth'], role=sink['role'],
                      final=True, closed=True, attempted=n, enqueued=n, written=n,
                      dropped=0, unknownOrUnadmitted=0, ioErrors=0, inFlight=0)
        sink['status'] = copy.deepcopy(status)
        sink['complete'] = status
