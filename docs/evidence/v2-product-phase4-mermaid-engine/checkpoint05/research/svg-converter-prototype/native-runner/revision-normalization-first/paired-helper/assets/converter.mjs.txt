/** Receives a live, attached browser SVG. Never accepts serialized helper claims. */
export function convertSvgElement(root){
 const coverage={declarations:[],attributes:[],rules:[],normalizations:[],errors:[]};

  if(!root||root.namespaceURI!==SVG||root.localName!=='svg'||!root.isConnected)fail('invalid-output','requires attached root SVG');
  const all=[root,...root.querySelectorAll('*')];if(all.length>MAX_NODES)fail('limit-exceeded','input element count');

  const charge=(n=1)=>{charged+=n;if(charged>MAX_NODES)fail('limit-exceeded','charged nodes');};
  const pathOf=n=>paths.get(n)??'?';
  const recurse=(n,p,d)=>{if(d>64)fail('limit-exceeded','input depth',p);paths.set(n,p);computed.set(n,getComputedStyle(n));Array.from(n.children).forEach((c,i)=>recurse(c,p+'/'+i,d+1));};recurse(root,'0',0);

  const definitionKinds={clipPath:'clipPath',marker:'marker',linearGradient:'linearGradient',filter:'dropShadowFilter',symbol:'symbol'};
  for(const n of all)if(definitionKinds[n.localName]){const id=n.getAttribute('id');if(!id)fail('invalid-output','definition without id',pathOf(n));localIds.set(id,'d'+localIds.size);}
  function ref(value,kind,n,computedValue=false){
    let raw=value.trim();const m=/^url\(\s*["']?([^"')]+)["']?\s*\)$/.exec(raw);if(m)raw=m[1];
    // Computed URLs may be expanded by the browser, but authorize only after a
    // matching literal local reference was found in the original declaration ledger.
    if(!raw.startsWith('#')){
      if(computedValue){let u;try{u=new URL(raw,document.URL);}catch{}const here=new URL(document.URL);if(u&&u.origin===here.origin&&u.pathname===here.pathname&&u.search===here.search&&u.hash){const candidate=u.hash;const authorized=coverage.declarations.some(x=>x.path===pathOf(n)&&x.localReferences?.includes(candidate));if(authorized)raw=candidate;}}
      if(!raw.startsWith('#'))fail('unsafe-output','external or unproven reference',pathOf(n));
    }
    const key=raw.slice(1),target=idNodes.get(key);if(!target||!localIds.has(key))fail('invalid-output','missing local definition',pathOf(n));
    if(kind&&definitionKinds[target.localName]!==kind)fail('invalid-output','reference type mismatch',pathOf(n));return localIds.get(key);
  }
  function rawLocalReferences(value,n){
    const found=[];
    for(const match of value.matchAll(/url\(\s*(["']?)(.*?)\1\s*\)/gi)){
      const v=match[2].trim();if(!/^#[A-Za-z0-9_.:-]+$/.test(v))fail('unsafe-output','nonlocal CSS URL',pathOf(n));found.push(v);
    }
    // CSSOM has already decoded escapes, but literal attributes also pass here.
    if(/(?:url|image-set)\s*\(/i.test(value)&&!found.length)fail('unsafe-output','unrecognized resource function',pathOf(n));
    return found;
  }
  function declaration(n,property,value,source){
    const p=property.toLowerCase(),final=css(n,p),record={path:pathOf(n),property:p,declared:value,computed:final,source};
    if(coverage.declarations.length>=20000)fail('limit-exceeded','CSS coverage rows');
    record.localReferences=rawLocalReferences(value,n);
    if(n.namespaceURI===SVG&&['text','tspan'].includes(n.localName)&&geometryProperties.has(p)&&source!=='presentation-attribute')fail('unsupported-feature','CSS text geometry requires explicit support: '+p,pathOf(n));
    if(p.startsWith('--'))record.disposition='custom-property-resolved-at-consumers';
    else if(['white-space-collapse','text-wrap-mode'].includes(p)&&['normal','pre','pre-wrap','nowrap'].includes(css(n,'white-space')))record.disposition='represented-by-effective-white-space';
    else if(styleProperties.has(p)||geometryProperties.has(p)||p==='stop-color'||p==='stop-opacity'||p==='flood-color'||p==='flood-opacity')record.disposition='typed-effective-value';
    else if(p.startsWith('animation')||p.startsWith('transition'))record.disposition='checked-no-effective-motion';
    else if(nonPaintingProperties.has(p))record.disposition='inert-in-image-no-interaction';
    else if(textBoxProperties.has(p)||['max-width','max-height','min-width','min-height','line-height','text-align'].includes(p))record.disposition='checked-box-or-inapplicable-SVG-layout';
    else if(p==='outline-width'&&css(n,'outline-style')==='none')record.disposition='inactive-outline-width';
    else if(neutralProperties.has(p)&&final===neutralProperties.get(p))record.disposition='effective-neutral';
    else {record.disposition='unsupported';coverage.declarations.push(record);fail('unsupported-feature','uncovered CSS declaration/effective value '+p+'='+final,pathOf(n));}
    coverage.declarations.push(record);
  }
  function declarations(n,style,source){for(let i=0;i<style.length;i++){const p=style[i];declaration(n,p,style.getPropertyValue(p),source);}}
  // Enumerate CSSOM. Unknown active at-rules are not silently discarded.
  function rules(list,source){
    for(const rule of Array.from(list)){
      if(rule.type===1){let matched=0;for(const n of all){let yes;try{yes=n.matches(rule.selectorText);}catch{fail('unsupported-feature','unparsed selector');}if(yes){matched++;declarations(n,rule.style,source+':'+rule.selectorText);}}coverage.rules.push({selector:rule.selectorText,matched});}
      else if(rule.type===7){coverage.rules.push({keyframes:rule.name,disposition:'only-discard-if-no-effective-animation'});}
      else if(rule.type===4){if(matchMedia(rule.conditionText).matches)rules(rule.cssRules,source+'@media');else coverage.rules.push({condition:rule.conditionText,disposition:'inactive-media'});}
      else if(rule.type===12){if(CSS.supports(rule.conditionText))rules(rule.cssRules,source+'@supports');else coverage.rules.push({condition:rule.conditionText,disposition:'inactive-supports'});}
      else if(rule.type===3||rule.type===5)fail('unsafe-output','import/font-face resource rule');
      else fail('unsupported-feature','unsupported CSS at-rule type '+rule.type);
    }
  }
  for(const sheet of Array.from(document.styleSheets)){let list;try{list=sheet.cssRules;}catch{fail('unsafe-output','unreadable stylesheet');}rules(list,'stylesheet');}
  for(const n of all){
    if(n.namespaceURI!==SVG&&n.namespaceURI!==XHTML)fail('unsupported-feature','unknown namespace',pathOf(n));
    if(n.namespaceURI===SVG&&!Object.hasOwn(attrsByTag,n.localName))fail(['a','image','script'].includes(n.localName)?'unsafe-output':'unsupported-feature','element '+n.localName,pathOf(n));
    if(n.namespaceURI===XHTML&&n.localName!=='div')fail('unsafe-output','non-textbox XHTML',pathOf(n));
    if(n.style)declarations(n,n.style,'inline');

      if(/^on/i.test(a.name))fail('unsafe-output','event attribute',pathOf(n));

      else if((n.namespaceURI===SVG&&n.localName==='g'&&['x','y'].includes(a.name))||(n.namespaceURI===SVG&&n.localName==='path'&&a.name==='viewBox')){disposition='standard-inapplicable-attribute';coverage.normalizations.push({path:pathOf(n),kind:'standard-inapplicable-attribute',tag:n.localName,attribute:a.name,value:a.value,standard:n.localName==='g'?'https://www.w3.org/TR/SVG2/struct.html#GElement':'https://www.w3.org/TR/SVG2/paths.html#PathElement',viewportStandard:'https://www.w3.org/TR/SVG2/coords.html#ViewBoxAttribute'});}
      else if(globalAttrs.has(a.name)||a.name.startsWith('data-'))disposition='consumed-selector-or-metadata';
      else if((attrsByTag[n.localName]??'').split(' ').includes(a.name))disposition='typed-geometry-definition';
      else fail('unsupported-feature','unknown attribute '+a.name,pathOf(n));
      coverage.attributes.push({path:pathOf(n),name:a.name,disposition});
    }
    const motion=css(n,'animation-name');
    if(motion&&motion!=='none')fail('unsupported-animation','effective CSS animation',pathOf(n));
    if(typeof n.getAnimations==='function'&&n.getAnimations().length)fail('unsupported-animation','active browser animation',pathOf(n));
    if(css(n,'transition-duration').split(',').some(x=>parseFloat(x)>0))fail('unsupported-animation','effective transition duration',pathOf(n));
    for(const [p,neutral] of neutralProperties){const value=css(n,p);if(value&&value!==neutral){if(p==='outline-width'&&css(n,'outline-style')==='none'){coverage.normalizations.push({path:pathOf(n),kind:'inactive-outline-width',value});continue;}if(p==='font-stretch'&&value==='normal')continue;fail('unsupported-feature','effective '+p+'='+value,pathOf(n));}}
    for(const p of ['background-image','mask-image'])if(css(n,p)&&css(n,p)!=='none')fail('unsafe-output','effective image resource '+p,pathOf(n));
  }
  const vb=numberList(root.getAttribute('viewBox'));if(vb.length!==4||vb[2]<=0||vb[3]<=0)fail('invalid-output','viewBox');
  const viewport={x:vb[2],y:vb[3],r:Math.hypot(vb[2],vb[3])/Math.SQRT2};
  function viewportFor(n){
    for(let parent=n.parentElement;parent&&parent!==root;parent=parent.parentElement){
      if(parent.localName==='clipPath'&&attr(parent,'clipPathUnits','userSpaceOnUse')==='objectBoundingBox')return {x:1,y:1,r:1};
      if(['marker','symbol','svg'].includes(parent.localName)){
        const local=parent.hasAttribute('viewBox')?numberList(parent.getAttribute('viewBox')):null;
        if(local?.length===4)return {x:local[2],y:local[3],r:Math.hypot(local[2],local[3])/Math.SQRT2};
        if(parent.localName==='marker'){const x=finite(attr(parent,'markerWidth','3')),y=finite(attr(parent,'markerHeight','3'));return {x,y,r:Math.hypot(x,y)/Math.SQRT2};}
        fail('unsupported-feature','percentage geometry in unresolved nested viewport',pathOf(n));
      }
    }
    return viewport;
  }
  function scalar(n,key,def=0,axis='x'){
    let value=css(n,key);if(!value||value==='auto')value=attr(n,key,String(def));
    const l=length(value);if(l.unit==='percent')return finite(l.value*viewportFor(n)[axis]/100,'resolved geometry');
    if(l.unit==='em')return finite(l.value*parseFloat(css(n,'font-size')),'resolved em geometry');return l.value;
  }
  function paint(value,n){if(value==='none')return {kind:'none'};if(value.startsWith('url('))return {kind:'gradient',ref:ref(value,'linearGradient',n,true)};return {kind:'color',rgba:color(value)};}
  function filter(value,n){
    if(!value||value==='none')return {kind:'none'};
    if(value.startsWith('url('))return {kind:'definition',ref:ref(value,'dropShadowFilter',n,true)};
    let rest=value,shadows=[];
    while(rest){if(!rest.startsWith('drop-shadow('))fail('unsupported-feature','filter function',pathOf(n));let depth=1,i=12;for(;i<rest.length&&depth;i++){if(rest[i]==='(')depth++;if(rest[i]===')')depth--;}if(depth)fail('invalid-output','filter parentheses');let body=rest.slice(12,i-1).trim();let colorToken;
      const cm=/^(rgba?\([^)]*\)|color\([^)]*\)|#[a-f\d]+)\s*/i.exec(body);
      if(cm){colorToken=cm[1];body=body.slice(cm[0].length);}else {const tail=/(rgba?\([^)]*\)|color\([^)]*\)|#[a-f\d]+)$/i.exec(body);if(!tail)fail('unsupported-feature','drop shadow color');colorToken=tail[1];body=body.slice(0,tail.index).trim();}
      const v=body.split(/\s+/).map(x=>length(x));if(v.length<2||v.length>3||v.some(x=>!['px','user'].includes(x.unit)))fail('unsupported-feature','drop shadow lengths');if((v[2]?.value??0)<0)fail('invalid-output','negative blur');if((v[2]?.value??0)>32)fail('limit-exceeded','shadow blur');shadows.push({dx:v[0].value,dy:v[1].value,blur:v[2]?.value??0,color:color(colorToken)});if(shadows.length>4)fail('limit-exceeded','shadow count');rest=rest.slice(i).trim();
    }
    return {kind:'cssDropShadow',shadows};
  }
  function style(n){
    const c=p=>css(n,p),len=p=>length(c(p)),a={};
    a.fill=paint(c('fill')||'rgb(0, 0, 0)',n);a.stroke=paint(c('stroke')||'none',n);
    for(const [key,p] of [['fillOpacity','fill-opacity'],['strokeOpacity','stroke-opacity'],['opacity','opacity']])a[key]=finite(c(p)||1,p,1);
    a.fillRule=ENUM(c('fill-rule')||'nonzero',['nonzero','evenodd'],'fill rule');a.clipRule=ENUM(c('clip-rule')||'nonzero',['nonzero','evenodd'],'clip rule');
    a.strokeWidth=len('stroke-width');a.dashArray=c('stroke-dasharray')==='none'?[]:c('stroke-dasharray').split(/[\s,]+/).filter(Boolean).map(x=>length(x));a.dashOffset=len('stroke-dashoffset');
    a.lineCap=ENUM(c('stroke-linecap'),['butt','round','square'],'line cap');a.lineJoin=ENUM(c('stroke-linejoin'),['miter','round','bevel'],'line join');a.miterLimit=finite(c('stroke-miterlimit'));
    a.display=c('display')==='none'?'none':'inline';a.visibility=ENUM(c('visibility'),['visible','hidden','collapse'],'visibility');
    a.overflow=ENUM(c('overflow'),['visible','hidden'],'overflow');a.mixBlendMode=ENUM(c('mix-blend-mode')||'normal',['normal','multiply'],'blend');a.isolation=ENUM(c('isolation')||'auto',['auto','isolate'],'isolation');a.filter=filter(c('filter'),n);
    if(c('clip-path')&&c('clip-path')!=='none')a.clipPath=ref(c('clip-path'),'clipPath',n,true);
    a.colorInterpolation=interpolation(c('color-interpolation')||'sRGB');a.colorInterpolationFilters=interpolation(c('color-interpolation-filters')||'linearRGB');
    a.fontId=fontId(c('font-family'));a.fontSize=len('font-size');const weight=c('font-weight')==='normal'?400:c('font-weight')==='bold'?700:Number(c('font-weight'));a.fontWeight=ENUM(weight,[100,200,300,400,500,600,700,800,900],'font weight');a.fontStyle=ENUM(c('font-style'),['normal','italic','oblique'],'font style');
    for(const [key,p] of [['letterSpacing','letter-spacing'],['wordSpacing','word-spacing']]){if(c(p)!=='normal')a[key]=len(p);else if(n.parentElement&&computed.has(n.parentElement)&&css(n.parentElement,p)!=='normal')fail('unsupported-feature','normal spacing reset requires typed normal',pathOf(n));}
    const rendering={'auto':'auto','optimizespeed':'optimizeSpeed','crispedges':'crispEdges','geometricprecision':'geometricPrecision'};a.shapeRendering=rendering[(c('shape-rendering')||'auto').toLowerCase()];if(!a.shapeRendering)fail('unsupported-feature','shape rendering',pathOf(n));
    a.textAnchor=ENUM(c('text-anchor')||'start',['start','middle','end'],'text anchor');a.dominantBaseline=ENUM(c('dominant-baseline')||'auto',['auto','alphabetic','central','middle','hanging','text-before-edge','text-after-edge'],'dominant baseline');a.alignmentBaseline=ENUM(c('alignment-baseline')||'auto',['auto','baseline','alphabetic','central','middle','hanging','before-edge','after-edge'],'alignment baseline');
    a.decoration=ENUM(c('text-decoration-line')||'none',['none','underline','overline','line-through'],'decoration');a.textTransform=ENUM(c('text-transform')||'none',['none','uppercase','lowercase','capitalize'],'text transform');a.whiteSpace=ENUM(c('white-space')||'normal',['normal','pre','pre-wrap','nowrap'],'whitespace');
    if(n===root&&color(c('background-color')||'rgba(0,0,0,0)')[3]!==0)fail('unsupported-feature','root background paint',pathOf(n));
    const key=JSON.stringify(a);let index=styleMap.get(key);if(index===undefined){charge();index=styles.length;styles.push(a);styleMap.set(key,index);}return index;
  }
  function effectiveLocalTransform(n){
    const declared=css(n,'transform');
    if((!declared||declared==='none')&&!n.hasAttribute('transform'))return undefined;
    const parent=n.parentElement;
    if(n===root||!parent||typeof n.getCTM!=='function'||typeof parent.getCTM!=='function')fail('unsupported-feature','local transform without comparable CTM',pathOf(n));
    if(n.localName==='svg'||n.localName==='symbol'||(parent!==root&&n.ownerSVGElement!==parent.ownerSVGElement))fail('unsupported-feature','transform crosses SVG viewport',pathOf(n));
    const own=n.getCTM(),up=parent.getCTM();if(!own||!up)fail('unsupported-feature','null local CTM',pathOf(n));
    const valuesOf=m=>[m.a,m.b,m.c,m.d,m.e,m.f];if([...valuesOf(own),...valuesOf(up)].some(v=>!Number.isFinite(v))||Math.abs(up.a*up.d-up.b*up.c)<1e-12)fail('unsupported-feature','noninvertible parent CTM',pathOf(n));
    let m;try{m=up.inverse().multiply(own);}catch{fail('unsupported-feature','parent CTM inverse failed',pathOf(n));}
    const values=valuesOf(m).map(v=>finite(v,'local matrix'));
    coverage.normalizations.push({path:pathOf(n),kind:'same-viewport-relative-CTM-origin-baked-once',attribute:n.getAttribute('transform'),computed:declared,parent:valuesOf(up),own:valuesOf(own),local:values});
    if(values.every((v,i)=>v===[1,0,0,1,0,0][i]))return undefined;
    return [{kind:'matrix',values}];
  }
  function common(n){
    const out={style:style(n)},transforms=effectiveLocalTransform(n);
    if(transforms){charge(transforms.length);out.transforms=transforms;}
    const title=Array.from(n.children).find(x=>x.localName==='title'),desc=Array.from(n.children).find(x=>x.localName==='desc');if(title)out.title=title.textContent;if(desc)out.description=desc.textContent;return out;
  }
  function marked(n){const out=common(n);for(const [key,p] of [['markerStart','marker-start'],['markerMid','marker-mid'],['markerEnd','marker-end']]){const v=css(n,p);if(v&&v!=='none')out[key]=ref(v,'marker',n,true);}return out;}
  function text(n){const out={kind:n.localName,...common(n),children:[]};for(const key of ['x','y','dx','dy']){const v=lengthList(attr(n,key));if(v)out[key]=v;}if(n.hasAttribute('xml:space'))out.xmlSpace=ENUM(n.getAttribute('xml:space'),['default','preserve'],'xml space');
    for(const child of Array.from(n.childNodes)){if(child.nodeType===3){charge();out.children.push({kind:'textRun',text:child.data});}else if(child.nodeType===1&&child.localName==='tspan'){charge();out.children.push(text(child));}else if(child.nodeType===1&&!['title','desc'].includes(child.localName))fail('unsupported-feature','text child',pathOf(n));}
    return out;}
  function textBox(n){
    const outer=n.firstElementChild,inner=outer?.firstElementChild;
    if(n.children.length!==1||outer?.namespaceURI!==XHTML||outer.localName!=='div'||outer.children.length!==1||inner?.localName!=='div'||inner.namespaceURI!==XHTML||inner.children.length)fail('unsafe-output','textbox must be fixed div/div/plain-text',pathOf(n));
    if(css(outer,'display')!=='table'||css(inner,'display')!=='table-cell')fail('unsupported-feature','textbox layout');
    const box=[scalar(n,'x'),scalar(n,'y',0,'y'),scalar(n,'width'),scalar(n,'height',0,'y')];
    for(const node of [outer,inner]){
      for(const [dimension,index] of [['width',2],['height',3]]){const raw=css(node,dimension);if(!/^[0-9.]+px$/.test(raw)||Math.abs(finite(parseFloat(raw))-box[index])>0.01)fail('unsupported-feature','textbox effective '+dimension+' differs from fixed container',pathOf(node));}
      for(const p of ['overflow-x','overflow-y','visibility'])if(css(node,p)!=='visible')fail('unsupported-feature','textbox '+p,pathOf(node));
