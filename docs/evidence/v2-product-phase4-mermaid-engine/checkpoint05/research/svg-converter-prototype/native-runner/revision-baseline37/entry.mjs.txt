// External one-case research entry. Native owner alone enforces deadline/network/transport limits.
import mermaid from './mermaid-full-12.2.0.mjs';
import {convertSvgElement} from './converter.mjs';
const base={startOnLoad:false,securityLevel:'strict',htmlLabels:false,suppressErrorRendering:true,maxTextSize:50000,maxEdges:500,deterministicIds:true,deterministicIDSeed:'offline-audit',fontFamily:'Arial, sans-serif',legacyMathML:false,forceLegacyMathML:false,secure:['secure','securityLevel','startOnLoad','maxTextSize','maxEdges','suppressErrorRendering','htmlLabels','fontFamily','themeCSS','legacyMathML','forceLegacyMathML','look','textPlacement','theme','darkMode']};
let preparation={status:'idle'},preparing,busy=false,sequence=0,task,host;
const encoder=new TextEncoder();
const textError=e=>String(e instanceof Error?e.message:e).slice(0,4096);
function ready(){if(preparing)return preparing;preparation={status:'pending'};return preparing=Promise.resolve(document.fonts?.ready).then(()=>{preparation={status:'ready',engine:'mermaid@12.2.0',converter:'external-prototype-only'};return preparation;},e=>{preparation={status:'error',error:textError(e)};throw e;});}
function readiness(){return preparation;}
function validate(r){if(!r||typeof r!=='object'||Array.isArray(r)||Object.keys(r).some(k=>!['id','source','profile','theme'].includes(k)))throw Error('invalid request');if(typeof r.id!=='string'||!/^[A-Za-z0-9_-]{1,96}$/.test(r.id)||typeof r.source!=='string'||r.source.length>50000||!['baseline','classic-tspan'].includes(r.profile)||!['light','dark'].includes(r.theme))throw Error('invalid request values');}
// Compact ledger only changes research encoding; the independent validator never trusts it.
function packCoverage(c){const strings=[],ids=new Map(),intern=s=>{s=String(s??'');let id=ids.get(s);if(id===undefined){id=strings.length;ids.set(s,id);strings.push(s);}return id;};return {strings,
 declarationColumns:['path','property','declared','computed','source','disposition','localReferences'],
 declarations:c.declarations.map(r=>[intern(r.path),intern(r.property),intern(r.declared),intern(r.computed),intern(r.source),intern(r.disposition),(r.localReferences??[]).map(intern)]),
 attributes:c.attributes.map(r=>[intern(r.path),intern(r.name),intern(r.disposition)]),
 rules:c.rules.map(r=>intern(JSON.stringify(r))),normalizations:c.normalizations.map(r=>intern(JSON.stringify(r))),errors:c.errors};}
function attachBytes(value){value.serializedBytes=0;value.over256KiB=false;for(let i=0;i<8;i++){const n=encoder.encode(JSON.stringify(value)).length,over=n>262144;if(n===value.serializedBytes&&over===value.over256KiB)break;value.serializedBytes=n;value.over256KiB=over;}return value;}
async function run(request){validate(request);if(preparation.status!=='ready')throw Error('prepare first');if(busy)throw Error('busy');busy=true;const started=performance.now();try{
 host?.remove();host=document.createElement('div');host.id='svg-converter-prototype-host';host.style.width='1000px';document.body.append(host);
 mermaid.initialize({...base,...(request.theme==='dark'?{theme:'dark',darkMode:true}:{}),...(request.profile==='classic-tspan'?{look:'classic',journey:{textPlacement:'tspan'}}:{})});
 const {svg,diagramType}=await mermaid.render('convert_'+request.id,request.source,host);const renderMs=performance.now()-started;host.innerHTML=svg;
 const result=convertSvgElement(host.querySelector('svg'));
 return attachBytes({...result,coverage:packCoverage(result.coverage),id:request.id,profile:request.profile,theme:request.theme,diagramType,originalSvgBytes:encoder.encode(svg).length,renderMs,elapsedMs:performance.now()-started,
 evidenceBoundary:'No independent validator or round-trip pixels run by this entry. Original live SVG remains in helper for separately authorized observation.'});
 }catch(e){return attachBytes({ok:false,id:request.id,error:{code:'invalid-output',reason:textError(e)},elapsedMs:performance.now()-started});}finally{busy=false;}}
function start(request){validate(request);if(preparation.status!=='ready')throw Error('prepare first');if(busy||task?.status==='pending')throw Error('busy');const taskId='conversion_'+(++sequence);task={taskId,status:'pending'};Promise.resolve().then(()=>run(request)).then(result=>{task={taskId,status:'complete',result};},e=>{task={taskId,status:'error',error:textError(e)};});return {taskId,status:'pending'};}
function poll(taskId){return task?.taskId===taskId?task:{status:'unknown'};}
function clear(){if(busy||task?.status==='pending')throw Error('pending: native termination required');task=undefined;host?.remove();host=undefined;}
window.__milkupSvgConverterProbe=Object.freeze({ready,readiness,run,start,poll,clear});
export {ready,readiness,run,start,poll,clear};
