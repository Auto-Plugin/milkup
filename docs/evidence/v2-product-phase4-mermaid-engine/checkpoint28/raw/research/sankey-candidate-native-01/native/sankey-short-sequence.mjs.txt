function assertCapturedPreview(dom, expected, stage) {
  const actual = dom.previews.find((item) => item.from === expected.from && item.to === expected.to)
  assert(actual && visible(actual.rect, dom), stage + ': same range remains visible')
  assert.equal(actual.inlineSvg, 0, stage + ': no injected inline SVG')
  assert.equal(actual.images.length, expected.images.length, stage + ': all images remain mounted')
  assert(
    actual.images.every(
      (item) =>
        item.src.startsWith('blob:') &&
        item.complete &&
        item.naturalWidth > 0 &&
        item.naturalHeight > 0,
    ),
    stage + ': same range remains fully decoded',
  )
}
function assertCapturedScrollEdge(dom, expected, stage) {
  assert(expected.after, stage + ': reached corner snapshot exists')
  assert.equal(expected.after.from, expected.from, stage + ': expected range start')
  assert.equal(expected.after.to, expected.to, stage + ': expected range end')
  assertCapturedPreview(dom, expected.after, stage)
  const actual = dom.previews.find((item) => item.from === expected.from && item.to === expected.to)
  assert(
    actual.images.length > 0 && actual.images.every((item) => item.width > 0 && item.height > 0),
    stage + ': mounted image dimensions remain positive',
  )
  assert(['bottom-right', 'top-left'].includes(expected.corner), stage + ': known corner')
  for (const [axis, extent, client] of [
    ['left', 'scrollWidth', 'clientWidth'],
    ['top', 'scrollHeight', 'clientHeight'],
  ]) {
    for (const snapshot of [expected.after, actual]) {
      assert(
        [snapshot[axis], snapshot[extent], snapshot[client]].every(Number.isFinite) &&
          snapshot[client] > 0 &&
          snapshot[extent] > 0,
        stage + ': finite positive scroll geometry ' + axis,
      )
    }
    const previousMax = Math.max(0, expected.after[extent] - expected.after[client])
    const currentMax = Math.max(0, actual[extent] - actual[client])
    assert(
      previousMax <= 1 || currentMax > 1,
      stage + ': previously overflowing axis remains scrollable ' + axis,
    )
    const corner = expected.corner === 'bottom-right' ? currentMax : 0
    const clamped = Math.min(currentMax, Math.max(0, expected.after[axis]))
    assert(
      Math.abs(actual[axis] - corner) <= 1,
      stage + ': remains at ' + expected.corner + ' ' + axis,
    )
    assert(Math.abs(actual[axis] - clamped) <= 1, stage + ': preserves reached offset ' + axis)
  }
}
function assertCapturedScrollCorner(dom, expected, stage) {
  const actual = dom.previews.find((item) => item.from === expected.from && item.to === expected.to)
  assert(actual, stage + ': corner preview remains mounted')
  const rect = actual.clientRect,
    viewport = dom.editorViewport
  assert(
    rect &&
      viewport &&
      [
        rect.left,
        rect.right,
        rect.top,
        rect.bottom,
        viewport.left,
        viewport.right,
        viewport.top,
        viewport.bottom,
      ].every(Number.isFinite) &&
      rect.right > rect.left &&
      rect.bottom > rect.top &&
      viewport.right > viewport.left &&
      viewport.bottom > viewport.top,
    stage + ': finite nonempty corner client geometry',
  )
  if (expected.corner === 'bottom-right') {
    assert(
      rect.right <= viewport.right && rect.right > viewport.left,
      stage + ': right corner is inside editor viewport',
    )
    assert(
      rect.bottom <= viewport.bottom && rect.bottom > viewport.top,
      stage + ': bottom corner is inside editor viewport',
    )
  } else {
    assert.equal(expected.corner, 'top-left', stage + ': known visible corner')
    assert(
      rect.left >= viewport.left && rect.left < viewport.right,
      stage + ': left corner is inside editor viewport',
    )
    assert(
      rect.top >= viewport.top && rect.top < viewport.bottom,
      stage + ': top corner is inside editor viewport',
    )
  }
}
function noticeIsVisible(notice, dom) {
  const rect = notice.rect
  return (
    notice.display !== 'none' &&
    notice.visibility === 'visible' &&
    notice.opacity > 0 &&
    rect.width > 0 &&
    rect.height > 0 &&
    rect.top >= Math.max(0, dom.editor.top) &&
    rect.bottom <= Math.min(dom.height, dom.editor.bottom) &&
    rect.left >= Math.max(0, dom.editor.left) &&
    rect.right <= Math.min(dom.width, dom.editor.right)
  )
}
function assertCapturedNotice(dom, expected, stage) {
  assert(
    dom.noticeBoxes.some((notice) => notice.text === expected && noticeIsVisible(notice, dom)),
    stage + ': the current isolated fallback notice is fully visible',
  )
  assert.equal(dom.previews.length, 0, stage + ': single fallback remains full source')
}
let assertionFailureSequence = 0
async function preserveAssertion(kind, dom, expected, stage, document = null) {
  try {
    if (kind === 'preview') assertCapturedPreview(dom, expected, stage)
    else if (kind === 'scroll-edge-offset') assertCapturedScrollEdge(dom, expected, stage)
    else if (kind === 'scroll-edge') {
      assertCapturedScrollEdge(dom, expected, stage)
      assertCapturedScrollCorner(dom, expected, stage)
    } else assertCapturedNotice(dom, expected, stage)
  } catch (error) {
    const file =
      'assertion-failure-' +
      String(++assertionFailureSequence).padStart(3, '0') +
      '-' +
      kind +
      '-' +
      stage.replaceAll(' ', '-') +
      '.json'
    // Preserve the observations already made by the failing check. A before-shot
    // check has no document probe; do not add one and change the observed timing.
    await writeFile(
      path.join(output, file),
      JSON.stringify(
        {
          recordedAt: new Date().toISOString(),
          kind,
          stage,
          document,
          documentBoundary:
            document === null
              ? 'Not read at this stage; no extra browser observation was made.'
              : 'Already-read public document probe from this record call.',
          dom,
          expected,
          error: {
            message: String(error),
            stack: error.stack,
            actual: error.actual,
            actualIsUndefined: error.actual === undefined,
            expected: error.expected,
            operator: error.operator,
          },
        },
        null,
        2,
      ) + '\n',
      { flag: 'wx' },
    ).catch((writeError) => {
      console.error('Failed to preserve assertion snapshot ' + file + ': ' + String(writeError))
    })
    throw error
  }
}
async function record(name, extra = {}) {
  const document = await app.probe()
  const dom = await observe()
  if (extra.preview)
    await preserveAssertion('preview', dom, extra.preview, 'after screenshot', document)
  if (extra.scrollEdge)
    await preserveAssertion('scroll-edge', dom, extra.scrollEdge, 'after screenshot', document)
  if (extra.fallbackNotice) {
    assert.equal(document.path, activeFile)
    await preserveAssertion('notice', dom, extra.fallbackNotice, 'after screenshot', document)
  }
  const currentRegistry = await registry()
  const audit = await diagramsAudit()
  if (extra.coldBundled) {
    const provider = currentRegistry.find((row) => row.id === 'milkup.diagrams')
    assert(provider?.sourcePath.startsWith('bundled:'))
    assert.equal(provider.state, 'enabled')
    assert.equal(provider.enabled, true)
    assert.deepEqual(provider.errors, [])
    assert(
      !audit.some((item) => item.denied && /execution deadline/i.test(item.detail || '')),
      'Cold bundled instance must not have a deadline-denied audit',
    )
  }
  results.push({
    name,
    at: new Date().toISOString(),
    ...extra,
    document,
    dom,
    registry: currentRegistry,
    diagramsAudit: audit,
  })
  await writeFile(journal, JSON.stringify(results, null, 2) + '\n')
  console.log('PASS ' + name)
}

    await preserveAssertion('preview', await observe(), extra.preview, 'before screenshot')

    await preserveAssertion('scroll-edge', await observe(), extra.scrollEdge, 'before screenshot')
  if (extra.fallbackNotice)
    await preserveAssertion('notice', await observe(), extra.fallbackNotice, 'before screenshot')
  await app.shot(name + '-webview')
  execFileSync('import', ['-window', 'root', path.join(output, name + '-x11.png')], {
    env: process.env,
  })
  const screenshots = await Promise.all(
    ['webview', 'x11'].map(async (kind) => {
      const file = name + '-' + kind + '.png',
        bytes = await readFile(path.join(output, file))
      return { file, bytes: bytes.length, sha256: sha256(bytes), reviewed: false }
    }),
  )
  await record(name, { ...extra, screenshots })
}

async function sampleSelectionProjection(sample, phase) {
  const markerTo = sample.from - 2
  const markerFrom = markerTo - sample.marker.length
  const fixture = sample.name ? contents.get(sample.name) : source
  assert.equal(fixture.slice(markerFrom, markerTo), sample.marker)
  const markerLine = fixture.slice(0, markerFrom).split('\n').length
  // Read lines and overlays in one task. Windowed rows may be unmounted even
  // while the logical selection remains unchanged; overscan rows still count.
  const projection = await app.execute(
    `
const editor=document.querySelector('.milkup-editor');
if(!editor) throw new Error('Selection projection: mounted editor required');
const rows=[...editor.querySelectorAll('.milkup-editor-content .milkup-line')]
  .map(e=>({from:Number(e.dataset.from),to:Number(e.dataset.to),line:Number(e.dataset.line)}));
return {
  windowed:editor.classList.contains('milkup-source-document-view'),
  mode:editor.dataset.mode,
  invalidRows:rows.filter(row=>!Number.isInteger(row.from)||!Number.isInteger(row.to)||!Number.isInteger(row.line)||row.from<0||row.to<row.from||row.line<1),
  markerLines:rows.filter(line=>line.from<arguments[1]&&line.to>arguments[0]),
  ranges:[...editor.querySelectorAll('.milkup-selection')].map(e=>({from:Number(e.dataset.from),to:Number(e.dataset.to),line:e.dataset.line===undefined?null:Number(e.dataset.line)})),
  cursors:[...editor.querySelectorAll('.milkup-cursor')].map(e=>Number(e.dataset.position))
};
`,
    markerFrom,
    markerTo,
  )
  await appendFile(
    path.join(output, 'selection-projections.jsonl'),
    JSON.stringify({
      at: new Date().toISOString(),
      sample: sample.id,
      phase,
      markerFrom,
      markerTo,
      markerLine,
      ...projection,
    }) + '\n',
  )
  assert.deepEqual(projection.invalidRows, [], 'Mounted source rows have valid source metadata')
  assert(projection.markerLines.length <= 1, 'Unique marker has at most one mounted source row')
  for (const line of projection.markerLines) {
    assert.equal(line.from, markerFrom, 'Mounted marker starts at its exact source offset')
    assert.equal(line.to, markerTo, 'Mounted marker ends at its exact source offset')
    assert.equal(line.line, markerLine, 'Mounted marker has its exact fixture source line')
  }
  if (!projection.windowed)
    assert.equal(projection.markerLines.length, 1, 'Ordinary fixture keeps its marker row mounted')
  for (const range of projection.ranges) {
    assert(
      Number.isInteger(range.from) && Number.isInteger(range.to) && range.from < range.to,
      'Every selection projection has valid source endpoints',
    )
    if (projection.windowed)
      assert(Number.isInteger(range.line), 'Every windowed projection has a source line')
  }
  const normalized = (ranges) => [...new Set(ranges.map((range) => JSON.stringify(range)))].sort()
  // Ordinary overlays carry the full range without data-line. SourceDocumentView
  // overlays carry intersections with mounted rows, including their line number.
  const expected = projection.windowed
    ? projection.markerLines.map((line) => ({
        from: Math.max(markerFrom, line.from),
        to: Math.min(markerTo, line.to),
        line: line.line,
      }))
    : [{ from: markerFrom, to: markerTo }]
  const actual = projection.ranges.map(({ from, to, line }) =>
    projection.windowed ? { from, to, line } : { from, to },
  )
  assert.deepEqual(
    normalized(actual),
    normalized(expected),
    'Selection projections must match the exact marker on currently mounted source rows',
  )
  assert.deepEqual(
    projection.cursors,
    projection.markerLines.length ? [markerTo] : [],
    'Cursor projection must match the mounted marker endpoint',
  )
  return { ...projection, expected }
}
async function positionSample(sample) {
  const before = await app.probe()
  const projectionBefore = await sampleSelectionProjection(sample, 'before-positioning')
  const fileBefore = await readFile(activeFile)
  const started = Date.now()
  const observations = []
  const fences = (sample.name ? [sample] : samples).map(({ from, to }) => ({ from, to }))
  const readPosition = () =>
    app.execute(
      `
const editor = document.querySelector('.milkup-editor');
const content = document.querySelector('.milkup-editor-content');
if (!editor || !content) throw new Error('Positioning: mounted editor required');
const root = editor.getBoundingClientRect(), owners = new Map(), measured = new Map();
const clientLeft=root.left+editor.clientLeft, clientTop=root.top+editor.clientTop;
const viewport = {top:Math.max(0,clientTop),bottom:Math.min(innerHeight,clientTop+editor.clientHeight),left:Math.max(0,clientLeft),right:Math.min(innerWidth,clientLeft+editor.clientWidth)};
const intersects = r => r.height>0 && r.width>0 && r.bottom>viewport.top && r.top<viewport.bottom && r.right>viewport.left && r.left<viewport.right;
const lines = [...content.querySelectorAll('.milkup-line')];
const effective = line => {
  const id = line.dataset.presentationOwner;
  if (id && !owners.has(id)) owners.set(id, content.querySelector('.milkup-line[data-line="'+CSS.escape(id)+'"]'));
  const owner = (id ? owners.get(id) : undefined) || line;
  if (!measured.has(owner)) measured.set(owner,owner.getBoundingClientRect());
  return measured.get(owner);
};
const fences = arguments[0].map(fence => {
  const matched = lines.filter(line => {
    const from=Number(line.dataset.from),to=Number(line.dataset.to);
    const blockFrom=Number(line.dataset.presentationBlockFrom),blockTo=Number(line.dataset.presentationBlockTo);
    return (blockFrom===fence.from && blockTo===fence.to) || (Number.isFinite(from)&&Number.isFinite(to)&&from<fence.to&&to>=fence.from);
  });
  const bounds = matched.map(effective).filter(r=>r.width>0&&r.height>0);
  return {from:fence.from,to:fence.to,mounted:matched.length>0,visible:bounds.some(intersects),top:bounds.length?Math.min(...bounds.map(r=>r.top)):null,bottom:bounds.length?Math.max(...bounds.map(r=>r.bottom)):null};
});
const target = fences.find(f=>f.from===arguments[1]&&f.to===arguments[2]);
const preceding = fences.filter(f=>f.to<=arguments[1]&&f.visible);
const contentRight=root.left+editor.clientLeft+editor.clientWidth;
const py=viewport.top+Math.min(40,(viewport.bottom-viewport.top)/3);
let gutter;
for(const px of [Math.max(viewport.left,root.left+editor.clientLeft)+4,Math.min(viewport.right,contentRight)-4]) {
  const hit=document.elementFromPoint(px,py);
  if(px>viewport.left&&px<contentRight&&hit&&editor.contains(hit)&&!hit.closest('[data-document-presentation],input,textarea,button,[role="scrollbar"]')) {gutter={x:px,y:py};break;}
}
const offsets=lines.map(line=>Number(line.dataset.from)).filter(Number.isFinite);
return {width:innerWidth,viewport,scrollTop:editor.scrollTop,scrollHeight:editor.scrollHeight,clientHeight:editor.clientHeight,target,preceding:preceding.map(f=>f.from),firstVisible:fences.find(f=>f.visible)?.from,gutter,mountedFrom:offsets.length?Math.min(...offsets):null,mountedTo:offsets.length?Math.max(...offsets):null};
`,
      fences,
      sample.from,
      sample.to,
    )
  let ticks = 0
  let reversals = 0
  let previousDirection
  let completed = false
  let completionReason
  let bottomCandidate
  let failure
  try {
    while (Date.now() - started < 10_000) {
      await app.delay(100)
      const position = await readPosition()
      observations.push({ elapsedMs: Date.now() - started, ticks, reversals, ...position })
      assert(position.target, 'Positioning: exact target fence required')
      assert(Date.now() - started < 10_000, 'Positioning: ten-second bound reached')
      if (position.target.visible && position.preceding.length === 0) {
        completed = true
        completionReason = 'first-visible-eligible-fence'
        break
      }
      if (
        position.target.visible &&
        position.scrollTop + position.clientHeight >= position.scrollHeight - 1
      ) {
        const stableBottom =
          bottomCandidate &&
          bottomCandidate.scrollHeight === position.scrollHeight &&
          bottomCandidate.clientHeight === position.clientHeight &&
          Math.abs(bottomCandidate.scrollTop - position.scrollTop) <= 1 &&
          bottomCandidate.target.from === position.target.from &&
          bottomCandidate.target.to === position.target.to &&
          Math.abs(bottomCandidate.target.top - position.target.top) <= 1 &&
          Math.abs(bottomCandidate.target.bottom - position.target.bottom) <= 1 &&
          ['top', 'bottom', 'left', 'right'].every(
            (key) => Math.abs(bottomCandidate.viewport[key] - position.viewport[key]) <= 1,
          ) &&
          JSON.stringify(bottomCandidate.preceding) === JSON.stringify(position.preceding)
        if (stableBottom) {
          completed = true
          completionReason = 'physical-scroll-end'
          break
        }
        bottomCandidate = position
        continue
      }
      bottomCandidate = undefined
      assert(ticks < 20, 'Positioning: twenty-wheel-tick bound reached')
      assert(position.gutter, 'Positioning: non-preview editor gutter hit required')
      let direction
      if (
        position.preceding.length ||
        (position.target.top !== null && position.target.top >= position.viewport.bottom)
      )
        direction = '5'
      else if (position.target.bottom !== null && position.target.bottom <= position.viewport.top)
        direction = '4'
      else if (
        !position.target.mounted &&
        position.mountedTo !== null &&
        position.mountedTo < sample.from
      )
        direction = '5'
      else if (
        !position.target.mounted &&
        position.mountedFrom !== null &&
        position.mountedFrom > sample.to
      )
        direction = '4'
      else throw new Error('Positioning: target has no usable mounted owner geometry')
      if (previousDirection && direction !== previousDirection) reversals++
      assert(reversals <= 2, 'Positioning: two-direction-reversal bound reached')
      const scale =
        Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', nativeWindow))[1]) /
        position.width
      assert(Number.isFinite(scale) && scale > 0, 'Positioning: actual native scale required')
      assert(Date.now() - started < 10_000, 'Positioning: ten-second bound reached')
      x(
        'mousemove',
        '--window',
        nativeWindow,
        String(Math.round(position.gutter.x * scale)),
        String(Math.round(position.gutter.y * scale)),
        'click',
        direction,
      )
      previousDirection = direction
      ticks++
    }
    assert(
      completed,
      'Positioning: target must be first visible or at stable physical scroll end within ten seconds',
    )
  } catch (error) {
    failure = error
  }
  const after = await app.probe()
  const projectionAfter = await sampleSelectionProjection(sample, 'after-positioning').catch(
    (error) => {
      if (!failure) throw error
      // Keep an earlier positioning failure authoritative while retaining this
      // additional projection failure and its already-written raw snapshot.
      return { error: String(error) }
    },
  )
  const fileUnchanged = (await readFile(activeFile)).equals(fileBefore)
  await appendFile(
    path.join(output, 'positioning.jsonl'),
    JSON.stringify({
      sample: sample.id,
      from: sample.from,
      to: sample.to,
      mode: before.mode,
      path: activeFile,
      elapsedMs: Date.now() - started,
      ticks,
      reversals,
      completed,
      completionReason,
      error: failure ? String(failure) : undefined,
      before: { version: before.version, selection: before.selection, dirty: before.dirty },
      after: { version: after.version, selection: after.selection, dirty: after.dirty },
      selectionProjection: { before: projectionBefore, after: projectionAfter },
      fileUnchanged,
      observations,
    }) + '\n',
  )
  assert.equal(after.version, before.version, 'Positioning must preserve source version')
  assert.equal(after.dirty, before.dirty, 'Positioning must preserve dirty state')
  assert(fileUnchanged, 'Positioning must preserve exact saved file bytes')
  if (failure) throw failure
}
function visible(rect, dom) {
  return (
    rect.bottom > Math.max(0, dom.editor.top) &&
    rect.top < Math.min(dom.height, dom.editor.bottom) &&
    rect.right > Math.max(0, dom.editor.left) &&
    rect.left < Math.min(dom.width, dom.editor.right)
  )
}
async function assertSample(sample, mode) {
  if (mode === 'source' || sample.expected === 'fallback') {
    if (mode === 'live')
      await app.until(
        async () => (await observe()).notices.length > 0,
        'explicit full-source fallback',
        30_000,
      )
    const dom = await observe()
    assert(
      !dom.previews.some((p) => p.from < sample.to && p.to > sample.from),
      'Literal/fallback range is never replaced',
    )
    const lines = await app.execute(
      'return [...document.querySelectorAll(".milkup-line")].filter(e=>Number(e.dataset.from)<arguments[1]&&Number(e.dataset.to)>=arguments[0]).map(e=>({from:Number(e.dataset.from),to:Number(e.dataset.to),text:e.textContent,rect:e.getBoundingClientRect().toJSON()}))',
      sample.from,
      sample.to,
    )
    assert(
      lines.some((line) => line.text.includes('```mermaid')),
      'Mapped opening fence remains present',
    )
    // Windowed source need not mount the whole long fence at once; exact file
    // bytes are asserted separately, and the visible source window is recorded.
    return { expected: sample.expected, sourceLines: lines, pixelReview: 'pending' }
  }
  let dom, preview
  await app.until(
    async () => {
      dom = await observe()
      preview = dom.previews.find(
        (p) => p.from === sample.from && p.to >= sample.to && p.to <= sample.to + 1,
      )
      return (
        preview?.images.length &&
        preview.images.every((i) => i.complete && i.naturalWidth > 0 && i.naturalHeight > 0)
      )
    },
    'actual image decode ' + sample.id,
    30_000,
  )
  assert(visible(preview.rect, dom), 'Preview must intersect the actual native viewport')
  assert.equal(preview.inlineSvg, 0)
  assert(preview.label)
  for (const image of preview.images) {
    assert(image.src.startsWith('blob:'))
    assert.equal(image.draggable, false)
    assert(image.tabIndex < 0)
    assert.equal(image.slotOverflow, 'visible')
    assert(image.rect.width > 0 && image.rect.height > 0)
  }
  return { preview, pixelReview: 'pending' }
}

async function alignScrollCorner(sample, expected) {
  const before = await app.probe()
  const projectionBefore = await sampleSelectionProjection(sample, 'before-corner-alignment')
  const fileBefore = await readFile(activeFile)
  const started = Date.now(),
    observations = []
  let ticks = 0,
    reversals = 0,
    previousDirection,
    failure,
    completed = false
  try {
    // Independent finite alignment stage: at most 5 seconds / 10 single ticks /
    // 2 reversals per corner. Existing sample and inner-wheel budgets are unchanged.
    while (Date.now() - started < 5000) {
      const dom = await observe()
      observations.push({ elapsedMs: Date.now() - started, ticks, reversals, dom })
      await preserveAssertion('scroll-edge-offset', dom, expected, 'during corner alignment')
      const actual = dom.previews.find(
        (item) => item.from === expected.from && item.to === expected.to,
      )
      const rect = actual.clientRect,
        viewport = dom.editorViewport
      const bottomRight = expected.corner === 'bottom-right'
      assert(
        bottomRight
          ? rect.right <= viewport.right && rect.right > viewport.left
          : rect.left >= viewport.left && rect.left < viewport.right,
        'Corner alignment: corresponding horizontal edge must already be inside editor viewport',
      )
      const edge = bottomRight ? rect.bottom : rect.top
      const aligned = bottomRight
        ? edge <= viewport.bottom && edge > viewport.top
        : edge >= viewport.top && edge < viewport.bottom
      if (aligned) {
        assert(Date.now() - started < 5000, 'Corner alignment: five-second bound reached')
        await preserveAssertion('scroll-edge', dom, expected, 'after corner alignment')
        completed = true
        break
      }
      assert(ticks < 10, 'Corner alignment: ten-wheel-tick bound reached')
      assert(dom.editorGutter, 'Corner alignment: real non-preview editor gutter hit required')
      const direction = bottomRight
        ? edge > viewport.bottom
          ? '5'
          : '4'
        : edge < viewport.top
          ? '4'
          : '5'
      if (previousDirection && direction !== previousDirection) reversals++
      assert(reversals <= 2, 'Corner alignment: two-direction-reversal bound reached')
      const scale =
        Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', nativeWindow))[1]) / dom.width
      assert(Number.isFinite(scale) && scale > 0, 'Corner alignment: actual native scale required')
      assert(Date.now() - started < 5000, 'Corner alignment: five-second bound reached')
      x(
        'mousemove',
        '--window',
        nativeWindow,
        String(Math.round(dom.editorGutter.x * scale)),
        String(Math.round(dom.editorGutter.y * scale)),
        'click',
        direction,
      )
      previousDirection = direction
      ticks++
      await app.delay(120)
    }
    assert(completed, 'Corner alignment: required edge must be visible within five seconds')
  } catch (error) {
    failure = error
  }
  const after = await app.probe()
  const projectionAfter = await sampleSelectionProjection(sample, 'after-corner-alignment').catch(
    (error) => {
      if (!failure) failure = error
      return { error: String(error) }
    },
  )
  const fileUnchanged = (await readFile(activeFile)).equals(fileBefore)
  await appendFile(
    path.join(output, 'corner-alignment.jsonl'),
    JSON.stringify({
      sample: sample.id,
      expected,
      elapsedMs: Date.now() - started,
      ticks,
      reversals,
      completed,
      error: failure ? String(failure) : undefined,
      before,
      after,
      selectionProjection: { before: projectionBefore, after: projectionAfter },
      fileUnchanged,
      observations,
    }) + '\n',
  )
  if (failure) throw failure
  assert.equal(after.path, before.path, 'Corner alignment must preserve document path')
  assert.equal(after.version, before.version, 'Corner alignment must preserve source version')
  assert.equal(after.dirty, before.dirty, 'Corner alignment must preserve dirty state')
  assert(fileUnchanged, 'Corner alignment must preserve exact saved file bytes')
}
async function scrollEdges(sample, key) {
  const read = async () => (await observe()).previews.find((p) => p.from === sample.from)
  const before = await read()
  let previousAim = null
  const aim = async (observations, phase, step) => {
    const dom = await observe()
    const preview = dom.previews.find((item) => item.from === before.from && item.to === before.to)
    const observation = {
      at: Date.now(),
      phase,
      step,
      previousAim,
      preview,
      viewport: dom.editorViewport,
    }
    observations.push(observation)
    assert(preview, 'Inner wheel target must remain mounted')
    assertCapturedPreview(dom, before, 'before inner wheel input')
    const rect = preview.clientRect,
      viewport = dom.editorViewport
    const intersection = {
      left: Math.max(rect.left, viewport.left),
      right: Math.min(rect.right, viewport.right),
      top: Math.max(rect.top, viewport.top),
      bottom: Math.min(rect.bottom, viewport.bottom),
    }
    observation.intersection = intersection
    assert(
      intersection.right > intersection.left && intersection.bottom > intersection.top,
      'Inner wheel requires a nonempty current client intersection',
    )
    const point = {
      x: (intersection.left + intersection.right) / 2,
      y: (intersection.top + intersection.bottom) / 2,
    }
    observation.requestedCssPoint = point
    const hit = (point) =>
      app.execute(
        `
const preview=[...document.querySelectorAll('[data-document-presentation]')].find(e=>Number(e.dataset.presentationFrom)===arguments[0]&&Number(e.dataset.presentationTo)===arguments[1]);
const target=document.elementFromPoint(arguments[2],arguments[3]);
return {inside:!!(preview&&target&&preview.contains(target)),targetTag:target?.tagName||null,targetRange:target?.closest('[data-document-presentation]')?.dataset.presentationFrom||null};
`,
        before.from,
        before.to,
        point.x,
        point.y,
      )
    observation.requestedHit = await hit(point)
    assert(observation.requestedHit.inside, 'Real wheel target hit test')
    const geometry = x('getwindowgeometry', '--shell', nativeWindow)
    const number = (text, key) => Number(new RegExp('^' + key + '=(-?\\d+)$', 'm').exec(text)?.[1])
    const windowGeometry = {
      raw: geometry,
      x: number(geometry, 'X'),
      y: number(geometry, 'Y'),
      width: number(geometry, 'WIDTH'),
      height: number(geometry, 'HEIGHT'),
    }
    const scale = windowGeometry.width / dom.width
    observation.windowGeometry = windowGeometry
    observation.scale = scale
    assert(
      Number.isFinite(scale) &&
        scale > 0 &&
        Number.isFinite(windowGeometry.x) &&
        Number.isFinite(windowGeometry.y),
      'Inner wheel requires actual native origin and scale',
    )
    const relativeNative = { x: Math.round(point.x * scale), y: Math.round(point.y * scale) }
    observation.requestedRelativeNativePoint = relativeNative
    x('mousemove', '--window', nativeWindow, String(relativeNative.x), String(relativeNative.y))
    const pointer = x('getmouselocation', '--shell')
    observation.actualRootPointer = {
      raw: pointer,
      x: number(pointer, 'X'),
      y: number(pointer, 'Y'),
    }
    const actualCssPoint = {
      x: (observation.actualRootPointer.x - windowGeometry.x) / scale,
      y: (observation.actualRootPointer.y - windowGeometry.y) / scale,
    }
    observation.actualCssPoint = actualCssPoint
    assert(
      Number.isFinite(actualCssPoint.x) && Number.isFinite(actualCssPoint.y),
      'Inner wheel actual pointer must be finite',
    )
    observation.actualHit = await hit(actualCssPoint)
    assert(observation.actualHit.inside, 'Actual X11 pointer must hit current target preview')
    previousAim = {
      requestedCssPoint: point,
      actualCssPoint,
      rootPointer: observation.actualRootPointer,
      windowGeometry,
      scale,
    }
  }
  for (const [corner, axes] of [
    [
      'bottom-right',
      [
        ['5', 'top', before.scrollHeight - before.clientHeight],
        ['7', 'left', before.scrollWidth - before.clientWidth],
      ],
    ],
    [
      'top-left',
      [
        ['4', 'top', 0],
        ['6', 'left', 0],
      ],
    ],
  ]) {
    for (const [button, axis, target] of axes) {
      const observations = []
      let failure,
        afterAxis,
        steps = 0
      try {
        await aim(observations, 'axis-start', null)
        for (let step = 0; step < 30 && Math.abs((await read())[axis] - target) > 1; step++) {
          await aim(observations, 'before-batch', step)
          x('click', '--repeat', '3', '--delay', '70', button)
          steps++
          await app.delay(120)
          const dom = await observe()
          observations.push({
            phase: 'after-batch',
            at: Date.now(),
            step,
            viewport: dom.editorViewport,
            preview: dom.previews.find(
              (item) => item.from === before.from && item.to === before.to,
            ),
          })
        }
        assert(
          Math.abs((await read())[axis] - target) <= 1,
          'Wheel reaches diagram ' + axis + ' ' + corner,
        )
      } catch (error) {
        failure = error
      } finally {
        try {
          const dom = await observe()
          afterAxis = {
            at: Date.now(),
            viewport: dom.editorViewport,
            diagnostics: dom.diagnostics,
            preview: dom.previews.find(
              (item) => item.from === before.from && item.to === before.to,
            ),
            rootPointer: x('getmouselocation', '--shell'),
          }
        } catch (error) {
          afterAxis = { error: String(error) }
          if (!failure) failure = error
        }
        await appendFile(
          path.join(output, 'inner-wheel.jsonl'),
          JSON.stringify({
            sample: sample.id,
            from: before.from,
            to: before.to,
            corner,
            axis,
            button,
            target,
            steps,
            observations,
            after: afterAxis,
            error: failure ? String(failure) : undefined,
          }) + '\n',
        ).catch((error) => {
          if (!failure) throw error
          console.error('Failed to retain inner wheel evidence: ' + String(error))
        })
      }
      if (failure) throw failure
    }
    const after = await read()
    const scrollEdge = { from: before.from, to: before.to, corner, after }
    await alignScrollCorner(sample, scrollEdge)
    await shot(`edges-${key}-${sample.id}-${corner}`, {
      sample: sample.id,
      before,
      after,
      scrollEdge,
    })
  }
}

async function finishNativeCapture() {
  const capture = process.env.MILKUP_COLD_REQUEST_OUTPUT_DIR
  const state = { requested: false, complete: false, deadlineMs: 1000 }
  try {
    assert(capture, 'Native cold capture directory is required')
    await writeFile(path.join(capture, 'finish.request'), 'sankey-short-final\n', { flag: 'wx' })
    state.requested = true
    const end = Date.now() + state.deadlineMs
    while (Date.now() < end) {
      try {
        const value = JSON.parse(await readFile(path.join(capture, 'complete.json'), 'utf8'))
        if (value.final === true) {
          state.sinkFinalObserved = true
          state.completion = value
          const drops = ['queueFull', 'disconnected', 'oversize', 'capped']
          state.complete = value.ownerBirth != null && value.inFlight === 0 &&
            value.ioErrors === 0 && value.incompleteTiming === 0 &&
            drops.every(key => value[key] === 0) && value.written === value.enqueued &&
            value.attempted === value.enqueued + drops.reduce((sum, key) => sum + value[key], 0)
          if (!state.complete) state.incomplete = 'native sink finalized but counters or identity are incomplete'
          break
        }
      } catch (error) {
        if (error.code !== 'ENOENT' && !(error instanceof SyntaxError)) throw error
      }
      await new Promise(resolve => setTimeout(resolve, 50))
    }
    if (!state.sinkFinalObserved) state.incomplete = 'native completion was not observed within 1000ms'
  } catch (error) { state.incomplete = String(error) }
  await writeFile(path.join(output, 'capture-finalization.json'), JSON.stringify(state, null, 2) + '\n')
}

// Post-outcome only: observe already-recorded ledgers; never retry UI or cancel work.
async function drainRecordedRequests() {
  const started = process.hrtime.bigint(), elapsed = () => Number(process.hrtime.bigint() - started) / 1e6
  const state = {budgetMs:5000,intervalMs:100,idleObserved:false,observations:[],clock:'controller hrtime; never subtracted from frontend/native clocks'}
  let stable, stableCount = 0
  while (elapsed() < 5000) {
    const row = {controllerStartNs:process.hrtime.bigint().toString()}
    try {
      const front = await app.execute(`const s=window.__milkupRenderQueueTrace; return s ? {
        events:s.events.map(e=>({client:e.client,phase:e.phase,requestId:e.requestId,nativeInstanceId:e.nativeInstanceId,frontendMs:e.frontendMs})),
        clients:s.clients,bytes:s.bytes,dropped:s.dropped,errors:s.errors,frontendMs:performance.now(),timeOriginMs:performance.timeOrigin
      } : null`)
      assert(front && Array.isArray(front.events) && front.events.length <= 1024)
      const capture = process.env.MILKUP_COLD_REQUEST_OUTPUT_DIR
      assert(capture)
      const [bytes, statusBytes] = await Promise.all([
        readFile(path.join(capture, 'events.jsonl')), readFile(path.join(capture, 'status.json')),
      ])
      assert(bytes.length <= 32 * 1024 * 1024, 'Bounded encoded native ledger read')
      const text = bytes.toString('utf8'), tailComplete = !text || text.endsWith('\n')
      const lines = text.split('\n'); if (!tailComplete) lines.pop()
      const native = lines.filter(Boolean).map(line => JSON.parse(line))
      assert(native.length <= 256)
      const status = JSON.parse(statusBytes)
      const frontJobs = new Map(), frontTerminals = new Set(), disposed = new Set()
      for (const e of front.events) {
        const key = `${e.client}:${e.requestId}`
        if (e.phase === 'queued' || e.phase === 'dispatch') frontJobs.set(key,e)
        if (['remote-settled','trusted-timeout','post-failed','cancel-queued'].includes(e.phase)) frontTerminals.add(key)
        if (e.phase === 'dispose') disposed.add(e.client)
      }
      const frontendOpen = [...frontJobs].filter(([key,e]) => !frontTerminals.has(key) && !disposed.has(e.client)).map(([key])=>key)
      const accepted = new Set(), terminals = new Set(), identityErrors = []
      for (const e of native) {
        if (!e.kind?.startsWith('render-')) continue
        if (!e.ownerPid || e.ownerBirth == null || !e.instanceId || !(e.eventMonotonicNs > 0)) identityErrors.push(e.sequence ?? null)
        if (e.requestId == null) continue
        const key = JSON.stringify([e.ownerPid,e.ownerBirth,e.instanceId,e.requestId])
        if (e.kind === 'render-accepted') accepted.add(key)
        if (['render-settled','render-deadline','render-failed','render-stop'].includes(e.kind)) terminals.add(key)
      }
      const nativeOpen = [...accepted].filter(key=>!terminals.has(key))
      const countersClean = ['queueFull','disconnected','oversize','capped','ioErrors','incompleteTiming'].every(k=>status[k]===0)
        && status.ownerBirth != null && status.inFlight===0 && status.attempted===status.enqueued && status.enqueued===status.written
        && status.written===native.length && tailComplete
      row.frontend = {count:front.events.length,clients:front.clients,bytes:front.bytes,dropped:front.dropped,errors:front.errors,frontendMs:front.frontendMs,timeOriginMs:front.timeOriginMs,open:frontendOpen}
      row.native = {count:native.length,accepted:accepted.size,terminalKeys:terminals.size,open:nativeOpen,status,tailComplete,identityErrors}
      const clean = countersClean && !identityErrors.length && front.dropped===0 && front.errors===0
      const signature = JSON.stringify([front.events.length,front.bytes,native.length,status.attempted,status.written,frontendOpen,nativeOpen])
      stableCount = clean && !frontendOpen.length && !nativeOpen.length && signature===stable ? stableCount+1 : 0
      stable = clean && !frontendOpen.length && !nativeOpen.length ? signature : undefined
      row.stableCount = stableCount
      if (stableCount >= 1 && elapsed() < 5000) state.idleObserved = true
    } catch (error) { row.error=String(error); stableCount=0; stable=undefined }
    row.controllerEndNs=process.hrtime.bigint().toString();row.elapsedMs=elapsed();state.observations.push(row)
    if (state.idleObserved || elapsed() >= 5000) break
    await new Promise(resolve=>setTimeout(resolve,Math.min(100,5000-elapsed())))
  }
  state.elapsedMs=elapsed();state.timedOut=!state.idleObserved;state.transportOverrun=state.elapsedMs>5100
  if (!state.idleObserved) state.incomplete='Dual recorded ledgers did not reach stable empty state within5s; no UI retry'
  await writeFile(path.join(output,'capture-ledger-drain.json'),JSON.stringify(state,null,2)+'\n')
}

async function exportFrontendCapture(name = 'frontend-queue-trace.json') {
  const state = {controllerStartNs:process.hrtime.bigint().toString(),complete:false}
  try {
    state.frontend = await app.execute(`return {
      frontendMs:performance.now(),timeOriginMs:performance.timeOrigin,dateNowMs:Date.now(),
      sink:window.__milkupRenderQueueTrace ?? null
    }`)
    const sink = state.frontend.sink
    state.complete = !!sink && Array.isArray(sink.events) && sink.dropped === 0 && sink.errors === 0
    if (!state.complete) state.incomplete = 'Missing frontend sink or nonzero dropped/errors'
  } catch (error) { state.incomplete = String(error) }
  state.controllerEndNs = process.hrtime.bigint().toString()
  await writeFile(path.join(output, name), JSON.stringify(state, null, 2) + '\n')
}
