use toml_edit::DocumentMut;
pub fn serialize_manifest(manifest: &DocumentMut) -> String {
  manifest
    .to_string()
    // apply some formatting fixes
    .replace(r#"" ,features =["#, r#"", features = ["#)
    .replace(r#"" , features"#, r#"", features"#)
    .replace("]}", "] }")
    .replace("={", "= {")
    .replace("=[", "= [")
    .replace(r#"",""#, r#"", ""#)
}

fn main() {
 let mut args=std::env::args().skip(1);
 let path=args.next().expect("manifest path");
 assert!(args.next().is_none());
 let original=std::fs::read_to_string(&path).unwrap();
 let parsed=original.parse::<DocumentMut>().unwrap();
 let serialized=serialize_manifest(&parsed);
 std::fs::write(path,serialized).unwrap();
}
