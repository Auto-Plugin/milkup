// Supplemental public-input checks prompted by actual partial-view screenshots.
// These do not replace the unchanged formal matrix or count as its acceptance.
async function caretGeometry() {
  return app.execute(`const e=document.querySelector('.milkup-editor'), c=document.querySelector('.milkup-cursor');return {editor:e.getBoundingClientRect().toJSON(),caret:c?.getBoundingClientRect().toJSON(),position:Number(c?.dataset.position),scrollLeft:e.scrollLeft,scrollTop:e.scrollTop}`)
}
async function vectorGeometry(sample, text) {
  const from = text.indexOf(sample.needle)
  return app.execute(`const e=document.querySelector('.milkup-editor');const p=[...document.querySelectorAll('[data-document-presentation]')].find(p=>Number(p.dataset.presentationFrom)<=arguments[0]&&Number(p.dataset.presentationTo)>arguments[0]);const s=p?.querySelector('svg');const er=e.getBoundingClientRect(),pr=p?.getBoundingClientRect();const clip=p?{left:Math.max(er.left,pr.left+p.clientLeft),right:Math.min(er.right,pr.left+p.clientLeft+p.clientWidth),top:Math.max(er.top,pr.top+p.clientTop),bottom:Math.min(er.bottom,pr.top+p.clientTop+p.clientHeight)}:null;return s?{editor:er.toJSON(),preview:pr.toJSON(),clip,previewScrollLeft:p.scrollLeft,previewScrollTop:p.scrollTop,svg:s.getBoundingClientRect().toJSON(),scrollTop:e.scrollTop,texts:[...s.querySelectorAll('text')].map(t=>({text:t.textContent,rect:t.getBoundingClientRect().toJSON()}))}:null`, from)
}
async function realWheel(direction, geometry) {
  const e = geometry.clip
  assert(e.right>e.left && e.bottom>e.top, 'Native wheel target intersects preview client area and editor')
  const physical = Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', nativeWindow))[1])
  const width = await app.execute('return innerWidth')
  x('mousemove', '--window', nativeWindow, String(Math.round(((e.left + e.right) / 2) * physical / width)), String(Math.round(((e.top + e.bottom) / 2) * physical / width)), 'click', String(direction))
  await app.delay(160)
}
async function revealVectorEdge(sample, text, edge) {
  const steps = []
  for (let i = 0; i <= 30; i++) {
    const g = await vectorGeometry(sample, text)
    assert(g, 'Previously displayed vector must remain queryable during bounded wheel reveal')
    steps.push(g)
    const value = g.svg[edge]
    if (value >= g.clip.top - 1 && value <= g.clip.bottom + 1) return steps
    assert(i < 30, 'Vector edge must be reachable with at most 30 real wheel notches')
    await realWheel(value > g.clip.bottom + 1 ? 5 : 4, g)
  }
}
async function revealLegend(sample, text, label) {
  const steps=[]
  for(let i=0;i<=40;i++) {
    const g=await vectorGeometry(sample,text)
    assert(g, 'Pie remains available during native legend navigation')
    const target=g.texts.find(t=>t.text.startsWith(label))
    assert(target, 'Expected literal pie legend '+label)
    const r=target.rect,c=g.clip
    steps.push({geometry:g,target})
    if(r.left>=c.left-1 && r.right<=c.right+1 && r.top>=c.top-1 && r.bottom<=c.bottom+1) return steps
    assert(i<40, 'Pie legend must be fully reachable by bounded real horizontal/vertical wheel input')
    await realWheel(r.right>c.right+1?7:r.left<c.left-1?6:r.bottom>c.bottom+1?5:4,g)
  }
}

  await start()
  await details('milkup.preferences')
  if (await ui.exists(action('approve', 'milkup.preferences'))) await click(action('approve', 'milkup.preferences'))
  if (await ui.exists(action('enable', 'milkup.preferences'))) await click(action('enable', 'milkup.preferences'))
  await closeUI()
  for (const name of ['ordinary', 'windowed']) {
    const text = contents.get(name)
    await openDocument(name)
    for (const theme of ['light', 'dark']) {
      x('windowsize', nativeWindow, '1100', '760')
      await appearance(theme, 2)
      x('windowsize', nativeWindow, '760', '520')
      await app.until(async () => {const d=await observe();return Math.abs(d.width-380)<2 && Math.abs(d.height-260)<2}, 'native 200 percent narrow window')
      await seek('SAMPLE-ROOT', 'source')
      const initial = await app.probe()
      const lineFrom = text.indexOf('\r\n', text.indexOf('SAMPLE-ROOT')) + 2
      const lineTo = text.indexOf('\r\n', lineFrom)
      await app.key('Right')
      await app.key('Down')
      await app.until(async () => {const p=await app.probe();return p.cursor>=lineFrom&&p.cursor<=lineTo}, 'next source line selected')
      await app.key('End')
      await app.until(async () => (await app.probe()).cursor === lineTo, 'native End reaches complete long source line')
      const end = await caretGeometry()
      assert(end.caret.left >= end.editor.left-1 && end.caret.right <= end.editor.right+1, 'End caret horizontally visible')
      assert(end.caret.top >= end.editor.top-1 && end.caret.bottom <= end.editor.bottom+1, 'End caret vertically visible')
      assert(end.scrollLeft > 0, 'This exact long source line actually uses horizontal scrolling')
      await shot(`reach-${name}-${theme}-source-End`, {initial, end, lineFrom, lineTo})
      await app.key('Home')
      await app.until(async () => (await app.probe()).cursor === lineFrom, 'native Home returns to source line start')
      const home = await caretGeometry()
      assert(home.caret.left >= home.editor.left-1 && home.caret.right <= home.editor.right+1, 'Home caret horizontally visible')
      assert(home.scrollLeft <= 26, 'Home returns horizontal reading origin')
      const after = await app.probe()
      assert.equal(after.version, initial.version)
      assert.equal(after.dirty, false)
      assert.deepEqual(await readFile(activeFile), Buffer.from(text))
      await shot(`reach-${name}-${theme}-source-Home`, {home, after})
      for (const sample of samples.filter(s => s.kind === 'vector')) {
        await seek(sample.marker, 'live')
        await assertSample(sample, 'live', text)
        const before = await vectorGeometry(sample, text)
        const bottom = await revealVectorEdge(sample, text, 'bottom')
        await shot(`reach-${name}-${theme}-${sample.id}-bottom`, {before, wheelSteps:bottom})
        const top = await revealVectorEdge(sample, text, 'top')
        await shot(`reach-${name}-${theme}-${sample.id}-top`, {wheelSteps:top})
        if(sample.id==='pie') for(const label of ['甲','乙']) {
          const legend=await revealLegend(sample,text,label)
          await shot(`reach-${name}-${theme}-pie-legend-${label}`, {wheelSteps:legend})
        }
        assert.equal((await app.probe()).version, initial.version)
        assert.equal((await app.probe()).dirty, false)
        assert.deepEqual(await readFile(activeFile), Buffer.from(text))
      }
    }
  }
  await record('SUPPLEMENTAL-REACHABILITY-COMPLETE', {formalAcceptance:false, sourceCaretCases:4, vectorEdgeCases:12, pieLegendCases:8, nativeInput:'XTest End/Home and vertical/horizontal wheel', windowsTested:false})
} catch (error) {
  await writeFile(path.join(output, 'failure.json'), JSON.stringify({error:String(error), stack:error.stack}, null, 2))
  if (app) await shot('FAIL', {error:String(error)}).catch(()=>{})
  throw error
} finally {
  if (app) await app.close()
}
