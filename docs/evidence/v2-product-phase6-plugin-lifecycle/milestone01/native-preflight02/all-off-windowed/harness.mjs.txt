async function install(pkg) {
  const id = pkg.manifest.id
  await ui.list()
  if (await ui.exists(action('details', id))) {
    await ui.details(id)
    if (await ui.exists(action('disable', id))) await ui.click(action('disable', id))
    await ui.click(action('remove', id))
    await app.until(
      () =>
        app.execute(
          'return !JSON.parse(localStorage.getItem("milkup.desktop.plugins.registry.v1")||"null")?.records.some(r=>r.manifest.id===arguments[0])',
          id,
        ),
      'removed bundled ' + id,
    )
  }
  await ui.list()
  await ui.click('[data-plugin-action="install"]')
  await ui.picker(pkg.file)
  await app.until(
    () =>
      app.execute(
        'return JSON.parse(localStorage.getItem("milkup.desktop.plugins.registry.v1")||"null")?.records.some(r=>r.manifest.id===arguments[0]&&r.sourcePath===arguments[1]&&r.manifest.version===arguments[2])',

        pkg.file,
        pkg.manifest.version,
      ),
    'installed ' + id,
    30000,
  )

  await ui.ready(action('approve', id))

const xmlText = (s) =>
  s.replace(/&(#x[\da-f]+|#\d+|amp|lt|gt|quot|apos);/gi, (_, entity) => {
    if (entity[0] === '#')
      return String.fromCodePoint(
        Number.parseInt(
          entity.slice(entity[1].toLowerCase() === 'x' ? 2 : 1),
          entity[1].toLowerCase() === 'x' ? 16 : 10,
        ),
      )
    return { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'" }[entity.toLowerCase()]
  })
function htmlGeometry(html) {
  const svgs = [...html.matchAll(/<svg\b[\s\S]*?<\/svg>/gi)].map((m) => m[0])
  assert.equal(svgs.length, current.spec.diagrams.length, 'Actual diagram SVG count')
  const allIds = svgs.flatMap((svg) =>
    [...svg.matchAll(/\bid="([^"]+)"/g)].map((m) => xmlText(m[1])),
  )
  assert.equal(new Set(allIds).size, allIds.length, 'SVG IDs unique across all diagrams')
  assert(!/<script\b|<iframe\b|<foreignObject\b|\bon(?:load|error)=/i.test(html), 'Inert export')
  assert(!html.includes('flowchart LR'), 'No Mermaid source fallback')
  return svgs.map((svg, i) => {
    const ids = new Set([...svg.matchAll(/\bid="([^"]+)"/g)].map((m) => xmlText(m[1])))
    const localRefs = [
      ...svg.matchAll(/url\(["']?#([^\s)"']+)["']?\)/g),
      ...svg.matchAll(/(?:href|xlink:href)="#([^"]+)"/g),
    ]
    for (const ref of localRefs)
      assert(ids.has(xmlText(ref[1])), 'SVG local URL/href reference resolves within diagram')
    const visibleText = xmlText(
      svg.replace(/<style\b[\s\S]*?<\/style>/gi, '').replace(/<[^>]+>/g, ' '),
    ).replace(/\s+/g, ' ')
    for (const label of current.spec.diagrams[i].nodeLabels)
      assert(
        visibleText.includes(label.replace(/\s+/g, ' ')),
        'Label in actual SVG, supplementary to geometry: ' + label,
      )
    const viewBox = /\bviewBox="([^"]+)"/
      .exec(svg)?.[1]
      .trim()
      .split(/[\s,]+/)
      .map(Number)
    const paths = [...svg.matchAll(/<path\b[^>]*\bd="([^"]+)"/g)].filter((m) =>
      /[MLCQ]/i.test(m[1]),
    )
    const arrows = [...svg.matchAll(/\bmarker-(?:end|start)="url\(#([^\)]+)\)"/g)]
    const shapes = (svg.match(/<(?:rect|polygon|ellipse|circle)\b/g) || []).length
    assert(
      viewBox?.length === 4 && viewBox.every(Number.isFinite) && viewBox[2] > 0 && viewBox[3] > 0,
      'Positive diagram viewBox',
    )
    assert(
      paths.length >= current.spec.diagrams[i].edgeCount &&
        shapes >= current.spec.diagrams[i].nodeCount,
      'Diagram has real edge paths and node geometry',
    )
    assert(arrows.length >= current.spec.diagrams[i].edgeCount, 'Actual edge arrow references')
    for (const ref of arrows)
      assert(svg.includes(`id="${ref[1]}"`), 'Arrow marker resolves in same SVG')
    return {
      viewBox,
      paths: paths.length,
      shapes,
      arrowReferences: arrows.length,
      svgSha256: sha(svg),
    }
  })
}
const markerCount = (text, marker) =>
  (
    text.match(
      new RegExp(
        '(?<![A-Za-z0-9_-])' + marker.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '(?![A-Za-z0-9_-])',
        'g',
      ),
    ) || []
  ).length
async function artifact(file) {
  const bytes = await readFile(file),
    result = {
      file,
      sha256: sha(bytes),
      bytes: bytes.length,
      visualReview: 'pending independent review',
    }
  if (format === 'html') {
    const html = bytes.toString('utf8')
    result.diagrams = htmlGeometry(html)
    const footnoteLinks = [...html.matchAll(/href="#source-(\d+)"/g)]
    assert(footnoteLinks.length >= 2, 'Cold footnote forward/back links are actual anchors')
    for (const link of footnoteLinks)
      assert(html.includes(`id="source-${link[1]}"`), 'Footnote destination resolves')
    assert(!html.includes('[^p5]'), 'No raw footnote source fallback')
    assert((html.match(/<math\b/g) || []).length >= 2, 'Original inline and display math retained')
    const images = [...html.matchAll(/data:image\/png;base64,([A-Za-z0-9+/=]+)/g)]
    assert(
      images.some((m) => sha(Buffer.from(m[1], 'base64')) === manifest.resources[0].sha256),
      'Original supported local PNG embedded',
    )
    result.footnoteLinks = footnoteLinks.length
    for (const marker of current.spec.bodyMarkersExactlyOnce)
      assert.equal(markerCount(html, marker), 1, marker)
    for (const marker of current.spec.forbiddenRevisionMarkers || [])
      assert(!html.includes(marker), 'No stale revision ' + marker)
  } else {
    assert(
      bytes.subarray(0, 5).equals(Buffer.from('%PDF-')) &&
        bytes.subarray(-1024).includes(Buffer.from('%%EOF')),
    )
    const info = execFileSync('pdfinfo', [file], { encoding: 'utf8' })
    await writeFile(file + '.info.txt', info)
    const dimensions = /^Page size:\s+([\d.]+) x ([\d.]+)/m.exec(info),
      expected = settings.landscape ? [792, 612] : [595.28, 841.89]
    assert(
      dimensions && expected.every((v, i) => Math.abs(v - Number(dimensions[i + 1])) < 2),
      'Actual paper size',
    )
    execFileSync('pdftoppm', ['-png', '-r', '96', file, file + '.page'], { timeout: 180000 })
    const extracted = execFileSync('pdftotext', ['-layout', file, '-'], { encoding: 'utf8' })
    await writeFile(file + '.text.txt', extracted)
    // These ASCII body/revision checks are independent of diagram/pixel fidelity.
    for (const marker of current.spec.bodyMarkersExactlyOnce)
      assert.equal(markerCount(extracted, marker), 1, marker)
    for (const marker of current.spec.forbiddenRevisionMarkers || [])
      assert(!extracted.includes(marker), 'No stale revision ' + marker)
    result.diagramFidelity =
      'NOT_ASSERTED: inspect all page rasters; extracted labels are not diagram evidence'
  }
  await json(path.basename(file) + '.artifact', result)
  return result
}

async function packageFiles(record) {
  const root = await realpath(record.rootPath)
  assert(root.startsWith(profile + path.sep), 'Only inspect this owned profile package')
  const files = []
  async function walk(dir) {
    for (const entry of await readdir(dir, { withFileTypes: true })) {
      const file = path.join(dir, entry.name)
      assert(!entry.isSymbolicLink(), 'Installed package tree must not contain links')
      if (entry.isDirectory()) await walk(file)
      else if (entry.isFile()) {
        const bytes = await readFile(file)
        files.push({ path: path.relative(root, file), bytes: bytes.length, sha256: sha(bytes) })
      }
    }
  }
  await walk(root)
  return files.sort((a, b) => a.path.localeCompare(b.path))
}

async function focusedBounds(selector) {
  await app.focus(selector)
  return app.execute(
    `const e=document.querySelector(arguments[0]),r=e.getBoundingClientRect();
    const rect=q=>({left:q.left,top:q.top,right:q.right,bottom:q.bottom,width:q.width,height:q.height});
    let clip={left:0,top:0,right:innerWidth,bottom:innerHeight};const ancestors=[];
    for(let p=e.parentElement;p;p=p.parentElement){const s=getComputedStyle(p),b=p.getBoundingClientRect();
      ancestors.push({tag:p.tagName,class:p.className,rect:rect(b),overflowX:s.overflowX,overflowY:s.overflowY,scrollTop:p.scrollTop,scrollHeight:p.scrollHeight,clientHeight:p.clientHeight});
      if(/auto|scroll|hidden|clip/.test(s.overflowX)){clip.left=Math.max(clip.left,b.left);clip.right=Math.min(clip.right,b.right)}
      if(/auto|scroll|hidden|clip/.test(s.overflowY)){clip.top=Math.max(clip.top,b.top);clip.bottom=Math.min(clip.bottom,b.bottom)}}
    const cx=(r.left+r.right)/2,cy=(r.top+r.bottom)/2,hit=document.elementFromPoint(cx,cy);
    return {selector:arguments[0],rect:rect(r),clip,ancestors,focused:document.activeElement===e,disabled:!!e.disabled,
      centerHit:e===hit||e.contains(hit),hit:hit?{tag:hit.tagName,class:hit.className}:null,
      fullyWithinClip:r.left>=clip.left-2&&r.right<=clip.right+2&&r.top>=clip.top-2&&r.bottom<=clip.bottom+2,
      viewport:{width:innerWidth,height:innerHeight,zoom:Number(document.querySelector('#app').dataset.appearanceZoom)}}`,
    selector,
  )
}

  lifecycle = await outputProcessEvidence(Number(x('getwindowpid', ui.windowId())), output)
  // Wait for the ordinary installed catalog; do not require unrelated plugins to run.
  await app.until(
    () =>
      app.execute(
        'return JSON.parse(localStorage.getItem("milkup.desktop.plugins.registry.v1")||"{\\"records\\":[]}").records.some(r=>r.manifest.id==="milkup.outline")',
      ),
    'bundled registry ready',
    45000,
  )
  for (const pkg of packages) await install(pkg)
  await ui.closeMenu()
  await app.focus('.milkup-input-proxy')
  await app.key('ctrl+1')
  await ui.click('[data-menu-toggle]')
  await ui.click('[data-menu-section="file"]')
  await ui.click('[data-command="open-sample"]')
  await ui.picker(sourcePath)
  await app.until(async () => {
    const p = await app.probe()
    return p.path === sourcePath && p.loading !== 'opening'
  }, 'fixture opened and source ready')
  const opened = await observe('fixture-opened-no-render-warmup')
  if (stage === 'all-off-windowed') {
    assert(opened.renderDom.nativeWindowed, 'Actual native windowed route required')
    assert.match(opened.document.notice, /已用原生大文件路径打开预览/)
  }
  if (stage === 'outline') await outlineLifecycle()
  else if (stage === 'reject') await rejectCandidates()
  else if (stage === 'manager-matrix') await managerMatrix()
  else if (stage === 'all-off' || stage === 'all-off-windowed') await disableAll()
  else {
    await openPanel()
    await field('format', format === 'html' ? 'html' : format === 'print-to-file' ? 'print' : 'pdf')
    if (format !== 'html') {
      await field('pageSize', settings.pageSize)
      await field('orientation', settings.landscape ? 'landscape' : 'portrait')
    }
    await exportOnce('cold-first')
    if (stage === 'cold-edit') await editRevision()
  }
  completed = true
} catch (error) {
  await json('failure', { error: String(error), stack: error.stack, rows })
  if (app) {
    await observe('failure-observation').catch(() => {})
    await app.shot('failure').catch(() => {})
  }
  throw error
} finally {
  try {
    await lifecycle?.sample('before-app-close')
  } finally {
    await app?.close()
  }
  await lifecycle?.released('after-app-close', true)
  await traces()
  if (completed)
    await json('completion', {
      automatedChecksPassed: true,
      scope: 'First checkpoint only; no upgrade rollback or permission-delta acceptance',
      visualReview: 'pending',
      diagramPdfFidelity: format === 'html' ? 'not applicable' : 'not asserted',
      stage,
      format,
      sourceSha256: sha(await readFile(sourcePath)),
      physicalPrinting: 'not exercised',
    })
}
