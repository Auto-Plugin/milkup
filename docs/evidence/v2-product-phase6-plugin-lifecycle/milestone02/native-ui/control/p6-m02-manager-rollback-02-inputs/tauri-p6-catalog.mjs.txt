const profile = path.resolve(required('MILKUP_NATIVE_PROFILE_ROOT'))

  profile.startsWith('/tmp/milkup-p6-profiles/') && profile !== '/tmp/milkup-p6-profiles',
  'Dedicated disposable P6 profile required',

assert(!binary.includes('/target/'), 'Frozen binary copy required')

await writeFile(sourcePath, catalogSource, { flag: 'wx' })

if (process.argv.includes('--preflight')) process.exit(0)

async function launch() {
  const launchOutput = path.join(output, 'launch-' + ++launches)
  await mkdir(launchOutput)
  app = await startProduction(sourcePath, launchOutput, { diagnostics: false })
  x('windowsize', ui.windowId(), '1280', '900')
  // Browser script only initiates the existing native command. No app/private
  // manager object is exposed or patched, and no native authority is mocked.
  assert(
    await app.execute('return typeof window.__TAURI_INTERNALS__?.invoke === "function"'),
    'Native invoke bridge available to trusted host',
  )
  await app.until(async () => (await catalog()).version === 1, 'native catalog readable', 45000)
  // A fresh profile has no legacy registry; migrationReceipt correctly remains
  // null. Shipped manager actions await initialization themselves.
  await ui.list()
  await ui.ready('[data-plugin-action="install"]')
  await observe('launch-' + launches)
}
async function invoke(command, args) {
  const key = '__p6CatalogProbe_' + randomUUID().replaceAll('-', '')
  await app.execute(
    `const [key,command,args]=arguments; window[key]={done:false};
    window.__TAURI_INTERNALS__.invoke(command,args).then(value=>window[key]={done:true,ok:true,value},error=>window[key]={done:true,ok:false,error:String(error)});`,
    key,
    command,
    args ?? {},
  )
  const result = await app.until(
    () => app.execute('return window[arguments[0]]?.done ? window[arguments[0]] : null', key),
    command,
    45000,
  )
  await app.execute('delete window[arguments[0]]', key)
  if (!result.ok) throw Error(result.error)
  return result.value
}
const catalog = () => invoke('plugin_catalog_read')
const find = (snapshot, pluginId = id) => snapshot.entries.find((entry) => entry.id === pluginId)
async function record() {
  return find(await catalog())
}
async function transition(entry, value) {
  return invoke('plugin_catalog_transition', {
    pluginId: entry.id,
    expectedRevision: entry.revision,
    action: value,
  })
}
function identity(entry) {
  assert(entry.pending)
  return {
    transactionId: entry.pending.transactionId,
    fingerprint: entry.pending.generation.capabilityFingerprint,
  }
}
async function observe(name) {
  const row = {
    name,
    at: new Date().toISOString(),
    catalog: await catalog(),
    document: await app.probe(),
    managerText: await app.execute(
      'return document.querySelector("[data-plugin-manager]")?.textContent ?? null',
    ),
    mirror: await app.execute('return localStorage.getItem("milkup.desktop.plugins.registry.v1")'),
  }
  events.push(row)
  await json('events', events)
  await app.shot(name)
  return row
}
async function restart() {
  await app.close()
  app = undefined
  await launch()
}
async function openPackage(kind) {
  await ui.list()
  await ui.click('[data-plugin-action="install"]')
  await ui.picker(fixtures[kind].file)
  await app.until(
    async () => {
      const entry = await record()
      const version = fixtures[kind].archive.manifest.version
      return (
        entry?.active?.manifest.version === version ||
        entry?.pending?.generation.manifest.version === version ||
        (await app.execute(
          'return [...document.querySelectorAll(".plugin-errors")].some(e=>e.textContent.includes(arguments[0]))',
          fixtures[kind].file,
        ))
      )
    },
    'candidate accepted or visible installation error',
    30000,
  )
}
async function activateCurrent() {
  await ui.details(id)
  if (await ui.exists(action('approve', id))) await ui.click(action('approve', id))
  const current = await record()
  if (current.pending || !current.enabled) await ui.click(action('enable', id))
  await app.until(
    async () => {
      const entry = await record()
      return entry?.enabled && !entry.startAttempt && !entry.pending
    },
    'verified activation committed and startup complete',
    30000,
  )
  await ui.ready(action('disable', id))
}
async function installBase() {
  assert(!find(await catalog()), 'Fresh fixture ID required')
  await openPackage('base')
  await activateCurrent()
  const entry = await record()
  assert.equal(entry.active.manifest.version, '1.0.0')
  assert(entry.active.verified && entry.enabled && !entry.pending)
  await observe('base-enabled')
  return entry
}
async function disable() {
  await ui.details(id)
  if (await ui.exists(action('disable', id))) await ui.click(action('disable', id))
  await app.until(async () => !(await record()).enabled, 'fixture disabled')
}
async function unicodeSave() {
  await ui.closeMenu()
  await app.focus('.milkup-input-proxy')
  await app.key('ctrl+1')
  await app.key('ctrl+End')
  const inserted = '\nP6 插件故障后仍可保存 😀 𠮷 e\u0301\n'
  const release = await ownNativeClipboard(inserted)
  try {
    await app.key('ctrl+v')
    await app.until(async () => (await app.probe()).dirty, 'Unicode edit dirty')
  } finally {
    release()
  }
  await app.key('ctrl+s')
  await app.until(
    async () => digest(await readFile(sourcePath)) === digest(catalogSource + inserted),
    'exact Unicode save',
  )
  await app.key('ctrl+z')
  await app.key('ctrl+s')
  await app.until(
    async () => digest(await readFile(sourcePath)) === digest(catalogSource),
    'exact Unicode undo save',
  )
  await app.key('ctrl+shift+z')
  await app.key('ctrl+s')
  await app.until(
    async () => digest(await readFile(sourcePath)) === digest(catalogSource + inserted),
    'exact Unicode redo save',
  )
  await app.key('ctrl+z')
  await app.key('ctrl+s')
  await app.until(
    async () => digest(await readFile(sourcePath)) === digest(catalogSource),
    'original bytes restored',
  )
  await observe('unicode-history-save')
}
async function prepare(kind, pluginId = id) {
  const current = find(await catalog(), pluginId)
  return invoke('plugin_catalog_prepare', {
    archive: catalogFixtureArchive(kind, pluginId),
    sourcePath: fixtures[kind].file,
    expectedRevision: current?.revision ?? null,
  })
}
async function stagedTrial() {
  await disable()
  let entry = find(await prepare('failure'))
  if (!entry.pending.generation.approvedFingerprint)
    entry = find(await transition(entry, { type: 'approve', ...identity(entry) }))
  entry = find(await transition(entry, { type: 'beginTrial', ...identity(entry) }))
  assert.equal(entry.pending.phase, 'trial')
  assert.equal(entry.enabled, false)
  return entry
}
async function ping(previousToken) {
  await ui.details(id)
  await ui.click('[data-plugin-command="catalogFixture.ping"]')
  return app.until(async () => {
    const notice = await app.execute(
      'return document.querySelector("[data-notice]")?.textContent ?? ""',
    )
    const token = /P6_BASE_ALIVE:([a-f0-9]{32})/.exec(notice)?.[1]
    return token && token !== previousToken ? token : false
  }, 'old code command responds from a fresh activation')
}
async function managerRollback(kind = 'failure') {
  const before = await installBase()
  const previousActivationToken = await ping()
  await openPackage(kind)
  await ui.details(id)
  let activationStarted
  if ((await record()).pending) {
    if (await ui.exists(action('approve', id))) await ui.click(action('approve', id))
    activationStarted = performance.now()
    await ui.click(action('enable', id))
  }
  assert(activationStarted !== undefined, 'The reviewed candidate must actually enter activation')
  await app.until(
    async () => {
      const entry = await record()
      return !entry.pending && entry.active?.id === before.active.id && entry.enabled
    },
    'failed candidate removed and old generation newly running',
    30000,
  )
  const after = await record()
  const recoveryElapsedMs = performance.now() - activationStarted
  if (kind === 'hang') {
    assert(
      recoveryElapsedMs >= 9000,
      'Infinite CPU activation must reach the real 10-second deadline, not fail immediately',
    )
    assert(
      recoveryElapsedMs < 30000,
      'Host must remain responsive and recover within the harness bound',
    )
  }
  assert.equal(after.active.contentSha256, before.active.contentSha256)
  assert.equal(after.active.approvedFingerprint, before.active.approvedFingerprint)
  const activationToken = await ping(previousActivationToken)
  assert.notEqual(activationToken, previousActivationToken)
  await json('candidate-recovery', {
    kind,
    recoveryElapsedMs,
    previousActivationToken,
    activationToken,
    oldGeneration: before.active.id,
    restoredGeneration: after.active.id,
    observation:
      'Public fixture activation nonce proves a fresh activation; private host epoch is not exposed to the harness',
  })
  await observe(kind === 'hang' ? 'cpu-hang-upgrade-restored' : 'failed-upgrade-restored')
}

async function cas() {
  const pluginId = 'example.catalog-concurrent'
  const entry = find(await prepare('base', pluginId), pluginId)
  const args = ['approve', 'abort'].map((type) => ({
    pluginId,
    expectedRevision: entry.revision,
    action: { type, ...identity(entry) },
  }))
  const key = '__p6Concurrent_' + randomUUID().replaceAll('-', '')
  await app.execute(
    `const [key,args]=arguments; window[key]=null; Promise.allSettled(args.map(arg=>window.__TAURI_INTERNALS__.invoke('plugin_catalog_transition',arg))).then(results=>window[key]=results.map(r=>r.status==='fulfilled'?{status:r.status,revision:r.value.revision}:{status:r.status,reason:String(r.reason)}));`,
    key,
    args,
  )
  const result = await app.until(
    () => app.execute('return window[arguments[0]]', key),
    'native concurrent CAS results',
  )
  assert.equal(result.filter((item) => item.status === 'fulfilled').length, 1)
  assert.equal(
    result.filter((item) => item.status === 'rejected' && item.reason.includes('conflict')).length,
    1,
  )
  await json('same-id-concurrent-cas', { expectedRevision: entry.revision, result })
  await app.execute('delete window[arguments[0]]', key)
  await observe('concurrent-cas')
}
async function interrupted(kind) {
  const before = await installBase()
  if (kind === 'trial') await stagedTrial()
  else {
    await disable()
    const entry = await record()
    await transition(entry, { type: 'startupBegin', generationId: entry.active.id })
    assert((await record()).startAttempt)
  }
  await observe('persisted-' + kind + '-before-restart')
  // A real process exit with incomplete persisted attempt; no completion/abort is
  // invoked. Graceful WebDriver close may trigger general shutdown but does not
  // know the trusted-host native-only injected transaction.
  await restart()
  const after = await record()
  assert.equal(after.active.id, before.active.id)
  assert.equal(after.pending, null)
  assert.equal(after.startAttempt, null)
  assert.equal(after.enabled, false)
  assert(after.diagnostics.length > 0)
  await observe('interrupted-' + kind + '-recovered')
}
async function mirrorRestart() {
  const before = await installBase()
  await app.execute(
    'localStorage.setItem("milkup.desktop.plugins.registry.v1", "{broken P6 mirror");',
  )
  await restart()
  const after = await record()
  assert.equal(after.active.id, before.active.id)
  assert.equal(after.active.contentSha256, before.active.contentSha256)
  assert.equal(after.active.approvedFingerprint, before.active.approvedFingerprint)
  assert.equal(after.enabled, true)
  await app.until(async () => {
    const mirror = await app.execute(
      'return localStorage.getItem("milkup.desktop.plugins.registry.v1")',
    )
    try {
      return JSON.parse(mirror).records.some((item) => item.manifest.id === id)
    } catch {
      return false
    }
  }, 'native authority rebuilt corrupted localStorage mirror')
  await observe('corrupt-mirror-rebuilt')
}

async function legacyMigration() {
  // This initial UI activation both verifies the fixture and waits for manager.ready.
  // Its new-profile native authority is parked after shutdown, never deleted or edited
  // in place, to synthesize an old release's pre-catalog profile.
  const initial = await installBase()
  const root = ownedNativePath(path.resolve(initial.active.packageRoot, '../../..'))
  const badIds = ['example.catalog-invalidmanifest', 'example.catalog-invalidrecord']
  const rows = []
  const diskFiles = []
  const sentinels = []
  for (const [index, pluginId] of [id, ...badIds].entries()) {
    const archive = catalogFixtureArchive('base', pluginId)
    if (index === 1) delete archive.manifest.name
    const packageRoot = ownedNativePath(path.join(root, pluginId, 'package'))
    const dataRoot = ownedNativePath(path.join(root, pluginId, 'data'))
    const storageRoot = ownedNativePath(path.join(root, pluginId, 'storage'))
    await mkdir(packageRoot, { recursive: true })
    await mkdir(dataRoot, { recursive: true })
    await mkdir(storageRoot, { recursive: true })
    const manifestPath = path.join(packageRoot, 'plugin.json')
    const manifestBytes = Buffer.from(JSON.stringify(archive.manifest, null, 2) + '\n')
    await writeFile(manifestPath, manifestBytes, { flag: 'wx' })
    diskFiles.push({ path: manifestPath, sha256: digest(manifestBytes) })
    for (const file of archive.files) {
      const target = ownedNativePath(path.join(packageRoot, file.path))
      assert(target.startsWith(packageRoot + path.sep), 'Legacy fixture stays in its package')
      await mkdir(path.dirname(target), { recursive: true })
      const bytes = Buffer.from(file.content, file.encoding === 'base64' ? 'base64' : 'utf8')
      await writeFile(target, bytes, { flag: 'wx' })
      diskFiles.push({ path: target, sha256: digest(bytes) })
    }
    const sentinel = path.join(dataRoot, 'p6-legacy-user-data.txt')
    const sentinelText = `旧插件数据 ${pluginId} 中文 😀 𠮷 e\u0301\n`
    await writeFile(sentinel, sentinelText, { flag: 'wx' })
    sentinels.push({ path: sentinel, text: sentinelText, sha256: digest(sentinelText) })
    rows.push({
      manifest: archive.manifest,
      manifestPath,
      rootPath: packageRoot,
      sourcePath: fixtures.base.file,
      dataRoot,
      storageRoot,
      installedAt: 1700000000000,
      updatedAt: 1700000000001,
      trust: 'development',
      // Even the valid old row claims enabled+approved. Migration must never promote it.
      enabled: true,
      state: 'enabled',
      errors: [],
      approvals:
        index === 2
          ? { permissions: 'not-an-array', hosts: ['worker'] }
          : { permissions: [], hosts: ['worker'] },
    })
  }
  const legacyRaw =
    JSON.stringify(
      { version: 1, records: rows, auditRecords: [], fixtureNote: '完整原文备份 中文 😀' },
      null,
      2,
    ) + '\n'
  const receipt = digest(legacyRaw)
  await writeFile(path.join(output, 'legacy-registry-original.json'), legacyRaw, { flag: 'wx' })
  await json('legacy-migration-inputs', {
    receipt,
    initialGeneration: initial.active.id,
    diskFiles,
    sentinels,
    badIds,
  })
  await app.execute(
    'localStorage.setItem("milkup.desktop.plugins.registry.v1", arguments[0]);',
    legacyRaw,
  )
  assert.equal(
    await app.execute('return localStorage.getItem("milkup.desktop.plugins.registry.v1")'),
    legacyRaw,
  )
  await app.close()
  app = undefined

  const parked = ownedNativePath(path.join(root, 'p6-before-legacy-migration'))
  await mkdir(parked)
  const parkedFiles = []
  for (const name of ['catalog-v1.json', 'catalog-v1.backup.json']) {
    const file = path.join(root, name)
    const bytes = await readFile(file)
    await writeFile(path.join(output, 'before-legacy-' + name), bytes, { flag: 'wx' })
    await rename(file, path.join(parked, name))
    parkedFiles.push({
      name,
      bytes: bytes.length,
      sha256: digest(bytes),
      preservedAt: path.join(parked, name),
    })
  }
  await json('legacy-profile-authority-parked', {
    root,
    parkedFiles,
    reason:
      'Simulate a pre-catalog release in this dedicated profile; preserve both original authority files and all generations',
  })
  // Only the real application's startup invokes migration. This harness never calls
  // plugin_catalog_migrate; read-only native snapshots observe its durable result.
  await launch()
  const migrated = await app.until(
    async () => {
      const snapshot = await catalog()
      return snapshot.migrationReceipt === receipt ? snapshot : false
    },
    'automatic startup migration exact receipt',
    45000,
  )
  const imported = find(migrated)
  assert(
    imported && !imported.active && imported.pending,
    'Valid legacy package is an unverified candidate only',
  )
  assert.equal(imported.pending.generation.verified, false)
  assert.equal(imported.pending.generation.approvedFingerprint, null)
  assert.equal(imported.enabled, false)
  assert.notEqual(
    imported.pending.generation.id,
    initial.active.id,
    'Migration stages actual legacy bytes as a new generation',
  )
  assert.equal(migrated.entries.filter((entry) => entry.id === id).length, 1)
  for (const badId of badIds)
    assert(!find(migrated, badId), 'Invalid old item has no native authority: ' + badId)
  for (const index of [1, 2])
    assert(
      migrated.diagnostics.some((message) =>
        message.includes(`Legacy record ${index} quarantined`),
      ),
      'Per-record quarantine diagnostic ' + index,
    )
  const backup = ownedNativePath(path.join(root, `legacy-registry-${receipt}.json`))
  assert.equal(
    await readFile(backup, 'utf8'),
    legacyRaw,
    'Native backup preserves exact complete original UTF-8 text',
  )
  await writeFile(path.join(output, 'native-legacy-raw-backup.json'), await readFile(backup), {
    flag: 'wx',
  })
  await observe('legacy-migration-quarantined-and-pending')
  await activateCurrent()
  const firstToken = await ping()
  const firstCommitted = await record()
  assert.equal(firstCommitted.active.id, imported.pending.generation.id)
  const generationNames = (await readdir(path.join(root, id, 'generations'))).sort()
  const backupNames = (await readdir(root))
    .filter((name) => /^legacy-registry-[a-f0-9]{64}\.json$/.test(name))
    .sort()
  assert.deepEqual(backupNames, [`legacy-registry-${receipt}.json`])

  // Reintroduce the same legacy mirror before a third real process startup. A durable
  // receipt/native entry must prevent another import or revival of the two bad rows.
  await app.execute(
    'localStorage.setItem("milkup.desktop.plugins.registry.v1", arguments[0]);',
    legacyRaw,
  )
  await restart()
  await app.until(
    async () => {
      const entry = await record()
      return entry?.enabled && !entry.startAttempt
    },
    'migrated verified plugin restored on later startup',
    30000,
  )
  const subsequent = await catalog()
  assert.equal(subsequent.migrationReceipt, receipt)
  assert.equal(find(subsequent).active.id, firstCommitted.active.id)
  assert.equal(subsequent.entries.filter((entry) => entry.id === id).length, 1)
  for (const badId of badIds) assert(!find(subsequent, badId))
  assert.deepEqual(
    (await readdir(path.join(root, id, 'generations'))).sort(),
    generationNames,
    'No duplicate migration generation',
  )
  assert.deepEqual(
    (await readdir(root))
      .filter((name) => /^legacy-registry-[a-f0-9]{64}\.json$/.test(name))
      .sort(),
    backupNames,
    'Single immutable legacy backup',
  )
  assert.equal(await readFile(backup, 'utf8'), legacyRaw)
  for (const file of diskFiles)
    assert.equal(
      digest(await readFile(file.path)),
      file.sha256,
      'Original legacy package bytes retained',
    )
  for (const sentinel of sentinels)
    assert.equal(
      await readFile(sentinel.path, 'utf8'),
      sentinel.text,
      'Valid and quarantined user data retained',
    )
  const laterToken = await ping(firstToken)
  await json('legacy-migration-result', {
    receipt,
    importedGeneration: firstCommitted.active.id,
    firstToken,
    laterToken,
    generationNames,
    backupNames,
    invalidIdsAbsent: badIds,
    launches,
  })
  await observe('legacy-migration-idempotent-after-restart')
}
function ownedNativePath(file) {
  const resolved = path.resolve(file)
  assert(resolved.startsWith(profile + path.sep), 'Fault injection limited to owned profile')
  return resolved
}
async function catalogCorruption() {
  const before = await installBase()
  await disable()
  const root = ownedNativePath(path.resolve(before.active.packageRoot, '../../..'))
  const primary = path.join(root, 'catalog-v1.json')
  const backup = path.join(root, 'catalog-v1.backup.json')
  await app.close()
  app = undefined
  const original = await readFile(primary)
  const backupBytes = await readFile(backup)
  await writeFile(path.join(output, 'catalog-before-corruption.json'), original, { flag: 'wx' })
  await writeFile(path.join(output, 'catalog-valid-backup.json'), backupBytes, { flag: 'wx' })
  await writeFile(primary, '{P6 intentionally corrupt primary')
  await launch()
  const recovered = await catalog()
  assert(recovered.diagnostics.some((message) => /corrupt|backup|recover/i.test(message)))
  assert(recovered.entries.every((entry) => !entry.enabled && !entry.startAttempt))
  assert(
    recovered.entries.every((entry) => !entry.active || entry.active.approvedFingerprint === null),
  )
  assert(find(recovered)?.active, 'valid backup package metadata retained')
  await observe('native-corruption-recovered-fail-closed')
}
async function removeFailure() {
  const before = await installBase()
  await disable()
  const generationRoot = ownedNativePath(path.dirname(before.active.packageRoot))
  const mode = (await stat(generationRoot)).mode & 0o777
  const dataRoot = ownedNativePath(before.dataRoot)
  await mkdir(dataRoot, { recursive: true })
  const sentinel = path.join(dataRoot, 'p6-retained-user-data.txt')
  await writeFile(sentinel, 'P6 user data 中文 😀', { flag: 'wx' })
  restorations.push(() => chmod(generationRoot, mode))
  await chmod(generationRoot, 0o500)
  const removed = find(await transition(await record(), { type: 'remove' }))
  assert(removed.tombstone && !removed.enabled)
  assert(
    removed.diagnostics.some((message) => /cleanup pending/i.test(message)),
    'actual deletion refusal recorded',
  )
  await observe('remove-failed-tombstone-persisted')
  await restart()
  const restarted = await record()
  assert(restarted.tombstone && !restarted.enabled)
  assert.equal(await readFile(sentinel, 'utf8'), 'P6 user data 中文 😀')
  await chmod(generationRoot, mode)
  await transition(restarted, { type: 'remove' })
  assert.equal(await readFile(sentinel, 'utf8'), 'P6 user data 中文 😀')
  await observe('remove-retried-data-retained')
}
