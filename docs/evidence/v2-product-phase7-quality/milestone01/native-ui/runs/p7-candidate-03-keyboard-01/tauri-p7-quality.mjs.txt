function check(condition, message, detail) {
  const item = { passed: !!condition, message, ...(detail === undefined ? {} : { detail }) }
  activeSample.checks.push(item)
  return item.passed
}
async function sample(name, callback) {
  const row = {
    name,
    fixture: { kind: fixture.kind, mode: fixture.mode, file: fixture.file },
    startedAt: new Date().toISOString(),
    checks: [],
  }
  activeSample = row
  try {
    row.before = await snapshot()
    await callback(row)
  } catch (error) {
    row.error = String(error)
    row.stack = error.stack
  }
  try {
    row.after = await snapshot()
  } catch (error) {
    row.observationError = String(error)
  }
  try {
    await app.shot(name)
    row.screenshot = name + '.png'
  } catch (error) {
    row.screenshotError = String(error)
  }
  const observedKeys = (row.after?.dom.events?.events ?? []).filter(
    (event) => event.sequence > (row.before?.dom.events?.total ?? 0) && event.type === 'keydown',
  )
  row.trustedKeyObservation = {
    count: observedKeys.length,
    allTrusted: observedKeys.every((event) => event.trusted),
  }
  row.checks.push({
    passed: observedKeys.length > 0 && row.trustedKeyObservation.allTrusted,
    message: 'nonempty observed keyboard events are trusted native input',
  })
  row.passed =
    !row.error &&
    !row.observationError &&
    !row.screenshotError &&
    row.checks.length > 0 &&
    row.checks.every((item) => item.passed)
  samples.push(row)
  await json('samples', samples)
  await json('operations', operations)
  return row
}

async function restoreBaseline() {
  await closeOverlays()

  await setupFocus(editor)
  // Only reverse this sample's possible native edit, preserving its prior evidence.
  for (let i = 0; i < 4; i++) {
    await key('ctrl+z')
    await key('ctrl+s')

      await app.until(
        async () =>
          !(await app.probe()).dirty && sha(await readFile(fixture.file)) === fixture.sha256,

async function routeCheck() {
  const state = await snapshot(),
    dom = state.dom
  const lines = dom.renderedLines.filter(
    (line) =>
      Number.isInteger(line.line) && Number.isInteger(line.from) && Number.isInteger(line.to),
  )
  check(
    dom.sourceDocumentView === (fixture.kind === 'windowed'),
    'windowed view class matches requested structure (not sole backend proof)',
    dom.sourceDocumentView,
  )
  check(lines.length > 0, 'nonempty exact DOM line/offset metrics', lines.length)
  if (fixture.kind === 'windowed') {
    check(
      fixture.bytes.length >= preflight.openPolicy.nativeLargeFileBytes,
      'frozen bytes exceed shipped native threshold',
      fixture.bytes.length,
    )
    check(
      dom.openDetails.scaleMode === 'performance' &&
        dom.openDetails.documentId?.startsWith('desktop-large-') &&
        dom.openDetails.timings?.includes('native-line-window'),
      'shipped open diagnostics confirm actual native backend rather than memory-windowed fallback',
      dom.openDetails,
    )
    check(dom.content?.data.virtualized === 'true', 'windowed content is virtualized')
    const from = Number(dom.content?.data.fromLine),
      to = Number(dom.content?.data.toLine)
    check(
      Number.isInteger(from) && Number.isInteger(to) && to > from && to - from < fixture.lines,
      'finite nonempty native window bounds',
      { from, to, total: fixture.lines },
    )
  }
  if (fixture.kind === 'ordinary')
    check(
      fixture.bytes.length < preflight.openPolicy.performanceBytes &&
        dom.openDetails.scaleMode === 'full',
      'ordinary bytes and shipped policy confirm full memory route',
      dom.openDetails,
    )
  await setupFocus(editor)
  await key('ctrl+End')
  await app.until(
    async () => (await app.probe()).cursor === fixture.utf16,
    'exact untruncated document end',
    30000,
  )
  check(
    (await app.probe()).cursor === fixture.utf16,
    'full UTF16 document length reachable',
    fixture.utf16,
  )
  await key('ctrl+Home')
}

async function activatePreferences() {
  const id = 'milkup.preferences'
  await ui.details(id)
  const candidate = async (kind) => {
    const scoped = '.plugin-candidate ' + action(kind, id)
    return (await ui.exists(scoped)) ? scoped : action(kind, id)
  }
  const approve = await candidate('approve')
  if (await ui.exists(approve)) {
    const review = await app.execute(
      'const e=document.querySelector(arguments[0]);return {fingerprint:e.dataset.pluginFingerprint,revision:Number(e.dataset.pluginEntryRevision),text:document.querySelector("[data-plugin-manager]")?.textContent}',
      approve,
    )
    assert.match(review.fingerprint, /^[a-f0-9]{64}$/)
    assert(Number.isInteger(review.revision) && review.revision >= 0)
    operations.push({ kind: 'UI-preferences-capability-review', ...review })
    await ui.click(approve)
    await app.until(async () => !(await ui.exists(approve)), 'preference approval acknowledged')
  }
  const enable = await candidate('enable')
  if (await ui.exists(enable)) await ui.click(enable)
  await ui.ready(action('disable', id))
  await ui.ready(action('open-ui', id))
}

async function revealLine(line) {
  await setupFocus(editor)

  const selector = `.milkup-line[data-line="${line}"]`

  // script scrollIntoView or a private revealSelection call.
  for (let i = 0; i < 60; i++) {
    const point = await visiblePoint(selector)
    if (point?.hit && point.visibleHeight >= Math.min(32, point.rect.bottom - point.rect.top) - 1)
      return selector
    const view = await app.execute(
      'const r=document.querySelector(".milkup-editor").getBoundingClientRect();return {x:(r.left+r.right)/2,y:Math.min(innerHeight-30,r.bottom-30),width:innerWidth}',
    )
    const win = ui.windowId(),
      scale = Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', win))[1]) / view.width
    x(
      'mousemove',
      '--window',
      win,
      String(Math.round(view.x * scale)),
      String(Math.round(view.y * scale)),
      'click',
      '5',
    )
    operations.push({ kind: 'native-wheel-down', targetLine: line })
    await app.delay(70)
  }
  throw Error('Native scroll did not reveal line ' + line)
}
async function nativeClick(selector) {
  const point = await visiblePoint(selector)
  assert(point?.hit, 'Target has a visible hit-tested native point: ' + selector)
  const win = ui.windowId(),
    scale =
      Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', win))[1]) / point.viewport.width
  x(
    'windowfocus',
    '--sync',
    win,
    'mousemove',
    '--window',
    win,
    String(Math.round(point.x * scale)),
    String(Math.round(point.y * scale)),
    'click',
    '1',
  )
  operations.push({ kind: 'native-mouse-click', selector, point, scale })
}
function cursorInside(state) {
  const r = state.dom.cursor?.rect,
    e = state.dom.editor?.rect,
    v = state.dom.viewport
  return (
    r &&
    e &&
    r.width > 0 &&
    r.height > 0 &&
    r.left >= Math.max(0, e.left) - 2 &&
    r.right <= Math.min(v.width, e.right) + 2 &&
    r.top >= Math.max(0, e.top) - 2 &&
    r.bottom <= Math.min(v.height, e.bottom) + 2
  )
}
async function matrix() {
  const prefix = `${fixture.kind}-${fixture.mode}`
  await sample(prefix + '-matrix-route', routeCheck)
  for (const theme of ['light', 'dark']) {
    const configured = await sample(prefix + '-theme-' + theme, async () => {
      await nativeZoom(1, 1280, 800)
      await setTheme(theme)
      check((await snapshot()).dom.theme === theme, 'theme persisted through actual preferences UI')
    })
    if (!configured.passed) continue
    for (const [width, height] of [
      [1280, 800],
      [760, 520],
    ]) {
      for (const zoom of [1, 1.25, 1.5, 2]) {
        const name = `${prefix}-${theme}-${width}x${height}-${zoom * 100}`
        const geometry = await sample(name + '-selection', async () => {
          await nativeZoom(zoom, width, height)
          await positionPlain()
          const state = await snapshot()
          check(
            state.dom.theme === theme && state.dom.zoom === zoom,
            'theme and real application zoom match cell',
          )
          check(
            cursorInside(state),
            'selected paragraph caret lies within editor and viewport',
            state.dom.cursor,
          )
          check(
            state.document.selection.some((range) => range.from === 0 && range.to === 8),
            'exact source selection survives display changes',
          )
          check(
            state.dom.selections.some((item) => item.rect.width > 0 && item.rect.height > 0),
            'nonempty painted selection',
          )
          check(
            state.diskSha256 === fixture.sha256 && !state.document.dirty,
            'display changes preserve exact source bytes',
          )
        })
        if (!geometry.passed && geometry.error) continue
        await sample(name + '-table', async (row) => {
          const selector = await revealLine(6)
          row.table = await app.execute(
            rectScript +
              `const line=document.querySelector(arguments[0]);return {line:describe(line),cells:[...line.querySelectorAll('.milkup-table-cell')].map(e=>{const s=getComputedStyle(e),range=document.createRange();range.selectNodeContents(e);return {text:e.textContent,rect:rect(e),align:s.textAlign,overflowWrap:s.overflowWrap,textRects:[...range.getClientRects()].map(r=>({left:r.left,right:r.right,top:r.top,bottom:r.bottom}))}})}`,
            selector,
          )
          check(!!row.table.line?.visible, 'table source row has visible geometry')
          if (fixture.mode === 'live') {
            check(
              JSON.stringify(row.table.cells.map((cell) => cell.align)) ===
                JSON.stringify(['left', 'center', 'right']),
              'live table delimiter alignment applied',
              row.table.cells,
            )
            check(
              row.table.cells.length === 3 &&
                row.table.cells.every(
                  (cell) =>
                    cell.textRects.length > 0 &&
                    cell.textRects.every(
                      (r) => r.left >= cell.rect.left - 2 && r.right <= cell.rect.right + 2,
                    ),
                ),
              'long unspaced cell text stays inside its own column',
              row.table.cells,
            )
          }
        })
        await sample(name + '-emphasis', async (row) => {
          const selector = await revealLine(8)
          row.emphasis = await app.execute(
            rectScript +
              `const line=document.querySelector(arguments[0]),nodes=[];const walk=document.createTreeWalker(line,NodeFilter.SHOW_TEXT);while(walk.nextNode()){const n=walk.currentNode,e=n.parentElement,r=document.createRange();r.selectNodeContents(n);if(r.getClientRects().length)nodes.push({text:n.textContent,fontStyle:getComputedStyle(e).fontStyle,fontWeight:getComputedStyle(e).fontWeight})}return {line:describe(line),strong:line.querySelectorAll('.milkup-inline-strong').length,emphasis:line.querySelectorAll('.milkup-inline-emphasis').length,nodes}`,
            selector,
          )
          check(!!row.emphasis.line?.visible, 'emphasis row has visible geometry')
          if (fixture.mode === 'live') {
            check(
              row.emphasis.strong > 0 && row.emphasis.emphasis > 0,
              'nested strong/emphasis rendered',
            )
            check(
              row.emphasis.nodes.some(
                (node) => node.text.includes('P7-NORMAL-TAIL') && node.fontStyle === 'normal',
              ),
              'following text does not inherit unintended italic',
              row.emphasis.nodes,
            )
            check(
              !row.emphasis.nodes.some((node) => node.text.includes('*')),
              'inactive nested emphasis leaves no visible star markers',
              row.emphasis.nodes,
            )
          }
        })
        await sample(name + '-task', async (row) => {
          const selector = await revealLine(10)
          row.task = await app.execute(
            rectScript +
              `const line=document.querySelector(arguments[0]);return {line:describe(line),controls:[...line.querySelectorAll('.milkup-task-marker')].map(e=>({description:describe(e),checked:e.getAttribute('aria-checked'),role:e.getAttribute('role')})),visibleBullets:[...line.querySelectorAll('.milkup-list-marker')].filter(e=>e.getClientRects().length&&getComputedStyle(e).display!=='none').map(e=>e.textContent)}`,
            selector,
          )
          check(!!row.task.line?.visible, 'task row has visible geometry')
          if (fixture.mode === 'live') {
            check(
              row.task.controls.length === 1 && row.task.controls[0].role === 'checkbox',
              'single semantic checkbox rendered',
            )
            check(
              row.task.visibleBullets.length === 0,
              'task does not also show a list bullet',
              row.task.visibleBullets,
            )
            await nativeClick(selector + ' .milkup-task-marker')
            await saved(
              sha(Buffer.from(fixture.text.replace('- [ ] P7 unchecked', '- [x] P7 unchecked'))),
            )
            row.checkedState = await snapshot()
            await app.shot(name + '-task-checked')
            check(
              row.checkedState.document.dirty === false,
              'native checkbox click edits and saves exact Markdown',
            )
            await setupFocus(editor)
            await key('ctrl+z')
            await saved()
            check(
              sha(await readFile(fixture.file)) === fixture.sha256,
              'checkbox edit undo restores exact source',
            )
          }
        })
        await restoreBaseline()
        await sample(name + '-code-End', async (row) => {
          const selector = await revealLine(14)
          await nativeClick(selector)
          await key('End')
          const end = fixtureLines.slice(0, 15).join('\n').length
          await app.until(
            async () => (await app.probe()).cursor === end,
            'long code logical End source offset',
          )
          const state = await snapshot()
          row.code = await app.execute(
            rectScript +
              'const e=document.querySelector(arguments[0]);return {line:describe(e),scrollLeft:e.scrollLeft,scrollWidth:e.scrollWidth,clientWidth:e.clientWidth}',
            selector,
          )
          check(cursorInside(state), 'End caret of long code stays inside visible viewport', {
            cursor: state.dom.cursor,
            code: row.code,
          })
          if (fixture.mode === 'live')
            check(
              row.code.scrollWidth <= row.code.clientWidth + 1 || row.code.scrollLeft > 0,
              'long live code scroll follows caret',
              row.code,
            )
          await saved()
          check(
            sha(await readFile(fixture.file)) === fixture.sha256,
            'matrix cell leaves complete source byte-exact',
          )
        })
        await restoreBaseline()
      }
    }
  }
}

async function installReadonlyObserver() {
  await app.execute(`
    window.__p7ReadonlyEvents={events:[],total:0,dropped:0};
    const identify=e=>e instanceof Element?{tag:e.tagName,id:e.id,class:e.className,role:e.getAttribute('role'),name:e.getAttribute('aria-label'),data:{...e.dataset}}:null;
    for(const type of ['keydown','keyup','focusin','beforeinput','input','paste','compositionstart','compositionupdate','compositionend']) {
      document.addEventListener(type,event=>{
        const log=window.__p7ReadonlyEvents,item={type,time:performance.now(),trusted:event.isTrusted,key:event.key??null,inputType:event.inputType??null,isComposing:event.isComposing??false,dataLength:typeof event.data==='string'?event.data.length:0,target:identify(event.target)};
        log.total++;item.sequence=log.total;log.events.push(item);if(log.events.length>128){log.events.shift();log.dropped++}
        queueMicrotask(()=>{item.defaultPrevented=event.defaultPrevented});
        if(type==='keydown'||type==='input') requestAnimationFrame(()=>requestAnimationFrame(()=>{
          item.twoFrame={time:performance.now(),cursor:Number(document.querySelector('.milkup-cursor')?.dataset.position),version:Number(document.querySelector('[data-stat="version"]')?.textContent)};
        }));
      },{capture:true,passive:true});
    }
  `)
}
const fixtureFailures = []
try {
  fixture = fixtures[0]
  app = await startProduction(fixture.file, output, { diagnostics: false })

      }

      else await matrix()
    } catch (error) {

      await json('fixture-failures', fixtureFailures)
      await app.shot(`${fixture.kind}-${fixture.mode}-fixture-failure`).catch(() => {})
      // Do not overwrite failed product bytes from Node. Opening the next fixture
      // remains a normal user operation; close-confirm failure remains explicit.
    }
  }

  fatal = { error: String(error), stack: error.stack }
} finally {
  const finalFiles = []
  for (const current of fixtures) {
    const bytes = await readFile(current.file)
    finalFiles.push({
      file: current.file,
      expected: current.sha256,
      actual: sha(bytes),
      unchanged: sha(bytes) === current.sha256,
    })
  }

    !fatal &&
    fixtureFailures.length === 0 &&
    samples.length > 0 &&
    samples.every((row) => row.passed) &&

    fixtureFailures,
    fatal,
    finalFiles,
    limitations: preflight.boundaries,
  })

    JSON.stringify({
      passed,
      samples: samples.length,
      failed: samples.filter((row) => !row.passed).map((row) => row.name),
      fixtureFailures,
      fatal,
    }),
  )
  if (!passed) process.exitCode = 1
}
