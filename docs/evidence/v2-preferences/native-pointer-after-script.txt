async function keyboardFocus(selector) {
  await ui.ready(selector)
  for (let i = 0; i < 100; i++) {
    if (await app.execute('return document.activeElement?.matches(arguments[0])', selector)) return
    const previous = await app.execute('return document.activeElement?.outerHTML')
    await app.key('Tab')
    await app.until(
      () => app.execute('return document.activeElement?.outerHTML !== arguments[0]', previous),
      'native Tab focus settled',
    )
  }
  throw Error('Keyboard focus did not reach ' + selector)
}
async function click(selector) {
  await ui.ready(selector)
  const geometrySample = () => app.execute(
    'const e=document.querySelector(arguments[0]),r=e.getBoundingClientRect(),x=r.left+r.width/2,y=r.top+r.height/2;return {x,y,w:innerWidth,h:innerHeight,zoom:Number(document.querySelector("#app")?.dataset.appearanceZoom),hit:e.contains(document.elementFromPoint(x,y))}',
    selector,
  )
  let r
  await app.until(async () => {
    const before = await geometrySample()
    await app.delay(80)
    const after = await geometrySample()
    if (JSON.stringify(before) !== JSON.stringify(after)) return false
    r = after
    return true
  }, 'stable native click geometry')
  await writeFile(path.join(output, 'last-pointer-action.json'), JSON.stringify({selector, sampled:r}, null, 2))
  if (!r.hit || r.x < 0 || r.x >= r.w || r.y < 0 || r.y >= r.h) {
    await keyboardFocus(selector)
    await app.key('Return')
  } else {
    const geometry = x('getwindowgeometry', '--shell', ui.windowId())
    const scale = Number(/WIDTH=(\d+)/.exec(geometry)[1]) / r.w
    x(
      'mousemove',
      '--window',
      ui.windowId(),
      String(Math.round(r.x * scale)),
      String(Math.round(r.y * scale)),
      'click',
      '1',
    )
  }
  await app.delay(150)
  await appendFile(path.join(output,'pointer-trace.jsonl'),JSON.stringify({selector,sampled:r,after:await app.execute('return {zoom:Number(document.querySelector("#app")?.dataset.appearanceZoom),innerWidth,events:window.pointerTrace}')})+'\n')
}

async function setField(key, value) {
  const selector = field(key)
  await keyboardFocus(selector)
  const options = await app.execute(
    'const e=document.querySelector(arguments[0]);return e.tagName==="SELECT"?[...e.options].map(o=>o.value):null',
    selector,
  )
  if (options) {
    const index = options.indexOf(String(value))
    assert(index >= 0, `supported option ${key}=${value}`)
    await app.key('Home')
    for (let i = 0; i < index; i++) {
      await app.key('Down')
      await app.delay(80)
    }
    await app.key('Tab')
  } else {
    await app.key('ctrl+a')
    await app.text(String(value))
    await app.key('Tab')
  }
  await app.until(
    () =>
      app.execute(
        'return document.querySelector(arguments[0])?.value===arguments[1]',
        selector,
        String(value),
      ),
    'controlled draft ' + key,
  )
  if (key === 'appearance.zoom') {
    const physicalWidth = Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', ui.windowId()))[1])
    await app.until(async () => {
      const dom = await observe()
      return dom.zoom === Number(value) && Math.abs(dom.width * Number(value) - physicalWidth) <= 2
    }, 'host and native WebView zoom applied')
  }
  await app.delay(200)
}

async function gutterRoundTrip(name) {
  await editorMode('live')
  const original = contents.get(name)
  const end = original.indexOf('\r\n')
  const r = await app.execute(
    'const e=document.querySelector(".milkup-editor").getBoundingClientRect(),c=document.querySelector(".milkup-editor-content").getBoundingClientRect(),l=document.querySelector(".milkup-line").getBoundingClientRect();return {left:e.left+20,right:e.right-20,y:l.top+l.height/2,contentLeft:c.left,contentRight:c.right,w:innerWidth}',
  )
  assert.ok(
    r.left < r.contentLeft && r.right > r.contentRight,
    'centered live body has both real gutters',
  )
  const geometry = x('getwindowgeometry', '--shell', ui.windowId())
  const scale = Number(/WIDTH=(\d+)/.exec(geometry)[1]) / r.w
  const pointer = (clientX, ...actions) =>
    x(
      'mousemove',
      '--window',
      ui.windowId(),
      String(Math.round(clientX * scale)),
      String(Math.round(r.y * scale)),
      ...actions,
    )
  pointer(r.left, 'click', '1')
  await app.until(
    async () =>
      (await app.probe()).cursor === 0 &&
      (await app.execute('return document.activeElement?.matches(".milkup-input-proxy")')),
    'left gutter focus and exact line start',
  )
  await app.text('L')
  await app.until(async () => (await app.probe()).cursor === 1, 'left gutter text input')
  pointer(r.right, 'click', '1')
  await app.until(
    async () => (await app.probe()).cursor === end + 1,
    'right gutter exact Unicode line end',
  )
  await app.text('R')
  await app.key('ctrl+s')
  const expected = 'L' + original.slice(0, end) + 'R' + original.slice(end)
  await app.until(
    async () =>
      !(await app.probe()).dirty && (await readFile(activeFile)).equals(Buffer.from(expected)),
    'gutter input full byte save',
    30000,
  )
  contents.set(name, expected)
  pointer(r.left, 'mousedown', '1')
  pointer(r.right)
  await app.delay(100)
  x('mouseup', '1')
  await app.until(
    () =>
      app.execute(
        'return !!document.querySelector(".milkup-selection") && document.activeElement?.matches(".milkup-input-proxy")',
      ),
    'gutter drag selection',
  )
  await shot(`gutters-${name}-live`, { sha256: hash(Buffer.from(expected)), unicodeEnd: end + 1 })
}

}

async function keymapAcceptance() {
  const shortcutId = 'milkup.shortcuts'
  const modal = '[data-plugin-ui="milkup.shortcuts:shortcuts-modal"]'
  const input = `${modal} [data-plugin-field-control="file.save"]`
  const button = (name) => `${modal} [data-plugin-command="shortcuts.${name}"]`
  const archive =
    process.env.MILKUP_SHORTCUTS_ARCHIVE ||
    '/workspace/milkup-preferences-phase3-evidence/keymap/external/dist/milkup.shortcuts-1.0.0.milkup-plugin'
  const bytes = await readFile(archive)
  assert.equal(JSON.parse(bytes).manifest.id, shortcutId)
  await list()
  if (await ui.exists(action('details', shortcutId))) {
    await details(shortcutId)
    await click(action('remove', shortcutId))
  }
  await list()
  await click('[data-plugin-action="install"]')
  await ui.picker(archive)
  await ui.ready(action('approve', shortcutId))
  await click(action('approve', shortcutId))
  await click(action('enable', shortcutId))
  async function open() {
    await details(shortcutId)
    await click(action('open-ui', shortcutId))
    await ui.ready(input)
  }
  async function close() {
    await click(`${modal} .plugin-ui-close`)
    await app.until(async () => !(await ui.exists(modal)), 'shortcut modal closed')
    await closeMenu()
  }
  async function draft(value) {
    await keyboardFocus(input)
    await app.key('ctrl+a')
    await app.text(value)
    await app.key('Tab')
    await app.until(
      () =>
        app.execute(
          'return document.querySelector(arguments[0])?.value===arguments[1]',
          input,
          value,
        ),
      'shortcut draft',
    )
  }
  async function message(text) {
    await app.until(
      () =>
        app.execute(
          'return document.querySelector(arguments[0])?.textContent.includes(arguments[1])',
          modal,
          text,
        ),
      'shortcut status ' + text,
    )
  }
  await open()
  await draft('Mod+O')
  await click(button('save'))
  await message('快捷键已被其他命令占用')
  await shot('keymap-01-conflict', {
    archive,
    archiveSha256: hash(bytes),
    text: await app.execute('return document.querySelector(arguments[0]).textContent', modal),
  })
  await click(button('cancel'))
  await message('已取消未保存的修改')
  await draft('Mod+K')
  await click(button('save'))
  await message('快捷键已保存')
  await draft('Mod+J')
  await click(button('cancel'))
  await message('已取消未保存的修改')
  assert.equal(
    await app.execute('return document.querySelector(arguments[0]).value', input),
    'Mod+K',
  )
  await shot('keymap-02-rebound-cancelled', { binding: 'Mod+K' })
  await close()
  async function verifySave(tag, checkOld) {
    const before = contents.get('ordinary')
    await editorMode('source')
    await app.text('KEYMAP-')
    await app.until(async () => (await app.probe()).dirty, 'keymap document edit')
    if (checkOld) {
      await app.key('ctrl+s')
      await app.delay(400)
      assert((await app.probe()).dirty, 'rebound save does not retain the previous Ctrl+S')
      assert.deepEqual(await readFile(activeFile), Buffer.from(before))
    }
    await app.key('ctrl+k')
    const expected = 'KEYMAP-' + before
    await app.until(
      async () =>
        !(await app.probe()).dirty && (await readFile(activeFile)).equals(Buffer.from(expected)),
      'native rebound save',
      30000,
    )
    contents.set('ordinary', expected)
    await shot(tag, { sha256: hash(Buffer.from(expected)) })
  }
  await verifySave('keymap-03-native-control-k-save', true)
  await app.close()
  app = undefined
  await start(path.join(output, 'ordinary.md'))
  await verifySave('keymap-04-restart-control-k-save', false)
  await open()
  await click(button('reset'))
  await message('已恢复默认快捷键')
  await close()
}
