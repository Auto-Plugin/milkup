async function archive(key, id, version) {

    bytes = await readFile(file)

async function click(selector) {
  await app.focus(selector)
  const r = await app.execute(
    'const e=document.querySelector(arguments[0]),r=e.getBoundingClientRect();return {x:r.left+r.width/2,y:r.top+r.height/2,w:innerWidth,h:innerHeight,hit:e.contains(document.elementFromPoint(r.left+r.width/2,r.top+r.height/2)),disabled:e.disabled}',
    selector,
  )
  assert(
    !r.disabled && r.hit && r.x > 0 && r.y > 0 && r.x < r.w && r.y < r.h,
    'Visible native target ' + selector,
  )
  const win = ui.windowId(),
    scale = Number(/WIDTH=(\d+)/.exec(x('getwindowgeometry', '--shell', win))[1]) / r.w
  x(
    'windowfocus',
    '--sync',
    win,
    'mousemove',
    '--window',
    win,
    String(Math.round(r.x * scale)),
    String(Math.round(r.y * scale)),
    'click',
    '1',
  )
}

const xmlText = (s) =>
  s.replace(/&(#x[\da-f]+|#\d+|amp|lt|gt|quot|apos);/gi, (_, entity) => {
    if (entity[0] === '#')
      return String.fromCodePoint(
        Number.parseInt(
          entity.slice(entity[1].toLowerCase() === 'x' ? 2 : 1),
          entity[1].toLowerCase() === 'x' ? 16 : 10,
        ),
      )
    return { amp: '&', lt: '<', gt: '>', quot: '"', apos: "'" }[entity.toLowerCase()]
  })
function htmlGeometry(html) {
  const svgs = [...html.matchAll(/<svg\b[\s\S]*?<\/svg>/gi)].map((m) => m[0])
  assert.equal(svgs.length, current.spec.diagrams.length, 'Actual diagram SVG count')
  const allIds = svgs.flatMap((svg) =>
    [...svg.matchAll(/\bid="([^"]+)"/g)].map((m) => xmlText(m[1])),
  )
  assert.equal(new Set(allIds).size, allIds.length, 'SVG IDs unique across all diagrams')
  assert(!/<script\b|<iframe\b|<foreignObject\b|\bon(?:load|error)=/i.test(html), 'Inert export')
  assert(!html.includes('flowchart LR'), 'No Mermaid source fallback')
  return svgs.map((svg, i) => {
    const ids = new Set([...svg.matchAll(/\bid="([^"]+)"/g)].map((m) => xmlText(m[1])))
    const localRefs = [
      ...svg.matchAll(/url\(["']?#([^\s)"']+)["']?\)/g),
      ...svg.matchAll(/(?:href|xlink:href)="#([^"]+)"/g),
    ]
    for (const ref of localRefs)
      assert(ids.has(xmlText(ref[1])), 'SVG local URL/href reference resolves within diagram')
    const visibleText = xmlText(
      svg.replace(/<style\b[\s\S]*?<\/style>/gi, '').replace(/<[^>]+>/g, ' '),
    ).replace(/\s+/g, ' ')
    for (const label of current.spec.diagrams[i].nodeLabels)
      assert(
        visibleText.includes(label.replace(/\s+/g, ' ')),
        'Label in actual SVG, supplementary to geometry: ' + label,
      )
    const viewBox = /\bviewBox="([^"]+)"/
      .exec(svg)?.[1]
      .trim()
      .split(/[\s,]+/)
      .map(Number)
    const paths = [...svg.matchAll(/<path\b[^>]*\bd="([^"]+)"/g)].filter((m) =>
      /[MLCQ]/i.test(m[1]),
    )
    const arrows = [...svg.matchAll(/\bmarker-(?:end|start)="url\(#([^\)]+)\)"/g)]
    const shapes = (svg.match(/<(?:rect|polygon|ellipse|circle)\b/g) || []).length
    assert(
      viewBox?.length === 4 && viewBox.every(Number.isFinite) && viewBox[2] > 0 && viewBox[3] > 0,
      'Positive diagram viewBox',
    )
    assert(
      paths.length >= current.spec.diagrams[i].edgeCount &&
        shapes >= current.spec.diagrams[i].nodeCount,
      'Diagram has real edge paths and node geometry',
    )
    assert(arrows.length >= current.spec.diagrams[i].edgeCount, 'Actual edge arrow references')
    for (const ref of arrows)
      assert(svg.includes(`id="${ref[1]}"`), 'Arrow marker resolves in same SVG')
    return {
      viewBox,
      paths: paths.length,
      shapes,
      arrowReferences: arrows.length,
      svgSha256: sha(svg),
    }
  })
}
const markerCount = (text, marker) =>
  (
    text.match(
      new RegExp(
        '(?<![A-Za-z0-9_-])' + marker.replace(/[.*+?^${}()|[\]\\]/g, '\\$&') + '(?![A-Za-z0-9_-])',
        'g',
      ),
    ) || []
  ).length
async function artifact(file) {
  const bytes = await readFile(file),
    result = {
      file,
      sha256: sha(bytes),
      bytes: bytes.length,
      visualReview: 'pending independent review',
    }
  if (format === 'html') {
    const html = bytes.toString('utf8')
    result.diagrams = htmlGeometry(html)
    for (const marker of current.spec.bodyMarkersExactlyOnce)
      assert.equal(markerCount(html, marker), 1, marker)
    for (const marker of current.spec.forbiddenRevisionMarkers || [])
      assert(!html.includes(marker), 'No stale revision ' + marker)
  } else {
    assert(
      bytes.subarray(0, 5).equals(Buffer.from('%PDF-')) &&
        bytes.subarray(-1024).includes(Buffer.from('%%EOF')),
    )
    const info = execFileSync('pdfinfo', [file], { encoding: 'utf8' })
    await writeFile(file + '.info.txt', info)
    const dimensions = /^Page size:\s+([\d.]+) x ([\d.]+)/m.exec(info),
      expected = settings.landscape ? [792, 612] : [595.28, 841.89]
    assert(
      dimensions && expected.every((v, i) => Math.abs(v - Number(dimensions[i + 1])) < 2),
      'Actual paper size',
    )
    execFileSync('pdftoppm', ['-png', '-r', '96', file, file + '.page'], { timeout: 180000 })
    const extracted = execFileSync('pdftotext', ['-layout', file, '-'], { encoding: 'utf8' })
    await writeFile(file + '.text.txt', extracted)
    // These ASCII body/revision checks are independent of diagram/pixel fidelity.
    for (const marker of current.spec.bodyMarkersExactlyOnce)
      assert.equal(markerCount(extracted, marker), 1, marker)
    for (const marker of current.spec.forbiddenRevisionMarkers || [])
      assert(!extracted.includes(marker), 'No stale revision ' + marker)
    result.diagramFidelity =
      'NOT_ASSERTED: inspect all page rasters; extracted labels are not diagram evidence'
  }
  await json(path.basename(file) + '.artifact', result)
  return result
}

async function exportOnce(name) {
  const start = await begin(),
    file = path.join(output, name + (format === 'html' ? '.html' : '.pdf'))
  if (format === 'print-to-file') await printFile(file)
  else await saveDialog(file, await dialog('Choose output destination'))
  await app.until(
    async () => {
      const error = await app.execute(
        'return document.querySelector(arguments[0]+" [role=alert]")?.textContent',
        panel,
      )
      if (error) throw Error(error)
      return /已保存|已提交打印/.test(await panelText())
    },
    'output committed',
    180000,
  )
  await lifecycle.released(name)
  const after = await observe(name, {
    elapsedMs: Date.now() - start.started,
    artifact: await artifact(file),
    traces: (await traces()).slice(start.traceStart),
  })
  assert.equal(after.sourceSha256, current.spec.sourceSHA256)
  assert.match(
    after.panel,
    new RegExp(`版本 ${start.before.document.version} ·`),
    'Visible captured version',
  )
  assert.equal(after.document.version, start.before.document.version, 'No edits during output')
  await app.shot(name)
}
async function cancelThenEdit() {
  const start = await begin()
  let evidence
  if (stage === 'cancel-chooser') {
    const win = await dialog(
      format === 'print-to-file' ? 'Print document' : 'Choose output destination',
    )
    x('windowfocus', '--sync', win, 'key', '--clearmodifiers', 'Escape')
    evidence = { checkpoint: 'native-chooser', activeNativeRenderObserved: false }
  } else {
    const ready = await app.until(
      async () => {
        const text = await panelText()
        if (!text.includes('正在准备内容与图片')) {
          if (/请选择保存位置|正在准备打印|已保存|已提交打印/.test(text))
            throw Error(
              'NOT_EXERCISED: preparation finished before cancellation; no active-cancel pass',
            )
          return false
        }
        const recent = (await traces()).slice(start.traceStart)
        const active = recent
          .filter((r) => r.phase === 'render-request' && r.request?.target === 'export')
          .filter(
            (r) =>
              !recent.some(
                (s) =>
                  s.instanceId === r.instanceId &&
                  s.requestId === r.requestId &&
                  ['owner-result-accepted', 'deadline-observed', 'control-emitted'].includes(
                    s.phase,
                  ),
              ),
          )
        return {
          checkpoint: 'preparing-ui',
          activeNativeRenderObserved: active.length > 0,
          activeRequests: active,
        }
      },
      'observable preparation',
      10000,
    )
    evidence = { ...ready, clickStartedMonotonicNs: Number(process.hrtime.bigint()) }
    await click(command('cancel'))
    evidence.clickFinishedMonotonicNs = Number(process.hrtime.bigint())
  }
  await app.until(async () => (await panelText()).includes('已取消'), 'confirmed cancellation')
  await lifecycle.released('cancelled')
  await observe('cancelled', {
    cancellation: evidence,

        s.requestId === r.requestId &&

        s.monotonicNs >= evidence.clickFinishedMonotonicNs,
    ),
  )
  evidence.activeRenderCoverage = evidence.activeAtClickConfirmed
    ? 'observed interval includes click; inspect terminal trace for cancellation outcome'
    : 'NOT_ESTABLISHED; preparing UI or prior dispatch alone does not prove render active at click'
  await json('cancellation', evidence)
  assert(
    !(await readdir(output)).some((name) => /\.(html|pdf)$/.test(name)),
    'No published target after cancellation',
  )

      !(await app.probe()).dirty && sha(await readFile(sourcePath)) === next.spec.sourceSHA256,
    'real edit/save produces exact r2',
  )

  current = next
  await observe('edited-r2')
  await openPanel()
  await exportOnce('retry-r2')

// Host editing acceptance only: this does not change outline or any plugin implementation.
async function disableAll() {
  const before = await observe('before-disable-all')
  for (const id of ['milkup.outline', ...packages.map((p) => p.manifest.id)])
    assert(
      before.registry.some((r) => r.id === id && r.state === 'enabled'),
      'Expected enabled test subject ' + id,
    )
  const enabled = before.registry.filter((r) => r.state === 'enabled' || r.enabled)
  for (const item of enabled) {
    await ui.details(item.id)
    await ui.click(action('disable', item.id))
    await app.until(
      () =>
        app.execute(
          'return JSON.parse(localStorage.getItem("milkup.desktop.plugins.registry.v1")||"null")?.records.some(r=>r.manifest.id===arguments[0]&&r.state==="disabled"&&r.enabled===false)',
          item.id,
        ),
      'disabled ' + item.id,
    )
    await observe('disabled-' + item.id)
  }
  async function inactive(label) {
    const row = await observe(label)
    assert.deepEqual(
      row.registry.map((r) => r.id).sort(),
      before.registry.map((r) => r.id).sort(),
      'Installed identities preserved',
    )
    assert(
      row.registry.every((r) => r.enabled === false && r.state !== 'enabled'),
      'Every installed plugin inactive',
    )
    for (const item of enabled)
      assert.equal(row.registry.find((r) => r.id === item.id).state, 'disabled')
    assert.equal(
      await app.execute(
        'return document.querySelectorAll(\'[data-plugin-ui],[data-plugin-action="command"],[data-plugin-action="open-ui"]\').length',
      ),
      0,
      'No contributed UI or commands remain',
    )
    return row
  }
  await ui.closeMenu()
  await inactive('all-plugins-disabled')
  for (const mode of ['source', 'live']) {
    await app.focus('.milkup-input-proxy')
    await app.key(mode === 'source' ? 'ctrl+1' : 'ctrl+2')
    await app.until(async () => (await app.probe()).mode === mode, mode + ' ready')
    await app.focus('.milkup-input-proxy')
    await app.key('ctrl+Home')
    const prefix = mode === 'source' ? 'S' : 'L'
    const editedSha = sha(Buffer.concat([Buffer.from(prefix), current.bytes]))
    await app.text(prefix)
    await app.until(
      async () => (await app.probe()).dirty && (await app.probe()).cursor === 1,
      mode + ' real input',
    )
    await inactive(mode + '-typed')
    for (const [operation, key, expectedSha] of [
      ['undo', 'ctrl+z', current.spec.sourceSHA256],
      ['redo', 'ctrl+shift+z', editedSha],
      ['restore', 'ctrl+z', current.spec.sourceSHA256],
    ]) {
      await app.key(key)
      await app.key('ctrl+s')
      await app.until(
        async () => !(await app.probe()).dirty && sha(await readFile(sourcePath)) === expectedSha,
        mode + ' ' + operation + ' exact saved bytes',
      )
      await inactive(mode + '-' + operation + '-saved')
    }
    await app.shot('all-disabled-' + mode)
  }
  await json('disable-all', {
    passed: true,
    disabledIds: enabled.map((r) => r.id),
    previouslyInactive: before.registry.filter((r) => !r.enabled && r.state !== 'enabled'),
    sourceSha256: sha(await readFile(sourcePath)),
    modes: ['source', 'live'],
    scope: 'Acceptance through product UI; no outline or plugin implementation changes',
  })
}
